'変数未宣言エラー設定ON
Option Explicit
'------------------------------------------------------------
'概要 : Sheet[配色テーマ集]を作成する
'引数 : -
'戻値 : -
'備考 : -
'------------------------------------------------------------
Public Sub CreateSheet_ThemeColorBook()
    '処理中描画OFF設定
    Call SettingRenderingProcess(False)

    Dim ws As Worksheet   'Sheet
    Dim rowIdx As Long    '行
    Dim colIdx As Long    '列
    Dim endRowIdx As Long '終了行
    Dim endColIdx As Long '終了列

    '●テーマ数
    Dim themeCnt As Long
    '●テーマリスト/行
    Dim themeCol As Collection
    Dim themeArr As Variant
    '●カテゴリ名(前/現在)
    Dim preCatNm As String
    Dim crntCatNm As String

    '初期化
    rowIdx = 1
    colIdx = -12
    themeCnt = 1

    '全テーマ取得
    Set themeCol = GetAllTheme()

    'Sheet[設定]作成(存在しない場合)
    If Not IsExistsSheet("設定") Then
        CreateSheet_ThemeParamSetting
    End If

    'Sheet作成 & Sheet名セット
    Set ws = ThisWorkbook.Sheets.Add(Before:=ThisWorkbook.Worksheets(1))
    ws.Name = ChangeUniqueSheetNm("配色テーマ集")

    '書込
    For Each themeArr In themeCol
        crntCatNm = themeArr(0)
        'カテゴリ名が変わった場合
        If preCatNm <> crntCatNm Then
            'タイトル書込
            rowIdx = 1
            colIdx = colIdx + 13
            Call WriteTitle(ws, rowIdx, colIdx, themeArr)
            rowIdx = 2
        End If
        'テーマ書込
        Call WriteOneTheme(ws, rowIdx, colIdx, themeArr, themeCnt)
        themeCnt = themeCnt + 1
        rowIdx = rowIdx + 6
        preCatNm = crntCatNm
    Next

    '終了(行/列) 取得
    endRowIdx = ws.Cells.SpecialCells(xlCellTypeLastCell).Row
    endColIdx = ws.Cells.SpecialCells(xlCellTypeLastCell).Column

    'Sheet設定
    ws.Activate
    With ws
        .Cells.Font.Name = "Noto Sans JP"       'フォント
        .Rows(1).Font.Size = 20                 '文字サイズ
        .Rows(1).Font.Bold = True               '太字
        .Cells.VerticalAlignment = xlTop        '文字揃え
        .Cells.HorizontalAlignment = xlLeft
        .Rows(1).HorizontalAlignment = xlCenter
        .Rows(1).RowHeight = 30                 '行幅
        .Columns.ColumnWidth = 4                '列幅
    End With
    ActiveWindow.DisplayGridlines = False       '目盛線
    ActiveWindow.Zoom = 85                      '表示倍率
    'グループ化
    rowIdx = 4
    Do While True
        ws.Rows(rowIdx & ":" & (rowIdx + 3)).Group
        rowIdx = rowIdx + 6
        If endRowIdx <= rowIdx Then Exit Do
    Loop
    'グループ閉じる
    ws.Outline.ShowLevels RowLevels:=1, ColumnLevels:=1
    '列幅
    colIdx = 1
    Do While True
        ws.Columns(colIdx).ColumnWidth = 2
        colIdx = colIdx + 13
        If endColIdx <= colIdx Then Exit Do
    Loop
    'ウィンドウ枠の固定
    ws.range("A2").Select
    ActiveWindow.FreezePanes = True

    'Sheet[比較]作成(存在しない場合)
    If Not IsExistsSheet("比較") Then
        CreateSheet_ThemeCompare
    End If

    '処理中描画ON設定
    Call SettingRenderingProcess(True)
End Sub
'------------------------------------------------------------
'概要 : 処理中の画面描画を設定する
'引数 : flg(True:ON / False:OFF)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub SettingRenderingProcess(ByVal flg As Boolean)
    Application.ScreenUpdating = flg '画面再描画
    Application.DisplayAlerts = flg  '確認ダイアログ表示
    Application.EnableEvents = flg   'イベント発火
    '再計算モード(セル値が変わるたびに再計算)
    If flg Then
        Application.Calculation = xlCalculationAutomatic '自動
    Else
        Application.Calculation = xlCalculationManual '手動
    End If
End Sub
'------------------------------------------------------------
'概要 : Sheet名のSheetが存在するか判定する
'引数 : wsNm : Sheet名
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsSheet(ByVal wsNm As String) As Boolean
    Dim ws As Worksheet
    For Each ws In Sheets
        If ws.Name = wsNm Then
            IsExistsSheet = True
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : 重複しないSheet名となるまで、末尾に連番を付与する
'     : (同じ名前のSheetが存在しない場合、そのまま返す)
'     : (同じ名前のSheetが存在する場合、2からの連番を末尾に付与し、重複しないSheet名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : wsNm : Sheet名
'戻値 : Sheet名
'備考 : -
'------------------------------------------------------------
Private Function ChangeUniqueSheetNm(ByVal wsNm As String) As String
    Dim i As Long
    Dim uniqueNo As String
    '存在しない場合
    If Not IsExistsSheet(wsNm) Then
        ChangeUniqueSheetNm = wsNm
        Exit Function
    End If
    '存在する場合
    For i = 2 To 1000
        '末尾付与
        uniqueNo = " (" & i & ")"
        '存在しない場合
        If Not IsExistsSheet(wsNm & uniqueNo) Then
            ChangeUniqueSheetNm = wsNm & uniqueNo
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : 列番号から列名(アルファベット)を取得する
'     : (範囲外の数値の場合、空を返す)
'引数 : colIdx : 列番号
'戻値 : 列名(アルファベット)
'備考 : -
'------------------------------------------------------------
Private Function GetColNm(ByVal colIdx As Long) As String
    Dim result As String
    '範囲外の場合
    If colIdx < 1 Or Application.Columns.Count < colIdx Then
        GetColNm = ""
        Exit Function
    End If
    Do While 0 < colIdx
        colIdx = colIdx - 1
        result = Chr$(65 + (colIdx Mod 26)) & result
        colIdx = colIdx \ 26
    Loop
    GetColNm = result
End Function
'------------------------------------------------------------
'概要 : 列名(アルファベット)から列番号を取得する
'     : (範囲外の列名の場合、0を返す)
'引数 : colNm : 列名(アルファベット)
'戻値 : 列番号
'備考 : -
'------------------------------------------------------------
Private Function GetColIdx(ByVal colNm As String) As Long
    Dim i As Long
    Dim result As Long
    Dim baseA As Long: baseA = Asc("A") - 1
    Dim colNmLen As Long: colNmLen = Len(UCase$(colNm))
    Dim ch As Long
    For i = 1 To colNmLen
        ch = Asc(Mid$(colNm, i, 1)) - baseA
        '範囲外の場合
        If ch < 1 Or 26 < ch Then
            GetColIdx = 0
            Exit Function
        End If
        result = result * 26 + ch
    Next
    GetColIdx = result
End Function

'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub WriteTitle(ByRef ws As Worksheet, ByVal rowIdx As Long, ByVal colIdx As Long, ByVal themeArr As Variant)
    '●範囲
    Dim rng As range: Set rng = ws.range("A" & rowIdx & ":L" & rowIdx).Offset(0, colIdx)
    '書込
    ws.Cells(rowIdx, colIdx).Offset(0, 1).Value = themeArr(0)
    'セル結合
    rng.Merge
    '文字色
    rng.Font.Color = RGB(themeArr(1), themeArr(2), themeArr(3))
End Sub

'引数 : -

Private Sub CreateSheet_ThemeCompare()
    Dim ws As Worksheet   'Sheet
    Dim endRowIdx As Long '終了行
    Dim endColIdx As Long '終了列
    '●検索番号セル番地(行/列)
    Dim searchNoRowList As Variant
    Dim searchNoColList As Variant
    Dim searchNoRow As Variant
    Dim searchNoCol As Variant
    '●罫線位置区分
    Dim borderType As Variant
    '●図形/サイズ[横/縦]
    Dim shp As Shape
    Dim shpWidth As Long: shpWidth = 2
    Dim shpHeight As Long: shpHeight = 1

    '検索番号セル番地 セット
    searchNoRowList = Array(3, 8, 13, 18, 23)
    searchNoColList = Array("C", "P", "AC")

    'Sheet作成 & Sheet名セット
    Set ws = ThisWorkbook.Sheets.Add(Before:=ThisWorkbook.Worksheets(1))
    ws.Name = ChangeUniqueSheetNm("比較")

    '書込
    For Each searchNoRow In searchNoRowList
        For Each searchNoCol In searchNoColList
            ws.Cells(searchNoRow, GetColIdx(CStr(searchNoCol)) - 1).Value = "番号"
            'いくつかサンプル番号出力
            If searchNoRow < 10 Then ws.Cells(searchNoRow, searchNoCol).Value = searchNoRow * GetColIdx(searchNoCol)
        Next
    Next

    '終了(行/列) 取得
    endRowIdx = ws.Cells.SpecialCells(xlCellTypeLastCell).Row + 3
    endColIdx = ws.Cells.SpecialCells(xlCellTypeLastCell).Column

    'Sheet設定
    ws.Activate
    With ws
        'フォント
        .Cells.Font.Name = "Noto Sans JP"
        '文字揃え
        .Cells.VerticalAlignment = xlTop
        .Cells.HorizontalAlignment = xlLeft
        .Rows(1).HorizontalAlignment = xlCenter
        '行幅
        .Rows(1).RowHeight = 36
        .Rows(2).RowHeight = 6
        '列幅
        .Columns.ColumnWidth = 4
        .Columns("A").ColumnWidth = 2
        .Columns("N").ColumnWidth = 2
        .Columns("AA").ColumnWidth = 2
        '文字揃え/行幅/罫線
        For Each searchNoRow In searchNoRowList
            .Rows(1).HorizontalAlignment = xlCenter
            .Rows(searchNoRow + 1).RowHeight = 10
            .Rows(searchNoRow + 3).RowHeight = 40
            .Rows(searchNoRow + 4).RowHeight = 10
            For Each searchNoCol In searchNoColList
                .Cells(searchNoRow, searchNoCol).HorizontalAlignment = xlRight
                For Each borderType In Array(xlEdgeTop, xlEdgeRight, xlEdgeBottom, xlEdgeLeft)
                    .Cells(searchNoRow, searchNoCol).Borders(borderType).LineStyle = xlContinuous
                    .Cells(searchNoRow, searchNoCol).Borders(borderType).Weight = xlThin
                Next
            Next
        Next
        'グループ化
        .Rows("1:2").Group
    End With
    '目盛線
    ActiveWindow.DisplayGridlines = False
    'ウィンドウ枠の固定
    ws.range("A2").Select
    ActiveWindow.FreezePanes = True

    'マクロボタン 作成
    Set shp = ws.Shapes.AddShape(msoShapeBevel, 0, 0, Application.CentimetersToPoints(shpWidth), Application.CentimetersToPoints(shpHeight))
    With shp
        .Fill.ForeColor.RGB = RGB(91, 155, 213)                            '#5B9BD5 塗りつぶし色
        .Line.ForeColor.RGB = RGB(91, 155, 213)                            '#5B9BD5 枠線色

        .TextFrame2.TextRange.Font.NameFarEast = "Noto Sans JP"            'フォント(JP)
        .TextFrame2.TextRange.Font.Size = 16                               '文字サイズ
        .TextFrame2.TextRange.Font.Bold = msoTrue                          '太字
        .TextFrame2.TextRange.Font.Fill.ForeColor.RGB = RGB(255, 255, 255) '#FFFFFF 文字色
        .TextFrame2.VerticalAnchor = msoAnchorMiddle                       '文字揃え(横)
        .TextFrame2.TextRange.ParagraphFormat.Alignment = msoAlignCenter   '文字揃え(縦)
        .OnAction = "Btn_RefreshCompareSheet"                              'マクロ登録
        .Top = 4                                                           '配置場所(縦)
        .Left = 16                                                         '配置場所(横)
    End With

    'Sheet更新
    Btn_RefreshCompareSheet
End Sub
'------------------------------------------------------------
'概要 : Sheet[比較]の表示を更新する
'     : (番号に一致するテーマを検索し、存在する場合、コピーする)
'引数 : -
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub Btn_RefreshCompareSheet()
    '処理中描画OFF設定
    Call SettingRenderingProcess(False)

    '●現在Sheet/検索先Sheet
    Dim ws As Worksheet: Set ws = ActiveSheet
    Dim searchWs As Worksheet: Set searchWs = ThisWorkbook.Sheets("配色テーマ集")
    '●終了行/列
    Dim endRowIdx As Long
    Dim endColIdx As Long
    '●検索一致セル
    Dim foundCell As range
    '●範囲 [コピー元/先]
    Dim srcRange As range, destRange As range
    '●検索番号セル番地 行/列
    Dim searchNoRowList As Variant
    Dim searchNoColList As Variant
    Dim searchNoRow As Variant
    Dim searchNoCol As Variant
    Dim searchText As String

    '検索番号セル番地 セット
    searchNoRowList = Array(3, 8, 13, 18, 23)
    searchNoColList = Array("C", "P", "AC")

    '終了(行/列) 取得
    endRowIdx = searchWs.Cells.SpecialCells(xlCellTypeLastCell).Row
    endColIdx = searchWs.Cells.SpecialCells(xlCellTypeLastCell).Column

    '検索/コピー
    For Each searchNoRow In searchNoRowList
        For Each searchNoCol In searchNoColList
            '検索値
            searchText = ws.Cells(searchNoRow, searchNoCol).Value2 & "："
            '一致セル検索(前方一致)
            Set foundCell = searchWs.range("A2:" & GetColNm(endColIdx) & endRowIdx).Find(What:=searchText & "*", LookIn:=xlValues, LookAt:=xlWhole)
            '存在しない場合
            If foundCell Is Nothing Then
                '既存テーマ削除
                ws.range(GetColNm(GetColIdx(searchNoCol) - 1) & (searchNoRow + 2) & ":" & GetColNm(GetColIdx(searchNoCol) + 10) & (searchNoRow + 2)).Value = ""
                ws.range(GetColNm(GetColIdx(searchNoCol) - 1) & (searchNoRow + 3) & ":" & GetColNm(GetColIdx(searchNoCol) + 10) & (searchNoRow + 3)).Interior.ColorIndex = xlNone
                '書込
                ws.Cells(searchNoRow, GetColIdx(searchNoCol) + 2).Value = "該当なし"
            Else
                '該当なし削除
                ws.Cells(searchNoRow, GetColIdx(searchNoCol) + 2).Value = ""
                'コピー(元/先) 指定
                Set srcRange = foundCell.Resize(2, 12)
                Set destRange = ws.range(searchNoCol & searchNoRow).Offset(2, -1).Resize(2, 12)
                'コピー
                srcRange.Copy
                destRange.PasteSpecial Paste:=xlPasteAll
                Application.CutCopyMode = False
            End If
        Next
    Next

    '選択
    ws.range("C3").Select

    '処理中描画ON設定
    Call SettingRenderingProcess(True)
End Sub

'------------------------------------------------------------
'概要 : Sheet[設定]にセットされた背景色からRGBの値と色コードをセットする
'引数 : -
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub Btn_SetRgbParam()
    '処理中描画OFF設定
    Call SettingRenderingProcess(False)

    'Sheet/行/列1/列2/終了行
    Dim ws As Worksheet: Set ws = ActiveSheet
    Dim rowIdx As Long
    Dim colIdx1 As Long
    Dim colIdx2 As Long
    Dim endRowIdx As Long

    '●色値/RGB値/色コード
    Dim colorVal As Long
    Dim r As Long, g As Long, b As Long
    Dim colorCd As String

    '最下行 取得
    endRowIdx = ws.Cells.SpecialCells(xlCellTypeLastCell).Row

    '書込
    For rowIdx = 4 To endRowIdx
        '========== カテゴリ色 ==========
        colorVal = ws.Cells(rowIdx, "C").Interior.Color
        '赤/緑/青/色コード 取得
        r = colorVal Mod 256
        g = (colorVal \ 256) Mod 256
        b = (colorVal \ 65536) Mod 256
        colorCd = "#" & Right("000000" & Hex(r) & Hex(g) & Hex(b), 6)
        '書込
        ws.Cells(rowIdx, "I").Value = r
        ws.Cells(rowIdx, "J").Value = g
        ws.Cells(rowIdx, "K").Value = b
        ws.Cells(rowIdx, "L").Value = colorCd
        '========== 色1,2,3 ==========
        colIdx2 = 1
        For colIdx1 = 1 To 3
            colorVal = ws.Cells(rowIdx, colIdx1 + 5).Interior.Color
            r = colorVal Mod 256
            g = (colorVal \ 256) Mod 256
            b = (colorVal \ 65536) Mod 256
            colorCd = "#" & Right("000000" & Hex(r) & Hex(g) & Hex(b), 6)
            ws.Cells(rowIdx, colIdx2 + 12).Value = r
            ws.Cells(rowIdx, colIdx2 + 13).Value = g
            ws.Cells(rowIdx, colIdx2 + 14).Value = b
            ws.Cells(rowIdx, colIdx2 + 15).Value = colorCd
            colIdx2 = colIdx2 + 4
        Next
    Next

    '処理中描画ON設定
    Call SettingRenderingProcess(True)
End Sub
