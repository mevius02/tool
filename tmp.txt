Private Sub AnalyzeJclFolder(ByVal ws As Worksheet, _
ByRef rowNo As Long)

Dim folderPath As String
Dim fileName As String

folderPath = ThisWorkbook.Path & "\PGM\"

fileName = Dir(folderPath & "*.jcl")

Do While fileName <> ""

    AnalyzeJclFile _
        folderPath & fileName, _
        ws, _
        rowNo

    fileName = Dir()

Loop
End Sub

Private Sub AnalyzeJclFile(ByVal filePath As String, _
ByVal ws As Worksheet, _
ByRef rowNo As Long)

Dim ff As Integer
Dim lineText As String

Dim jobName As String
Dim stepName As String
Dim pgmName As String
Dim procName As String

ff = FreeFile

Open filePath For Input As #ff

Do Until EOF(ff)

    Line Input #ff, lineText

    lineText = Trim(lineText)

    If Left(lineText, 2) <> "//" Then
        GoTo ContinueLoop
    End If

    'JOB判定
    If InStr(lineText, " JOB ") > 0 Then

        jobName = Replace(Split(lineText)(0), "//", "")

    End If

    'EXEC判定
    If InStr(lineText, " EXEC ") > 0 Then

        stepName = Replace(Split(lineText)(0), "//", "")

        pgmName = ""
        procName = ""

        If InStr(lineText, "PGM=") > 0 Then
            pgmName = GetParameterValue(lineText, "PGM")
        End If

        If InStr(lineText, "PROC=") > 0 Then
            procName = GetParameterValue(lineText, "PROC")
        End If

        ws.Cells(rowNo, 1).Value = jobName
        ws.Cells(rowNo, 2).Value = stepName
        ws.Cells(rowNo, 3).Value = pgmName
        ws.Cells(rowNo, 4).Value = procName

        rowNo = rowNo + 1

    End If
ContinueLoop:

Loop

Close #ff
End Sub

共通関数

Private Function CreateOrClearSheet( _
    sheetName As String) As Worksheet

    Dim ws As Worksheet

    On Error Resume Next
    Set ws = Worksheets(sheetName)
    On Error GoTo 0

    If ws Is Nothing Then

        Set ws = Worksheets.Add

        ws.Name = sheetName

    Else

        ws.Cells.Clear

    End If

    Set CreateOrClearSheet = ws

End Function


Private Function GetParameterValue( _
    lineText As String, _
    parameterName As String) As String

    Dim pos As Long
    Dim tmp As String

    pos = InStr(lineText, parameterName & "=")

    If pos = 0 Then Exit Function

    tmp = Mid(lineText, pos + Len(parameterName) + 1)

    If InStr(tmp, ",") > 0 Then
        tmp = Left(tmp, InStr(tmp, ",") - 1)
    End If

    GetParameterValue = Trim(tmp)

End Function

ファイル	抽出可能な情報

.cob	PROGRAM-ID・ファイル定義・データ項目・段落構造・CALL先・ファイルI/O操作・PERFORM構造
.cpy	共通レコードレイアウト・共通データ項目定義
作成できる仕様書（8種類）
✅ ほぼ自動生成できるもの
1. プログラム一覧表
PROGRAM-ID / ファイル名 / COPYブック使用一覧 / CALL関係
2. JCLジョブ仕様書

5. コピーブック使用一覧

🔶 ある程度自動生成できるもの（精度に限界あり）
6. プログラム間呼び出し関係図（テキスト表形式）

※ 段落名が意味のある命名なら有効。PARA-001のような名前は推測不能

結論から言うと、あの7シートは仕様書ではなく、仕様書を作るための地図です。ただし、この地図があるかどうかで作業効率が全く変わります。

Javaで例えると一番わかりやすい
Springプロジェクトで「仕様書なし・ドキュメントなし」の状態を想像してください。

こんな状況↓
src/
├── service/   （何十個ものServiceクラス）
├── mapper/    （何十個ものMapperクラス）
└── batch/     （何十個ものJobクラス）
この状況で「このシステムは何をしてるの？」と聞かれたら、まずやることは：

クラス一覧を把握する
どのクラスがどのクラスを呼んでるか把握する
どのテーブル・ファイルを使ってるか把握する
COBOLのケースも全く同じで、あの7シートはこの「把握する」フェーズを自動化するものです。

7シートの正体と価値
シート	正体	Javaで例えると
プログラム一覧	存在確認	クラス一覧
JCLジョブ仕様	実行設定	application.yml + cronの定義
ファイル割当一覧	入出力の配線	@Valueで読んでるファイルパス一覧
ファイルレイアウト	データ構造	DTOクラスのフィールド定義
コピーブック使用一覧	共通クラスの依存関係	どのServiceが共通Utilを使ってるか
呼び出し関係	依存グラフ	@Autowiredの依存関係マップ
処理フロー	メソッド一覧	クラス内のメソッド名一覧
「地図」があると何が変わるか
地図なし（今の状態）
.cobファイルを1個開く
  → COPY文でcpyファイルを参照してると判明
  → そのcpyを開く
  → CALLで別プログラム呼んでると判明
  → その.cobを開く
  → JCLでどう動いてるか気になる
  → JCLを探す...
ファイルを行ったり来たりしながら、頭の中で繋いでいく地獄

地図あり（7シート完成後）
シート①でプログラム全体像を把握
  → シート⑥でCALL関係を確認
  → シート④でデータ構造を確認
  → あとはコードを読んで「業務の意図」だけ人間が補足する
つまりこういうこと
7シート（地図）
  ＋
人間が補足する「なぜそうしてるか」
  ↓
  = 本当の仕様書
7シートは仕様書の骨格であって、肉付けは人間がやる必要があります。ただその骨格を手作業で作ると、大規模システムだと数週間〜数ヶ月かかる作業が、VBAなら数分〜数時間で完成するのが最大のメリットです。

正直なところ
「コードから自動生成した仕様書です」とは言えない。「仕様書を作るために必要な下調べを自動化したもの」が正確。

ただ、COBOLの大規模システムは何百本というプログラムがあるのが普通なので、この下調べを自動化できるだけで実務上の価値は十分あります。

VBAを作る方向で進めますか？

じゃあ7sheetを作成するVBAを作って欲しい。ただし、VBA処理の最初に、7sheetをそれぞれ1sheet単位で作るか作らないかをbooleanのフラグで管理したい。例えばcreateSheet1Flgがtrueなら1sheet目を作る。falseなら作らないとして、実行ユーザーが作りたいsheetだけ作るようにしたい。なので各sheetの作成処理が干渉しない、独立した処理にしたい。まずはsheet1を作成するVBAを教えて

    GetCpyFolderPath = ThisWorkbook.Path & "\CPY\"

'==================================================
' シート初期化（既存削除→末尾に新規作成）
'==================================================
Public Sub InitSheet(sheetName As String, ByRef ws As Worksheet)
    Dim existWs As Worksheet
    For Each existWs In ThisWorkbook.Worksheets
        If existWs.Name = sheetName Then
            Application.DisplayAlerts = False
            existWs.Delete
            Application.DisplayAlerts = True
            Exit For
        End If
    Next existWs
    Set ws = ThisWorkbook.Worksheets.Add( _
        After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
    ws.Name = sheetName
End Sub

'==================================================
' 配列内重複チェック
'==================================================
Public Function IsInArray(targetVal As String, _
                           arr() As String, _
                           arrCount As Integer) As Boolean
    Dim i As Integer
    For i = 0 To arrCount - 1
        If arr(i) = targetVal Then
            IsInArray = True
            Exit Function
        End If
    Next i
    IsInArray = False
End Function

'==================================================
' COBOLコメント行判定
' → 7列目が * か / ならコメント行
'==================================================
Public Function IsCobolCommentLine(line As String) As Boolean
    If Len(line) < 7 Then
        IsCobolCommentLine = False
        Exit Function
    End If
    Dim indicator As String
    indicator = Mid(line, 7, 1)
    IsCobolCommentLine = (indicator = "*" Or indicator = "/")
End Function

'==================================================
' ヘッダー行スタイル（紺背景・白文字・太字）

Public Sub SetHeaderStyle(ws As Worksheet, lastCol As Long)
    With ws.Range(ws.Cells(1, 1), ws.Cells(1, lastCol))
        .Interior.Color = RGB(31, 78, 121)
        .Font.Color = RGB(255, 255, 255)
        .Font.Bold = True
        .HorizontalAlignment = xlCenter
    End With
End Sub

Option Explicit

'==================================================
' Sheet1: プログラム一覧 作成
' 出力列: No / ファイル名 / PROGRAM-ID / 使用COPYブック / CALL先
'==================================================
Public Sub CreateSheet1_ProgramList()

    Dim ws       As Worksheet
    Dim cobFolder As String
    Dim fileName  As String
    Dim rowIdx    As Long
    Dim programId As String
    Dim copyList  As String
    Dim callList  As String

    ' シート初期化
    Call InitSheet("Sheet1_プログラム一覧", ws)

    ' ヘッダー設定
    ws.Cells(1, 1).Value = "No"
    ws.Cells(1, 2).Value = "ファイル名"
    ws.Cells(1, 3).Value = "PROGRAM-ID"
    ws.Cells(1, 4).Value = "使用COPYブック"
    ws.Cells(1, 5).Value = "CALL先プログラム"
    Call SetHeaderStyle(ws, 5)

    ' フォルダ存在確認
    cobFolder = GetCobFolderPath()
    If Dir(cobFolder, vbDirectory) = "" Then
        MsgBox "COBフォルダが見つかりません。" & vbCrLf & cobFolder, vbExclamation
        Exit Sub
    End If

    ' COBファイルを1件ずつ処理
    fileName = Dir(cobFolder & "*.cob")
    rowIdx = 2

    Do While fileName <> ""

        Call ParseCobFile_Sheet1(cobFolder & fileName, programId, copyList, callList)

        ws.Cells(rowIdx, 1).Value = rowIdx - 1
        ws.Cells(rowIdx, 2).Value = fileName
        ws.Cells(rowIdx, 3).Value = programId
        ws.Cells(rowIdx, 4).Value = copyList
        ws.Cells(rowIdx, 5).Value = callList

        ' 偶数行に薄いグレー背景（視認性向上）
        If rowIdx Mod 2 = 0 Then
            ws.Range(ws.Cells(rowIdx, 1), ws.Cells(rowIdx, 5)) _
              .Interior.Color = RGB(242, 242, 242)
        End If

        rowIdx = rowIdx + 1
        fileName = Dir()
    Loop

    ' 列幅調整
    ws.Columns("A:C").AutoFit
    ws.Columns("D:E").ColumnWidth = 45

    ' 枠線
    With ws.Range(ws.Cells(1, 1), ws.Cells(rowIdx - 1, 5)).Borders
        .LineStyle = xlContinuous
        .Color = RGB(180, 180, 180)
    End With

    ws.Cells(1, 1).Select

    MsgBox "Sheet1（プログラム一覧）完了。" & vbCrLf & _
           "件数: " & (rowIdx - 2) & " 件", vbInformation, "COBOL仕様書生成"
End Sub

'==================================================
' COBファイル解析（Sheet1専用）
'==================================================
Private Sub ParseCobFile_Sheet1(filePath As String, _
                                 ByRef programId As String, _
                                 ByRef copyList  As String, _
                                 ByRef callList  As String)
    Dim fileNum     As Integer
    Dim line        As String
    Dim upperLine   As String
    Dim copyBooks() As String
    Dim callProgs() As String
    Dim copyCount   As Integer
    Dim callCount   As Integer
    Dim tmpName     As String

    ReDim copyBooks(0)
    ReDim callProgs(0)
    programId = ""
    copyCount = 0
    callCount = 0

    fileNum = FreeFile()
    On Error GoTo FileOpenError
    Open filePath For Input As #fileNum
    On Error GoTo 0

    Do While Not EOF(fileNum)
        Line Input #fileNum, line

        ' コメント行スキップ
        If IsCobolCommentLine(line) Then GoTo ContinueLoop

        ' 73列目以降はシーケンス番号領域なので切り捨て
        If Len(line) > 72 Then line = Left(line, 72)

        upperLine = UCase(line)

        ' ---- PROGRAM-ID ----
        If programId = "" And InStr(upperLine, "PROGRAM-ID") > 0 Then
            programId = S1_ExtractProgramId(upperLine)
        End If

        ' ---- COPY文 ----
        If InStr(upperLine, "COPY ") > 0 Then
            tmpName = S1_ExtractCopyName(upperLine)
            If tmpName <> "" Then
                If Not IsInArray(tmpName, copyBooks, copyCount) Then
                    ReDim Preserve copyBooks(copyCount)
                    copyBooks(copyCount) = tmpName
                    copyCount = copyCount + 1
                End If
            End If
        End If

        ' ---- CALL文 ----
        If InStr(upperLine, "CALL ") > 0 Then
            tmpName = S1_ExtractCallName(upperLine)
            If tmpName <> "" Then
                If Not IsInArray(tmpName, callProgs, callCount) Then
                    ReDim Preserve callProgs(callCount)
                    callProgs(callCount) = tmpName
                    callCount = callCount + 1
                End If
            End If
        End If

        ContinueLoop:
    Loop

    Close #fileNum

    ' 結果を文字列化
    copyList = IIf(copyCount > 0, Join(copyBooks, ", "), "（なし）")
    callList = IIf(callCount > 0, Join(callProgs, ", "), "（なし）")
    Exit Sub

FileOpenError:
    programId = "★読取エラー"
    copyList  = "★読取エラー"
    callList  = "★読取エラー"
End Sub

'--------------------------------------------------
' PROGRAM-ID 値を抽出
' 例: "       PROGRAM-ID. MYPROG01." → "MYPROG01"
'--------------------------------------------------
Private Function S1_ExtractProgramId(upperLine As String) As String
    Dim pos      As Integer
    Dim after    As String
    Dim result   As String

    pos = InStr(upperLine, "PROGRAM-ID")
    If pos = 0 Then Exit Function

    after = Trim(Mid(upperLine, pos + Len("PROGRAM-ID")))

    ' 先頭ピリオド除去
    If Left(after, 1) = "." Then after = Trim(Mid(after, 2))
    If after = "" Then Exit Function

    ' 最初のトークンを取得
    result = Split(after, " ")(0)

    ' 末尾ピリオド除去
    If Right(result, 1) = "." Then result = Left(result, Len(result) - 1)

    S1_ExtractProgramId = Trim(result)
End Function

'--------------------------------------------------
' COPY文のコピーブック名を抽出
' 例: "           COPY COPYBK01."             → "COPYBK01"
'      "           COPY COPYBK02 REPLACING..." → "COPYBK02"
'--------------------------------------------------
Private Function S1_ExtractCopyName(upperLine As String) As String
    Dim pos    As Integer
    Dim after  As String
    Dim result As String

    pos = InStr(upperLine, "COPY ")
    If pos = 0 Then Exit Function

    after = Trim(Mid(upperLine, pos + 5))
    If after = "" Then Exit Function

    result = Split(after, " ")(0)
    If Right(result, 1) = "." Then result = Left(result, Len(result) - 1)

    S1_ExtractCopyName = Trim(result)
End Function

'--------------------------------------------------
' CALL文のプログラム名を抽出
' 静的: CALL 'SUBPROG1'  → "SUBPROG1"
' 動的: CALL WK-PROG     → "[動的:WK-PROG]"
'--------------------------------------------------
Private Function S1_ExtractCallName(upperLine As String) As String
    Dim pos      As Integer
    Dim after    As String
    Dim result   As String
    Dim endPos   As Integer

    pos = InStr(upperLine, "CALL ")
    If pos = 0 Then Exit Function

    after = Trim(Mid(upperLine, pos + 5))
    If after = "" Then Exit Function

    If Left(after, 1) = "'" Or Left(after, 1) = Chr(34) Then
        ' 静的CALL（リテラル）
        endPos = InStr(2, after, Left(after, 1))
        If endPos > 0 Then
            result = Mid(after, 2, endPos - 2)
        Else
            result = Mid(after, 2)
        End If
    Else
        ' 動的CALL（変数）
        result = "[動的:" & Split(after, " ")(0) & "]"
    End If

    S1_ExtractCallName = Trim(result)
End Function
