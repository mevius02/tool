vbOption Explicit

'==================================================
' Sheet2: JCLジョブ仕様 作成
' 出力列: No / JCLファイル名 / ジョブ名 / ステップ名 /
'         実行PGM・PROC / COND条件 / 使用DD名一覧
' 1行 = 1ステップ
'==================================================
Public Sub CreateSheet2_JclJobSpec()

    Dim ws        As Worksheet
    Dim jclFolder As String
    Dim fileName  As String
    Dim rowIdx    As Long

    Call InitSheet("Sheet2_JCLジョブ仕様", ws)

    ws.Cells(1, 1).Value = "No"
    ws.Cells(1, 2).Value = "JCLファイル名"
    ws.Cells(1, 3).Value = "ジョブ名"
    ws.Cells(1, 4).Value = "ステップ名"
    ws.Cells(1, 5).Value = "実行PGM / PROC"
    ws.Cells(1, 6).Value = "COND条件"
    ws.Cells(1, 7).Value = "使用DD名一覧"
    Call SetHeaderStyle(ws, 7)

    jclFolder = GetJclFolderPath()
    If Dir(jclFolder, vbDirectory) = "" Then
        MsgBox "JCLフォルダが見つかりません。" & vbCrLf & jclFolder, vbExclamation
        Exit Sub
    End If

    fileName = Dir(jclFolder & "*.jcl")
    rowIdx = 2

    Do While fileName <> ""
        Call ParseJclFile_Sheet2(jclFolder & fileName, fileName, ws, rowIdx)
        fileName = Dir()
    Loop

    ws.Columns("A:F").AutoFit
    ws.Columns("G").ColumnWidth = 50

    If rowIdx > 2 Then
        With ws.Range(ws.Cells(1, 1), ws.Cells(rowIdx - 1, 7)).Borders
            .LineStyle = xlContinuous
            .Color = RGB(180, 180, 180)
        End With
    End If

    ws.Cells(1, 1).Select

    MsgBox "Sheet2（JCLジョブ仕様）完了。" & vbCrLf & _
           "ステップ数: " & (rowIdx - 2) & " 件", vbInformation, "COBOL仕様書生成"
End Sub

'==================================================
' JCLファイル解析（1ファイル → 複数行書き込み）
'
' JCLの構造:
'   //JOBNAME  JOB  ...
'   //STEP01   EXEC PGM=PROG01,COND=(0,NE)
'   //INFILE   DD   DSN=...
'   //OUTFILE  DD   DSN=...
'   //STEP02   EXEC PGM=PROG02
'   ...
' → EXECを検出したら「前のステップ」を書き込み、新ステップを初期化
'==================================================
Private Sub ParseJclFile_Sheet2(filePath As String, _
                                  fileName As String, _
                                  ws As Worksheet, _
                                  ByRef rowIdx As Long)
    Dim fileNum   As Integer
    Dim line      As String
    Dim jobName   As String
    Dim stepName  As String
    Dim pgmOrProc As String
    Dim condValue As String
    Dim ddNames() As String
    Dim ddCount   As Integer
    Dim stepNo    As Integer

    jobName = ""
    stepNo  = 0
    ReDim ddNames(0)
    ddCount = 0

    fileNum = FreeFile()
    On Error GoTo FileOpenError
    Open filePath For Input As #fileNum
    On Error GoTo 0

    Do While Not EOF(fileNum)
        Line Input #fileNum, line

        ' "//"で始まらない行はスキップ（インストリームデータ等）
        If Left(line, 2) <> "//" Then GoTo NextJclLine

        ' コメント行スキップ（//*）
        If Left(line, 3) = "//*" Then GoTo NextJclLine

        ' 継続行スキップ（"//"の直後がスペース = 名前フィールドなし）
        If Len(line) >= 3 Then
            If Mid(line, 3, 1) = " " Then GoTo NextJclLine
        End If

        ' JCL行をname/verb/operandsに分解
        Dim cardName     As String
        Dim cardVerb     As String
        Dim cardOperands As String
        Call S2_ParseJclLine(line, cardName, cardVerb, cardOperands)

        Select Case UCase(cardVerb)

            Case "JOB"
                ' ジョブ名を取得
                jobName = cardName

            Case "EXEC"
                ' 前のステップをシートに書き込む
                If stepNo > 0 Then
                    Call S2_WriteStepToSheet(ws, rowIdx, stepNo, fileName, jobName, _
                                              stepName, pgmOrProc, condValue, _
                                              ddNames, ddCount)
                    rowIdx = rowIdx + 1
                End If
                ' 新ステップ初期化
                stepNo    = stepNo + 1
                stepName  = cardName
                pgmOrProc = S2_ExtractPgmOrProc(cardOperands)
                condValue = S2_ExtractCond(cardOperands)
                ReDim ddNames(0)
                ddCount = 0

            Case "DD"
                ' DD名を現ステップに追加（重複除外）
                If cardName <> "" And stepNo > 0 Then
                    If Not IsInArray(cardName, ddNames, ddCount) Then
                        ReDim Preserve ddNames(ddCount)
                        ddNames(ddCount) = cardName
                        ddCount = ddCount + 1
                    End If
                End If

        End Select

        NextJclLine:
    Loop

    Close #fileNum

    ' 最後のステップを書き込み
    If stepNo > 0 Then
        Call S2_WriteStepToSheet(ws, rowIdx, stepNo, fileName, jobName, _
                                  stepName, pgmOrProc, condValue, _
                                  ddNames, ddCount)
        rowIdx = rowIdx + 1
    End If

    Exit Sub

FileOpenError:
    MsgBox "ファイルを開けません: " & filePath, vbExclamation
End Sub

'--------------------------------------------------
' JCL1行を name / verb / operands に分解
'
' 入力: "//STEP01   EXEC PGM=PROG01,COND=(0,NE)"
' 出力: name="STEP01", verb="EXEC", operands="PGM=PROG01,COND=(0,NE)"
'--------------------------------------------------
Private Sub S2_ParseJclLine(line As String, _
                              ByRef cardName     As String, _
                              ByRef cardVerb     As String, _
                              ByRef cardOperands As String)
    cardName     = ""
    cardVerb     = ""
    cardOperands = ""

    Dim content As String
    content = Mid(line, 3)  ' "//"を除去

    Dim tokens() As String
    tokens = S2_SplitMultiSpace(content)

    Dim tokenCount As Integer
    tokenCount = UBound(tokens) + 1

    If tokenCount = 0 Then Exit Sub
    If tokenCount >= 1 Then cardName = tokens(0)
    If tokenCount >= 2 Then cardVerb = tokens(1)

    ' operandsをcontentの元文字列から取得（コンマ・括弧を正確に保持するため）
    If tokenCount >= 3 Then
        Dim pos As Long
        pos = 1
        ' name分スキップ
        pos = pos + Len(cardName)
        ' nameの後のスペースをスキップ
        Do While pos <= Len(content) And Mid(content, pos, 1) = " "
            pos = pos + 1
        Loop
        ' verb分スキップ
        pos = pos + Len(cardVerb)
        ' verbの後のスペースをスキップ
        Do While pos <= Len(content) And Mid(content, pos, 1) = " "
            pos = pos + 1
        Loop
        cardOperands = Trim(Mid(content, pos))
    End If
End Sub

'--------------------------------------------------
' 複数スペースで分割して空文字列を除去
'--------------------------------------------------
Private Function S2_SplitMultiSpace(str As String) As String()
    Dim parts()  As String
    Dim result() As String
    Dim count    As Integer
    Dim i        As Integer

    parts = Split(str, " ")
    count = 0
    ReDim result(UBound(parts))

    For i = 0 To UBound(parts)
        If parts(i) <> "" Then
            result(count) = parts(i)
            count = count + 1
        End If
    Next i

    If count = 0 Then
        ReDim result(-1)
    Else
        ReDim Preserve result(count - 1)
    End If

    S2_SplitMultiSpace = result
End Function

'--------------------------------------------------
' EXEC の PGM名 または PROC名を取得
'   PGM=PROG01   → "PGM:PROG01"
'   PROC=MYPROC  → "PROC:MYPROC"
'   MYPROC       → "PROC:MYPROC"（暗黙的PROC呼び出し）
'--------------------------------------------------
Private Function S2_ExtractPgmOrProc(operands As String) As String
    Dim upperOp As String
    upperOp = UCase(Trim(operands))

    If upperOp = "" Then
        S2_ExtractPgmOrProc = "（不明）"
        Exit Function
    End If

    Dim pos    As Integer
    Dim after  As String
    Dim result As String

    ' PGM= を優先検索
    pos = InStr(upperOp, "PGM=")
    If pos > 0 Then
        after = Mid(operands, pos + 4)
        pos = InStr(after, ",")
        If pos > 0 Then
            result = Left(after, pos - 1)
        Else
            result = Trim(Split(after, " ")(0))
        End If
        S2_ExtractPgmOrProc = "PGM:" & Trim(result)
        Exit Function
    End If

    ' PROC= を検索
    pos = InStr(upperOp, "PROC=")
    If pos > 0 Then
        after = Mid(operands, pos + 5)
        pos = InStr(after, ",")
        If pos > 0 Then
            result = Left(after, pos - 1)
        Else
            result = Trim(Split(after, " ")(0))
        End If
        S2_ExtractPgmOrProc = "PROC:" & Trim(result)
        Exit Function
    End If

    ' 暗黙的PROC呼び出し（最初のトークンがPROC名）
    Dim tokens() As String
    tokens = S2_SplitMultiSpace(operands)
    If UBound(tokens) >= 0 Then
        result = tokens(0)
        pos = InStr(result, ",")
        If pos > 0 Then result = Left(result, pos - 1)
        S2_ExtractPgmOrProc = "PROC:" & Trim(result)
    Else
        S2_ExtractPgmOrProc = "（不明）"
    End If
End Function

'--------------------------------------------------
' EXEC/JOB の COND値を取得
' 括弧の深さを追跡して正確に取得する
'   COND=(0,NE)            → "(0,NE)"
'   COND=((0,NE),(4,LT))  → "((0,NE),(4,LT))"
'   COND=EVEN              → "EVEN"
'--------------------------------------------------
Private Function S2_ExtractCond(operands As String) As String
    Dim upperOp  As String
    Dim condPos  As Integer
    Dim after    As String

    upperOp = UCase(operands)
    condPos = InStr(upperOp, "COND=")

    If condPos = 0 Then
        S2_ExtractCond = "（なし）"
        Exit Function
    End If

    after = Mid(operands, condPos + 5)

    ' 括弧始まりの場合 → 深さ追跡
    If Left(after, 1) = "(" Then
        Dim depth As Integer
        Dim i     As Integer
        Dim c     As String
        depth = 0
        For i = 1 To Len(after)
            c = Mid(after, i, 1)
            If c = "(" Then depth = depth + 1
            If c = ")" Then
                depth = depth - 1
                If depth = 0 Then
                    S2_ExtractCond = Left(after, i)
                    Exit Function
                End If
            End If
        Next i
        S2_ExtractCond = after  ' 括弧が閉じなかった場合（異常時）
    Else
        ' 括弧なし（EVEN/ONLY等）
        Dim commaPos As Integer
        commaPos = InStr(after, ",")
        If commaPos > 0 Then
            S2_ExtractCond = Left(after, commaPos - 1)
        Else
            S2_ExtractCond = Trim(Split(after, " ")(0))
        End If
    End If
End Function

'--------------------------------------------------
' 1ステップ分をシートに書き込む
'--------------------------------------------------
Private Sub S2_WriteStepToSheet(ws As Worksheet, _
                                  rowIdx As Long, _
                                  stepNo As Integer, _
                                  fileName As String, _
                                  jobName As String, _
                                  stepName As String, _
                                  pgmOrProc As String, _
                                  condValue As String, _
                                  ddNames() As String, _
                                  ddCount As Integer)
    Dim ddListStr As String
    ddListStr = IIf(ddCount > 0, Join(ddNames, ", "), "（なし）")

    ws.Cells(rowIdx, 1).Value = stepNo
    ws.Cells(rowIdx, 2).Value = fileName
    ws.Cells(rowIdx, 3).Value = jobName
    ws.Cells(rowIdx, 4).Value = stepName
    ws.Cells(rowIdx, 5).Value = pgmOrProc
    ws.Cells(rowIdx, 6).Value = condValue
    ws.Cells(rowIdx, 7).Value = ddListStr

    If rowIdx Mod 2 = 0 Then
        ws.Range(ws.Cells(rowIdx, 1), ws.Cells(rowIdx, 7)).Interior.Color = RGB(242, 242, 242)
    End If
End Sub
