'変数未宣言エラー設定ON
Option Explicit
'●[定数](RGB値)A5M2緑色
Private Const COLOR_A5M2 As Long = 135 + 231 * 256& + 173 * 65536 'RGB(135, 231, 173) #87E7AD
'●[定数]DB種別(Oracle/PostgreSQL/SQLServer)
Private Const ORACLE = "Oracle"
Private Const POSTGRESQL = "PostgreSQL"
Private Const SQLSERVER = "SQLServer"
'●[定数]名称変換種別
Private Const L_SNAKE = "LOWER_SNAKE"
Private Const U_SNAKE = "UPPER_SNAKE"
Private Const L_CAMEL = "LOWER_CAMEL"
Private Const U_CAMEL = "UPPER_CAMEL"
'------------------------------------------------------------
'概要 : Sheet[TBL定義書]を作成する
'引数 : -
'戻値 : -
'備考 : -
'------------------------------------------------------------
Public Sub CreateSheet_TableInfo()
    '処理中描画OFF設定
    Call SettingRenderingProcess(False)

    'Sheet
    Dim ws As Worksheet
    'Loopカウンタ/終了行/書込行
    Dim i As Long
    Dim endRowIdx As Long
    Dim wtriteRowIdx As Long: wtriteRowIdx = 2
    '●図形/サイズ(横/縦)
    Dim shp As Shape
    Dim shpWidth As Double: shpWidth = 3.5
    Dim shpHeight As Double: shpHeight = 1

    '新規Sheet作成 & 新規Sheet名セット
    Set ws = ThisWorkbook.Sheets.Add(Before:=ThisWorkbook.Worksheets(1))
    ws.Name = ChangeUniqueSheetNm("TBL定義書")

    '========== 書込 ==========
    ws.Range("A1").Value = "DB"
    ws.Range("B1").Validation.Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, Operator:=xlBetween, Formula1:=ORACLE & "," & POSTGRESQL & "," & SQLSERVER
    ws.Range("B1").Value = ORACLE
    ws.Range("A2").Value = "SCHEMA"
    ws.Range("A3").Value = "TBL-No"
    ws.Range("B3").Value = "TBL(論理)"
    ws.Range("C3").Value = "TBL(物理)"
    ws.Range("D3").Value = "COL-No"
    ws.Range("E3").Value = "PK"
    ws.Range("F3").Value = "COL(論理)"
    ws.Range("G3").Value = "COL(物理)"
    ws.Range("H3").Value = "型"
    ws.Range("I3").Value = "桁"
    ws.Range("J3").Value = "NULL不可"
    ws.Range("K3").Value = "初期値"

    '========== Sheet設定 ==========
    With ws
        'Tab色
        .Tab.Color = COLOR_A5M2
        '背景色
        .Range("A1:A2").Interior.Color = RGB(248, 203, 173) '#F8CBAD
        .Range("B1:B2").Interior.Color = RGB(252, 228, 214) '#FCE4D6
        .Range("A3:K3").Interior.Color = COLOR_A5M2
        '太字
        .Range("A1:A2").Font.Bold = True
        .Range("A3:K3").Font.Bold = True
        '文字揃え
        .Cells.VerticalAlignment = xlTop
        .Cells.HorizontalAlignment = xlLeft
        .Columns("A").HorizontalAlignment = xlCenter
        .Columns("D:E").HorizontalAlignment = xlCenter
        .Columns("I").HorizontalAlignment = xlRight
        .Columns("J").HorizontalAlignment = xlCenter
        .Rows(3).HorizontalAlignment = xlCenter
        '列幅
        .Columns("A").ColumnWidth = 10
        .Columns("B:C").ColumnWidth = 30
        .Columns("D").ColumnWidth = 10
        .Columns("E").ColumnWidth = 5
        .Columns("F:G").ColumnWidth = 30
        .Columns("H").ColumnWidth = 20
        .Columns("I").ColumnWidth = 10
        .Columns("J").ColumnWidth = 12
        .Columns("K").ColumnWidth = 20
        '罫線
        .Range("A1:B2").Borders(xlInsideHorizontal).Weight = xlThin
        .Range("A1:B2").Borders(xlInsideHorizontal).Color = RGB(255, 255, 255)
        'フィルター
        .Rows(3).AutoFilter
        'グループ化
        .Rows("1:2").Group
        '折り返して表示
        .Range("A2:A3").WrapText = True
        .Range("A4:K6").WrapText = True
    End With
    '罫線
    Call DrawRuledLine_Grid(ws.Range("A3:K6"))
    ws.Activate
    '表示倍率
    ActiveWindow.Zoom = 85
    '目盛線
    ActiveWindow.DisplayGridlines = False
    'ウィンドウ枠の固定
    ws.Range("C4").Select
    ActiveWindow.FreezePanes = True

    '========== マクロボタン 作成 ==========
    Set shp = ws.Shapes.AddShape(msoShapeBevel, 0, 0, Application.CentimetersToPoints(shpWidth), Application.CentimetersToPoints(shpHeight))
    With shp
        .Fill.ForeColor.RGB = RGB(91, 155, 213)                            '#5B9BD5 塗りつぶし色
        .line.ForeColor.RGB = RGB(91, 155, 213)                            '#5B9BD5 枠線色
        .TextFrame2.TextRange.Text = "作成"                                '表示文字列
        .TextFrame2.TextRange.Font.NameFarEast = "游ゴシック"              'フォント(JP)
        .TextFrame2.TextRange.Font.Size = 16                               '文字サイズ
        .TextFrame2.TextRange.Font.Bold = True                             '太字
        .TextFrame2.TextRange.Font.Fill.ForeColor.RGB = RGB(255, 255, 255) '#FFFFFF 文字色
        .TextFrame2.VerticalAnchor = msoAnchorMiddle                       '文字揃え(横)
        .TextFrame2.TextRange.ParagraphFormat.Alignment = msoAlignCenter   '文字揃え(縦)
        .OnAction = "Btn_MyBatisGenerator"                                 'マクロ登録
        .Top = ws.Range("C1").Top + 4                                      '配置場所(縦)
        .Left = ws.Range("C1").Left + 4                                    '配置場所(横)
    End With

    '処理中描画ON設定
    Call SettingRenderingProcess(True)
End Sub
'------------------------------------------------------------
'概要 : MyBatisGeneratorファイル作成
'     : (このExcelファイルと同じ場所に作成する)
'引数 : -
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub Btn_MyBatisGenerator()
    '処理中描画OFF設定
    Call SettingRenderingProcess(False)

    Dim i As Long, j As Long                  'Loopカウンタ
    Dim ws As Worksheet: Set ws = ActiveSheet 'Sheet
    'Dim rowIdx As Long                        '行
    'Dim colIdx As Long                        '列
    Dim endRowIdx1 As Long                     '終了行1
    Dim endRowIdx2 As Long                     '終了行2
    'Dim rng As Range                          '範囲

    '●生成日時
    Dim genTime As Date: genTime = Now

    '●TBL名(前/現在)
    Dim preTblNm As String
    Dim crntTblNm As String

    '●DB/スキーマ
    Dim db As String: db = ws.Range("B1").Value2
    Dim schema As String: schema = Trim$(ws.Range("B2").Value2)
    Dim schemaLC As String: schemaLC = ConvertCamelSnakeNm(schema, L_CAMEL)
    Dim schemaUC As String: schemaUC = ConvertCamelSnakeNm(schema, U_CAMEL)
    '●TBL定義データ
    Dim tblInfo As Variant
    '●テーブル名(論理/物理/lowCamel/upCamel)/項目名(論理/物理)
    Dim logiTblNm As String
    Dim physiTblNm As String
    Dim physiTblNmLC As String
    Dim physiTblNmUC As String
    Dim logiColNm As String
    Dim physiColNm As String
    '●PK/型/桁/NULL不可/正数部/小数部
    Dim pk As String
    Dim colType As String
    Dim colLen As Variant
    Dim notNull As String
    Dim intPart As Long
    Dim dcmlPart As Long
    '●JDBC型/Java型
    Dim jdbcType As String
    Dim javaType As String

    '●日時スタンプ
    Dim timestamp As String: timestamp = Format(Now, "yyyymmddhhnnss")
    Dim timestampEn As String: timestampEn = FormatEnglishDate(genTime)
    '●ファイルパス(元/MapperJava/MapperXml)/ファイルパス/ファイル名/拡張子
    Dim filePathBase As String: filePathBase = ThisWorkbook.path & "\generated_" & timestamp & "\"
    Dim filePath_entity As String: filePath_entity = filePathBase & "com\example\entity\"
    Dim filePath_mapperJava As String: filePath_mapperJava = filePathBase & "com\example\mapper\"
    Dim filePath_mapperXml As String: filePath_mapperXml = filePathBase & "mapper\"
    Dim filePath As Variant
    Dim fileNm As String
    Dim fileExt As String
    '●ファイル内容/行
    Dim contents As String
    'Dim line As String

    '●メッセージ内容
    Dim msg As String

    '●PK数
    Dim pkCnt As Long
    '●1番目のPK行番号
    Dim firstPkRowIdx As Long
    '●有無フラグ
    Dim blobFlg As Boolean   'BLOB系項目
    Dim jBDcmlFlg As Boolean 'Java(BigDecimal)
    Dim jDateFlg As Boolean  'Java(Date)

    '●追加文字
    Dim addStr As String

    '終了行 セット
    endRowIdx1 = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    'DB 未選択の場合
    If LenB(db) = 0 Then
        Call MsgBox("(B1) DBを選択して下さい", vbCritical, "ERROR")
        Exit Sub
    End If
    'スキーマ 未入力の場合
    If LenB(schema) = 0 Then
        Call MsgBox("(B2) スキーマを入力して下さい", vbCritical, "ERROR")
        Exit Sub
    End If
    'TBL定義 無しの場合
    If endRowIdx1 <= 3 Then
        Call MsgBox("TBL定義がありません", vbCritical, "ERROR")
        Exit Sub
    End If

    '========== ファイル作成先フォルダ 作成 ==========
    For Each filePath In Array(filePath_entity, filePath_mapperJava, filePath_mapperXml)
        If Not CreateFolderPath(filePath) Then
            msg = "フォルダの作成に失敗しました" & vbCrLf & _
                  "PATH : " & filePath
            Call MsgBox(msg, vbCritical, "")
            Exit Sub
        End If
    Next

    'TBL定義データ 取得
    tblInfo = ws.Range("A4:K" & endRowIdx1).Value2
    Debug.Print UBound(tblInfo, 1)

    For i = 1 To UBound(tblInfo, 1)
        'TBL定義 取得(テーブル名(論理/物理)/項目名(論理/物理)/PK/型/桁/NULL不可/正数部/小数部)
        logiTblNm = CStr(tblInfo(i, 2))
        physiTblNm = CStr(tblInfo(i, 3))
        pk = CStr(tblInfo(i, 5))
        logiColNm = CStr(tblInfo(i, 6))
        physiColNm = CStr(tblInfo(i, 7))
        colType = CStr(tblInfo(i, 8))
        colLen = CStr(tblInfo(i, 9))
        notNull = CStr(tblInfo(i, 10))
        'TBL名(変換) セット
        physiTblNmLC = ConvertCamelSnakeNm(physiTblNm, L_CAMEL)
        physiTblNmUC = ConvertCamelSnakeNm(physiTblNm, U_CAMEL)
        'TBL名(現在) セット
        crntTblNm = physiTblNm
        'テーブル終了行 セット
        endRowIdx2 = 0
        For j = i To UBound(tblInfo, 1)
            If crntTblNm <> tblInfo(j, 3) Then
                endRowIdx2 = j - 1
                Exit For
            End If
        Next
        If endRowIdx2 = 0 Then endRowIdx2 = endRowIdx1 - 3
        'PK数/1番目PK行番号 セット
        pkCnt = 0
        firstPkRowIdx = 0
        For j = i To endRowIdx2
            If tblInfo(j, 5) = "Y" Then
                pkCnt = pkCnt + 1
                If firstPkRowIdx = 0 Then firstPkRowIdx = j
            End If
        Next
        '初期化
        blobFlg = False
        jBDcmlFlg = False
        jDateFlg = False
        pkCnt = 0
        'セット
        For j = i To endRowIdx2
            '有無フラグ(BLOB系項目)
            If IsBlobColType(db, tblInfo(j, 8)) Then
                blobFlg = True
            End If
            '有無フラグ(Java型)
            javaType = GetJavaType(db, GetJdbcType(db, tblInfo(j, 8)), tblInfo(j, 9))
            javaType = Mid$(javaType, InStrRev(javaType, ".") + 1)
            If javaType = "BigDecimal" Then
                jBDcmlFlg = True
            ElseIf javaType = "Date" Then
                jDateFlg = True
            End If
            'PK数
            If tblInfo(j, 5) = "Y" Then
                pkCnt = pkCnt + 1
                If firstPkRowIdx = 0 Then firstPkRowIdx = j
            End If
        Next

        '■■■■■■■■■■ ENTITY ■■■■■■■■■■
        contents = ""
        contents = contents & "package com.example.entity." & schemaLC & ";" & vbCrLf
        contents = contents & "" & vbCrLf
        If jBDcmlFlg Then
            contents = contents & "import java.math.BigDecimal;" & vbCrLf
        End If
        If jDateFlg Then
            contents = contents & "import java.util.Date;" & vbCrLf
        End If
        contents = contents & "" & vbCrLf
        contents = contents & "public class " & physiTblNmUC & " {" & vbCrLf
        For j = i To endRowIdx2
            If Not IsBlobColType(db, tblInfo(j, 8)) Then
                contents = contents & "    /**" & vbCrLf
                contents = contents & "     *" & vbCrLf
                contents = contents & "     * This field was generated by MyBatis Generator." & vbCrLf
                contents = contents & "     * This field corresponds to the database column " & schema & "." & physiTblNm & "." & tblInfo(j, 7) & vbCrLf
                contents = contents & "     *" & vbCrLf
                contents = contents & "     * @mbg.generated " & timestampEn & vbCrLf
                contents = contents & "     */" & vbCrLf
                javaType = GetJavaType(db, GetJdbcType(db, tblInfo(j, 8)), tblInfo(j, 9))
                javaType = Mid$(javaType, InStrRev(javaType, ".") + 1)
                contents = contents & "    private " & javaType & " " & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & ";" & vbCrLf
                contents = contents & vbCrLf
            End If
        Next
        For j = i To endRowIdx2
            If Not IsBlobColType(db, tblInfo(j, 8)) Then
                contents = contents & "    /**" & vbCrLf
                contents = contents & "     * This method was generated by MyBatis Generator." & vbCrLf
                contents = contents & "     * This method returns the value of the database column " & schema & "." & physiTblNm & "." & tblInfo(j, 7) & vbCrLf
                contents = contents & "     *" & vbCrLf
                contents = contents & "     * @return the value of " & schema & "." & physiTblNm & "." & tblInfo(j, 7) & vbCrLf
                contents = contents & "     *" & vbCrLf
                contents = contents & "     * @mbg.generated " & timestampEn & vbCrLf
                contents = contents & "     */" & vbCrLf
                javaType = GetJavaType(db, GetJdbcType(db, tblInfo(j, 8)), tblInfo(j, 9))
                javaType = Mid$(javaType, InStrRev(javaType, ".") + 1)
                contents = contents & "    public " & javaType & " get" & ConvertCamelSnakeNm(tblInfo(j, 7), U_CAMEL) & "() {" & vbCrLf
                contents = contents & "        return " & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & ";" & vbCrLf
                contents = contents & "    }" & vbCrLf
                contents = contents & "" & vbCrLf
                contents = contents & "    /**" & vbCrLf
                contents = contents & "     * This method was generated by MyBatis Generator." & vbCrLf
                contents = contents & "     * This method sets the value of the database column " & schema & "." & physiTblNm & "." & tblInfo(j, 7) & vbCrLf
                contents = contents & "     *" & vbCrLf
                contents = contents & "     * @param " & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & " the value for " & schema & "." & physiTblNm & "." & tblInfo(j, 7) & vbCrLf
                contents = contents & "     *" & vbCrLf
                contents = contents & "     * @mbg.generated " & timestampEn & vbCrLf
                contents = contents & "     */" & vbCrLf
                javaType = GetJavaType(db, GetJdbcType(db, tblInfo(j, 8)), tblInfo(j, 9))
                javaType = Mid$(javaType, InStrRev(javaType, ".") + 1)
                contents = contents & "    public void set" & ConvertCamelSnakeNm(tblInfo(j, 7), U_CAMEL) & "(" & javaType & " " & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & ") {" & vbCrLf
                contents = contents & "        this." & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & " = " & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & ";" & vbCrLf
                contents = contents & "    }" & vbCrLf
                contents = contents & "" & vbCrLf
            End If
        Next
        '末尾除去/付与
        contents = Left$(contents, Len(contents) - Len(vbCrLf))
        contents = contents & "}"

        'ファイル作成(UTF-8(BOM無し))
        Call CreateTxtFile(filePath_entity, physiTblNmUC, "java", contents, "UTF-8", False)

        '■■■■■■■■■■ ENTITY(Example) ■■■■■■■■■■

        '■■■■■■■■■■ ENTITY(Key) ■■■■■■■■■■
        '初期化
        blobFlg = False
        jBDcmlFlg = False
        jDateFlg = False
        pkCnt = 0
        'セット
        For j = i To endRowIdx2
            '有無フラグ(BLOB系項目)
            If IsBlobColType(db, tblInfo(j, 8)) Then
                blobFlg = True
            End If
            '有無フラグ(Java型)
            javaType = GetJavaType(db, GetJdbcType(db, tblInfo(j, 8)), tblInfo(j, 9))
            javaType = Mid$(javaType, InStrRev(javaType, ".") + 1)
            If javaType = "BigDecimal" Then
                jBDcmlFlg = True
            ElseIf javaType = "Date" Then
                jDateFlg = True
            End If
            'PK数
            If tblInfo(j, 5) = "Y" Then
                pkCnt = pkCnt + 1
                If firstPkRowIdx = 0 Then firstPkRowIdx = j
            End If
        Next

        '複数PKの場合
        If 2 <= pkCnt Then
            contents = ""
            contents = contents & "package com.example.entity." & schemaLC & ";" & vbCrLf
            contents = contents & "" & vbCrLf
            If jBDcmlFlg Then
                contents = contents & "import java.math.BigDecimal;" & vbCrLf
            End If
            If jDateFlg Then
                contents = contents & "import java.util.Date;" & vbCrLf
            End If
            contents = contents & "" & vbCrLf
            contents = contents & "public class " & physiTblNmUC & "Key {" & vbCrLf
            For j = i To endRowIdx2
                If Not IsBlobColType(db, tblInfo(j, 8)) And tblInfo(j, 5) = "Y" Then
                    contents = contents & "    /**" & vbCrLf
                    contents = contents & "     *" & vbCrLf
                    contents = contents & "     * This field was generated by MyBatis Generator." & vbCrLf
                    contents = contents & "     * This field corresponds to the database column " & schema & "." & physiTblNm & "." & tblInfo(j, 7) & vbCrLf
                    contents = contents & "     *" & vbCrLf
                    contents = contents & "     * @mbg.generated " & timestampEn & vbCrLf
                    contents = contents & "     */" & vbCrLf
                    javaType = GetJavaType(db, GetJdbcType(db, tblInfo(j, 8)), tblInfo(j, 9))
                    javaType = Mid$(javaType, InStrRev(javaType, ".") + 1)
                    contents = contents & "    private " & javaType & " " & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & ";" & vbCrLf
                    contents = contents & vbCrLf
                End If
            Next
            For j = i To endRowIdx2
                If Not IsBlobColType(db, tblInfo(j, 8)) And tblInfo(j, 5) = "Y" Then
                    contents = contents & "    /**" & vbCrLf
                    contents = contents & "     * This method was generated by MyBatis Generator." & vbCrLf
                    contents = contents & "     * This method returns the value of the database column " & schema & "." & physiTblNm & "." & tblInfo(j, 7) & vbCrLf
                    contents = contents & "     *" & vbCrLf
                    contents = contents & "     * @return the value of " & schema & "." & physiTblNm & "." & tblInfo(j, 7) & vbCrLf
                    contents = contents & "     *" & vbCrLf
                    contents = contents & "     * @mbg.generated " & timestampEn & vbCrLf
                    contents = contents & "     */" & vbCrLf
                    javaType = GetJavaType(db, GetJdbcType(db, tblInfo(j, 8)), tblInfo(j, 9))
                    javaType = Mid$(javaType, InStrRev(javaType, ".") + 1)
                    contents = contents & "    public " & javaType & " get" & ConvertCamelSnakeNm(tblInfo(j, 7), U_CAMEL) & "() {" & vbCrLf
                    contents = contents & "        return " & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & ";" & vbCrLf
                    contents = contents & "    }" & vbCrLf
                    contents = contents & "" & vbCrLf
                    contents = contents & "    /**" & vbCrLf
                    contents = contents & "     * This method was generated by MyBatis Generator." & vbCrLf
                    contents = contents & "     * This method sets the value of the database column " & schema & "." & physiTblNm & "." & tblInfo(j, 7) & vbCrLf
                    contents = contents & "     *" & vbCrLf
                    contents = contents & "     * @param " & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & " the value for " & schema & "." & physiTblNm & "." & tblInfo(j, 7) & vbCrLf
                    contents = contents & "     *" & vbCrLf
                    contents = contents & "     * @mbg.generated " & timestampEn & vbCrLf
                    contents = contents & "     */" & vbCrLf
                    javaType = GetJavaType(db, GetJdbcType(db, tblInfo(j, 8)), tblInfo(j, 9))
                    javaType = Mid$(javaType, InStrRev(javaType, ".") + 1)
                    contents = contents & "    public void set" & ConvertCamelSnakeNm(tblInfo(j, 7), U_CAMEL) & "(" & javaType & " " & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & ") {" & vbCrLf
                    contents = contents & "        this." & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & " = " & ConvertCamelSnakeNm(tblInfo(j, 7), L_CAMEL) & ";" & vbCrLf
                    contents = contents & "    }" & vbCrLf
                    contents = contents & "" & vbCrLf
                End If
            Next
            '末尾除去/付与
            contents = Left$(contents, Len(contents) - Len(vbCrLf))
            contents = contents & "}"
        End If

        'ファイル作成(UTF-8(BOM無し))
        Call CreateTxtFile(filePath_entity, physiTblNmUC & "Key", "java", contents, "UTF-8", False)

        '■■■■■■■■■■ ENTITY(WithBLOBs) ■■■■■■■■■■

        '■■■■■■■■■■ MAPER(java) ■■■■■■■■■■
        contents = ""
        contents = contents & "package com.example.mapper" & "." & schemaLC & ";" & vbCrLf
        contents = contents & vbCrLf
        contents = contents & "import com.example.entity." & schemaLC & "." & physiTblNmUC & ";" & vbCrLf
        contents = contents & "import com.example.entity." & schemaLC & "." & physiTblNmUC & "Example;" & vbCrLf
        If 2 <= pkCnt Then
            contents = contents & "import com.example.entity." & schemaLC & "." & physiTblNmUC & "Key;" & vbCrLf
        End If
        If blobFlg Then
            contents = contents & "import com.example.entity." & schemaLC & "." & physiTblNmUC & "WithBLOBs;" & vbCrLf
        End If
        contents = contents & "import java.util.List;" & vbCrLf
        contents = contents & "import org.apache.ibatis.annotations.Param;" & vbCrLf
        contents = contents & vbCrLf
        contents = contents & "public interface " & physiTblNmUC & "Mapper {" & vbCrLf
        contents = contents & "    /**" & vbCrLf
        contents = contents & "     * This method was generated by MyBatis Generator." & vbCrLf
        contents = contents & "     * This method corresponds to the database table " & schema & "." & physiTblNm & vbCrLf
        contents = contents & "     *" & vbCrLf
        contents = contents & "     * @mbg.generated " & timestampEn & vbCrLf
        contents = contents & "     */" & vbCrLf
        contents = contents & "    long countByExample(" & physiTblNmUC & "Example example);" & vbCrLf
        contents = contents & vbCrLf
        contents = contents & "    /**" & vbCrLf
        contents = contents & "     * This method was generated by MyBatis Generator." & vbCrLf

'------------------------------------------------------------
'概要 : フォルダが存在するか判定する
'引数 : path     : フォルダまでのパス(絶対パス)
'     : folderNm : フォルダ名
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsFolder(ByVal path As String, ByVal folderNm As String) As Boolean
    '末尾付与
    If Right$(path, 1) <> "\" Then path = path & "\"
    '判定(Dirだけだと同名ファイルにもヒットするためGetAttrで再確認)
    If Dir(path & folderNm, vbDirectory) = "" Then
        IsExistsFolder = False
    Else
        IsExistsFolder = (GetAttr(path & folderNm) And vbDirectory) = vbDirectory
    End If
End Function
'------------------------------------------------------------
'概要 : 重複しないフォルダ名となるまで、末尾に連番を付与する
'     : (同じ名前のフォルダが存在しない場合、そのまま返す)
'     : (同じ名前のフォルダが存在する場合、2からの連番を末尾に付与し、重複しないフォルダ名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : path     : フォルダまでのパス
'     : folderNm : フォルダ名
'戻値 : フォルダ名
'備考 : -
'------------------------------------------------------------
Private Function ChangeUniqueFolderNm(ByVal path As String, ByVal folderNm As String) As String
    Dim i As Long
    Dim uniqueNo As String
    '存在する場合
    For i = 2 To 1000
        '存在しない場合
        If Not IsExistsFolder(path, folderNm & uniqueNo) Then
            ChangeUniqueFolderNm = folderNm & uniqueNo
            Exit Function
        End If
        '末尾付与
        uniqueNo = " (" & i & ")"
    Next
End Function

    Dim cratCnt As Long
    '最大分だけ確保
    ReDim cratPaths(UBound(Split(path, "\")))
    cratCnt = 0
    '末尾除去
    If Right$(path, 1) = "\" Then path = Left$(path, Len(path) - 1)
    parts = Split(path, "\")
    crntPath = parts(0)
    For i = 1 To UBound(parts)
        If Not IsExistsFolder(crntPath, parts(i)) Then
            MkDir crntPath & "\" & parts(i)
            '作成フォルダパス 追加
            cratPaths(cratCnt) = crntPath & "\" & parts(i)
            cratCnt = cratCnt + 1
        End If
        crntPath = crntPath & "\" & parts(i)
    Next
    CreateFolderPath = True
    Exit Function
ErrHandler:
    'ロールバック(逆順で削除)
    For i = cratCnt - 1 To 0 Step -1
        '削除中の追加エラーは無視して続行
        On Error Resume Next
        RmDir cratPaths(i)
        On Error GoTo 0
    Next
    CreateFolderPath = False
End Function

'     : ext    : 拡張子
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsFile(ByVal path As String, ByVal fileNm As String, ByVal ext As String) As Boolean
    'パス末尾付与
    If Right$(path, 1) <> "\" Then path = path & "\"
    '判定
    IsExistsFile = (Dir(path & fileNm & "." & ext) <> "")
End Function

'概要 : 重複しないファイル名となるまで、末尾に連番を付与する
'     : (同じ名前のファイルが存在しない場合、そのまま返す)
'     : (同じ名前のファイルが存在する場合、2からの連番を末尾に付与し、重複しないファイル名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : path   : ファイルまでのパス
'     : fileNm : ファイル名
'     : ext    : 拡張子
'戻値 : ファイル名(拡張子付き)

Private Function ChangeUniqueFileNm(ByVal path As String, ByVal fileNm As String, ByVal ext As String) As String
    Dim i As Long
    Dim uniqueNo As String
    For i = 2 To 1000
        '存在しない場合
        If Not IsExistsFile(path, fileNm & uniqueNo, ext) Then
            ChangeUniqueFileNm = fileNm & uniqueNo & "." & ext
            Exit Function
        End If
        '末尾付与
        uniqueNo = " (" & i & ")"
    Next
End Function

'     : (同じ名前のSheetが存在する場合、2からの連番を末尾に付与し、重複しないSheet名で返す)

'------------------------------------------------------------
'概要 : 列番号から列名(アルファベット)を取得する
'     : (範囲外の数値の場合、空を返す)
'引数 : colIdx : 列番号
'戻値 : 列名(アルファベット)
'備考 : -
'------------------------------------------------------------
Private Function GetColNm(ByVal colIdx As Long) As String
    Dim result As String
    '範囲外の場合
    If colIdx < 1 Or Application.Columns.Count < colIdx Then
        GetColNm = ""
        Exit Function
    End If
    Do While 0 < colIdx
        colIdx = colIdx - 1
        result = Chr$(65 + (colIdx Mod 26)) & result
        colIdx = colIdx \ 26
    Loop
    GetColNm = result
End Function
'------------------------------------------------------------
'概要 : 列名(アルファベット)から列番号を取得する
'     : (範囲外の列名の場合、0を返す)
'引数 : colNm : 列名(アルファベット)
'戻値 : 列番号
'備考 : -
'------------------------------------------------------------
Private Function GetColIdx(ByVal colNm As String) As Long
    Dim i As Long
    Dim result As Long
    Dim baseA As Long: baseA = Asc("A") - 1
    Dim colNmLen As Long: colNmLen = Len(UCase$(colNm))
    Dim ch As Long
    For i = 1 To colNmLen
        ch = Asc(Mid$(colNm, i, 1)) - baseA
        '範囲外の場合
        If ch < 1 Or 26 < ch Then
            GetColIdx = 0
            Exit Function
        End If
        result = result * 26 + ch
    Next
    GetColIdx = result
End Function
'------------------------------------------------------------
'概要 : テキストファイルを作成する(UTF-8)
'引数 : filePath  : ファイルパス
'     : fileNm    : ファイル名
'     : fileExt   : ファイル拡張子
'     : contents  : 書込内容
'     : charsetNm : 文字コード名(UTF-8 / Shift-JIS など)
'     : bomFlg    : BOM有無フラグ(True=有 / False=無)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub CreateTxtFile(ByVal filePath As String, ByVal fileNm As String, ByVal fileExt As String, ByVal contents As String, ByVal charsetNm As String, ByVal bomFlg As Boolean)
    '●ファイル書込用
    Dim stream As Object: Set stream = CreateObject("ADODB.Stream")
    Dim binStream As Object
    Dim byteData() As Byte
    Dim bomLen As Integer: bomLen = GetBomLen(charsetNm)

    '末尾付与
    If Right$(filePath, 1) <> "\" Then filePath = filePath & "\"
    '先頭付与
    If Left$(fileExt, 1) <> "." Then fileExt = "." & fileExt
    '末尾除去
    If Right$(fileNm, Len(fileExt)) = fileExt Then fileNm = Left$(fileNm, Len(fileNm) - Len(fileExt))

    With stream
        .Type = 2
        .Charset = charsetNm
        .Open
        .WriteText contents
        .SaveToFile filePath & fileNm & fileExt, 2 '既存ファイル上書き設定
        .Close
    End With
    Set stream = Nothing

    'BOMなしの場合、先頭nバイト(BOM)除去で作り直す
    If Not bomFlg And 0 < bomLen Then
        '===== 読込 =====
        Set binStream = CreateObject("ADODB.Stream")
        With binStream
            .Type = 1
            .Open
            .LoadFromFile filePath & fileNm & fileExt
            .Position = bomLen
            byteData = .Read
            .Close
        End With
        '===== 書込 =====
        With binStream
            .Type = 1
            .Open
            .Write byteData
            .SaveToFile filePath & fileNm & fileExt, 2
            .Close
        End With
        Set binStream = Nothing
    End If
End Sub
'------------------------------------------------------------
'概要 : 文字コード名のBOMバイト数を取得する
'引数 : charsetNm : 文字コード名(UTF-8 / Shift-JIS など)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Function GetBomLen(ByVal charsetNm As String) As Integer
    Select Case UCase$(charsetNm)
        Case "UTF-8"
            GetBomLen = 3
        Case "UTF-16", "UNICODE", "UTF-16LE", "UTF-16BE"
            GetBomLen = 2
        Case "UTF-32"
            GetBomLen = 4
        Case Else
            GetBomLen = 0
    End Select
End Function

'------------------------------------------------------------
'概要 : MyBatisGenerator用の日時フォーマットに変換する
'     : (Thu Jan 1 00:00:00 JST 2026)
'引数 : dt : 日時
'戻値 : 英語日時文字列
'備考 : -
'------------------------------------------------------------
Function FormatEnglishDate(dt As Date) As String
    '●曜日/月
    Dim days As Variant: days = Array("Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat")
    Dim months As Variant: months = Array("Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec")
    '各要素セット (Weekdayは1=日曜なのでインデックスは -1)
    Dim w As String: w = days(Weekday(dt, vbSunday) - 1)
    Dim m As String: m = months(Month(dt) - 1)
    Dim d As String: d = Format(Day(dt), "00")
    Dim time As String: time = Format(dt, "hh:nn:ss")
    Dim tz As String: tz = "JST"
    Dim y As String: y = Year(dt)
    '結合
    FormatEnglishDate = w & " " & m & " " & d & " " & time & " " & tz & " " & y
End Function
'------------------------------------------------------------
'概要 : キャメルケースとスネークケースの変換をする
'     : [convertType=UPPER_SNAKE] の場合 "USER_ID"
'     : [convertType=LOWER_SNAKE] の場合 "user_id"
'     : [convertType=LOWER_CAMEL] の場合 "userId"
'     : [convertType=UPPER_CAMEL] の場合 "UserId"
'引数 : str         : 文字列
'     : convertType : 名称変換区分
'戻値 : 文字列
'備考 : -
'------------------------------------------------------------
Private Function ConvertCamelSnakeNm(ByVal str As String, ByVal convertType As String) As String
    Dim wordCol As Collection
    Dim word As Variant
    Dim result As String
    Dim i As Long
    '名前を単語に分解
    Set wordCol = ParseNm(str)
    If wordCol.Count <> 0 Then
        Select Case convertType
            Case U_SNAKE
                For Each word In wordCol
                    If result <> "" Then
                        result = result & "_"
                    End If
                    result = result & UCase$(CStr(word))
                Next
            Case L_SNAKE
                For Each word In wordCol
                    If result <> "" Then
                        result = result & "_"
                    End If
                    result = result & LCase$(CStr(word))
                Next
            Case L_CAMEL
                For i = 1 To wordCol.Count
                    word = LCase$(CStr(wordCol(i)))
                    If i = 1 Then
                        result = result & word
                    Else
                        result = result & UCase$(Left$(word, 1)) & Mid$(word, 2)
                    End If
                Next
            Case U_CAMEL
                For Each word In wordCol
                    word = LCase$(CStr(word))
                    result = result & UCase$(Left$(word, 1)) & Mid$(word, 2)
                Next
        End Select
    End If
    ConvertCamelSnakeNm = result
End Function
'------------------------------------------------------------
'概要 : 文字列を単語ごとに分解する
'     : USER_ID    → USER / ID
'     : userId     → user / Id
'     : UserName01 → User / Name / 01
'     : A_B_C_01   → A / B / C / 01
'引数 : str : 文字列
'戻値 : 単語リスト
'備考 : -
'------------------------------------------------------------
Private Function ParseNm(ByVal str As String) As Collection
    Dim col As New Collection
    Dim i As Long
    Dim ch As String
    Dim preCh As String
    Dim nextCh As String
    Dim crntWord As String
    str = Trim$(str)
    If str <> "" Then
        For i = 1 To Len(str)
            ch = Mid$(str, i, 1)
            If 1 < i Then
                preCh = Mid$(str, i - 1, 1)
            Else
                preCh = ""
            End If
            If i < Len(str) Then
                nextCh = Mid$(str, i + 1, 1)
            Else
                nextCh = ""
            End If
            If ch = "_" Then
                Call AddCol(col, crntWord)
                crntWord = ""
            '小文字 → 大文字
            ElseIf IsLowerCaseChar(preCh) And IsUpperCaseChar(ch) Then
                Call AddCol(col, crntWord)
                crntWord = ch
            '大文字 → 大文字 → 小文字
            ElseIf IsUpperCaseChar(preCh) And IsUpperCaseChar(ch) And IsLowerCaseChar(nextCh) Then
                Call AddCol(col, crntWord)
                crntWord = ch
            Else
                crntWord = crntWord & ch
            End If
        Next
        '最後の単語
        Call AddCol(col, crntWord)
    End If
    Set ParseNm = col
End Function
'------------------------------------------------------------
'概要 : 文字列をリストに追加
'引数 : col : リスト
'     : srt : 文字列
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub AddCol(ByRef col As Collection, ByVal str As String)
    str = Trim$(str)
    If str <> "" Then
        Call col.Add(str)
    End If
End Sub
'------------------------------------------------------------
'概要 : 文字列が全て大文字の英字か判定する
'引数 : str : 文字列
'戻値 : 判定結果(True:のみ / False:以外含む)
'備考 : -
'------------------------------------------------------------
Private Function IsUpperCaseChar(ByVal str As String) As Boolean
    If Len(str) <> 1 Then Exit Function
    IsUpperCaseChar = (str Like "[A-Z]")
End Function
'------------------------------------------------------------
'概要 : 文字列が全て小文字の英字か判定する
'引数 : str : 文字列
'戻値 : 判定結果(True:のみ / False:以外含む)
'備考 : -
'------------------------------------------------------------
Private Function IsLowerCaseChar(ByVal str As String) As Boolean
    If Len(str) <> 1 Then Exit Function
    IsLowerCaseChar = (str Like "[a-z]")
End Function

'概要 : 対応するJDBCタイプ名を取得する

Private Function GetJdbcType(ByVal dbType As String, ByVal colType As String) As String

        Select Case colType
            Case "CHAR"
                result = "CHAR"

                result = "VARCHAR"

                result = "NVARCHAR"

            Case "BINARY_DOUBLE"
                result = "DOUBLE"
            Case "DATE"

            Case "BLOB"
                result = "BLOB"
            Case "RAW"
                result = "VARBINARY"

            Case "uniqueidentifier"
                result = "CHAR"

                If InStr(colLen, ",") = 0 Then

                        result = "java.lang.Short"
                    ElseIf colLen <= 9 Then
                        result = "java.lang.Integer"
                    ElseIf colLen <= 18 Then
                        result = "java.lang.Long"

                Else
                    intPart = Split(colLen, ",")(0)
                    dcmlPart = Split(colLen, ",")(1)
                    If dcmlPart = 0 Then
                        If intPart <= 4 Then

                        ElseIf intPart <= 9 Then

                result = "java.time.LocalDateTime"
            Case "TIME_WITH_TIMEZONE"
                result = "java.time.OffsetTime"
            Case "TIMESTAMP_WITH_TIMEZONE"
                result = "java.time.OffsetDateTime"
            Case "VARBINARY", "LONGVARBINARY", "BINARY", "BLOB"
                result = "byte[]"
        End Select
        'TIMESTAMP(TIME ZONE無し)の場合
        If Left$(jdbcType, 9) = "TIMESTAMP" And InStr(jdbcType, "TIME ZONE") = 0 Then
            result = "java.util.Date"
        'TIMESTAMP(TIME ZONE有り)の場合
        ElseIf Left$(jdbcType, 9) = "TIMESTAMP" And 0 < InStr(jdbcType, "TIME ZONE") Then
            result = "java.time.OffsetDateTime"
        '(TIME ZONE有り)の場合
        ElseIf 0 < InStr(jdbcType, "TIME ZONE") Then

    ElseIf dbType = POSTGRESQL Then
        Select Case jdbcType
            Case "character varying", "text"

            Case "character"
                result = "CHAR"
            Case "numeric"
                result = "NUMERIC"

                result = "DATE"

                result = "BINARY"

    ElseIf dbType = SQLSERVER Then

                result = "VARCHAR"

'概要 : 項目型がBLOB系か判定する
'引数 : dbType  : DB種別(Oracle/PostgreSQL/SQLServer)
'     : colType : 項目型名
'戻値 : 判定結果(True:である / False:でない)
'備考 : -
'------------------------------------------------------------
Private Function IsBlobColType(ByVal dbType As String, ByVal colType As String) As Boolean
    IsBlobColType = False
    If dbType = ORACLE Then
        If colType = "BLOB" Or colType = "RAW" Or colType = "LONG RAW" Then IsBlobColType = True
    ElseIf dbType = POSTGRESQL Then
        If colType = "bytea" Then IsBlobColType = True
    ElseIf dbType = SQLSERVER Then
        If colType = "varbinary" Or colType = "xml" Then IsBlobColType = True
    End If
End Function
