        colLen = CStr(tblInfo(i, 9))

'------------------------------------------------------------
'概要 : フォルダが存在するか判定する
'引数 : path     : フォルダまでのパス(絶対パス)
'     : folderNm : フォルダ名
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsFolder(ByVal path As String, ByVal folderNm As String) As Boolean
    '末尾付与
    If Right$(path, 1) <> "\" Then path = path & "\"
    '判定(Dirだけだと同名ファイルにもヒットするためGetAttrで再確認)
    If Dir(path & folderNm, vbDirectory) = "" Then
        IsExistsFolder = False
    Else
        IsExistsFolder = (GetAttr(path & folderNm) And vbDirectory) = vbDirectory
    End If
End Function
'------------------------------------------------------------
'概要 : 重複しないフォルダ名となるまで、末尾に連番を付与する
'     : (同じ名前のフォルダが存在しない場合、そのまま返す)
'     : (同じ名前のフォルダが存在する場合、2からの連番を末尾に付与し、重複しないフォルダ名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : path     : フォルダまでのパス
'     : folderNm : フォルダ名
'戻値 : フォルダ名
'備考 : -
'------------------------------------------------------------
Private Function ChangeUniqueFolderNm(ByVal path As String, ByVal folderNm As String) As String
    Dim i As Long
    Dim uniqueNo As String
    '存在する場合
    For i = 2 To 1000
        '存在しない場合
        If Not IsExistsFolder(path, folderNm & uniqueNo) Then
            ChangeUniqueFolderNm = folderNm & uniqueNo
            Exit Function
        End If
        '末尾付与
        uniqueNo = " (" & i & ")"
    Next
End Function
'------------------------------------------------------------
'概要 : 指定パスとなるフォルダを、存在しない階層も含めて全て作成する
'     : (途中でエラーが発生した場合、今回新規作成した分だけを削除してロールバックする)
'引数 : path : 作成したいフォルダの絶対パス
'戻値 : 判定結果(True:全て作成できた(元々あった場合も含む) / False:途中でエラー発生、ロールバック済み)
'備考 : -
'------------------------------------------------------------
Private Function CreateFolderPath(ByVal path As String) As Boolean
    On Error GoTo ErrHandler
    Dim i As Long
    Dim parts() As String
    Dim crntPath As String
    '作成フォルダパス(ロールバック用)
    Dim cratPaths() As String
    Dim cratCnt As Long
    '最大分だけ確保
    ReDim cratPaths(UBound(Split(path, "\")))
    cratCnt = 0
    '末尾除去
    If Right$(path, 1) = "\" Then path = Left$(path, Len(path) - 1)
    parts = Split(path, "\")
    crntPath = parts(0)
    For i = 1 To UBound(parts)
        If Not IsExistsFolder(crntPath, parts(i)) Then
            MkDir crntPath & "\" & parts(i)
            '作成フォルダパス 追加
            cratPaths(cratCnt) = crntPath & "\" & parts(i)
            cratCnt = cratCnt + 1
        End If
        crntPath = crntPath & "\" & parts(i)
    Next
    CreateFolderPath = True
    Exit Function
ErrHandler:
    'ロールバック(逆順で削除)
    For i = cratCnt - 1 To 0 Step -1
        '削除中の追加エラーは無視して続行
        On Error Resume Next
        RmDir cratPaths(i)
        On Error GoTo 0
    Next
    CreateFolderPath = False
End Function
'------------------------------------------------------------
'概要 : ファイルが存在するか判定する
'引数 : path   : ファイルまでのパス(絶対パス)
'     : fileNm : ファイル名
'     : ext    : 拡張子
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsFile(ByVal path As String, ByVal fileNm As String, ByVal ext As String) As Boolean
    'パス末尾付与
    If Right$(path, 1) <> "\" Then path = path & "\"
    '判定
    IsExistsFile = (Dir(path & fileNm & "." & ext) <> "")
End Function
'------------------------------------------------------------
'概要 : 重複しないファイル名となるまで、末尾に連番を付与する
'     : (同じ名前のファイルが存在しない場合、そのまま返す)
'     : (同じ名前のファイルが存在する場合、2からの連番を末尾に付与し、重複しないファイル名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : path   : ファイルまでのパス
'     : fileNm : ファイル名
'     : ext    : 拡張子
'戻値 : ファイル名(拡張子付き)
'備考 : -
'------------------------------------------------------------

            Exit Function
        End If
        '末尾付与
        uniqueNo = " (" & i & ")"
    Next
End Function

'     : (同じ名前のSheetが存在する場合、2からの連番を末尾に付与し、重複しないSheet名で返す)

'------------------------------------------------------------
'概要 : 列番号から列名(アルファベット)を取得する
'     : (範囲外の数値の場合、空を返す)
'引数 : colIdx : 列番号
'戻値 : 列名(アルファベット)
'備考 : -
'------------------------------------------------------------
Private Function GetColNm(ByVal colIdx As Long) As String
    Dim result As String
    '範囲外の場合
    If colIdx < 1 Or Application.Columns.Count < colIdx Then
        GetColNm = ""
        Exit Function
    End If
    Do While 0 < colIdx
        colIdx = colIdx - 1
        result = Chr$(65 + (colIdx Mod 26)) & result
        colIdx = colIdx \ 26
    Loop
    GetColNm = result
End Function
'------------------------------------------------------------
'概要 : 列名(アルファベット)から列番号を取得する
'     : (範囲外の列名の場合、0を返す)
'引数 : colNm : 列名(アルファベット)
'戻値 : 列番号
'備考 : -
'------------------------------------------------------------
Private Function GetColIdx(ByVal colNm As String) As Long
    Dim i As Long
    Dim result As Long
    Dim baseA As Long: baseA = Asc("A") - 1
    Dim colNmLen As Long: colNmLen = Len(UCase$(colNm))
    Dim ch As Long
    For i = 1 To colNmLen
        ch = Asc(Mid$(colNm, i, 1)) - baseA
        '範囲外の場合
        If ch < 1 Or 26 < ch Then
            GetColIdx = 0
            Exit Function
        End If
        result = result * 26 + ch
    Next
    GetColIdx = result
End Function
'------------------------------------------------------------
'概要 : テキストファイルを作成する(UTF-8)
'引数 : filePath  : ファイルパス
'     : fileNm    : ファイル名
'     : fileExt   : ファイル拡張子
'     : fileContents  : 書込内容
'     : charsetNm : 文字コード名(UTF-8 / Shift-JIS など)
'     : bomFlg    : BOM有無フラグ(True=有 / False=無)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub CreateTxtFile(ByVal filePath As String, ByVal fileNm As String, ByVal fileExt As String, ByVal contents As String, ByVal charsetNm As String, ByVal bomFlg As Boolean)
    '●ファイル書込用
    Dim stream As Object: Set stream = CreateObject("ADODB.Stream")
    Dim binStream As Object
    Dim byteData() As Byte
    Dim bomLen As Integer: bomLen = GetBomLen(charsetNm)

    '末尾付与
    If Right$(filePath, 1) <> "\" Then filePath = filePath & "\"
    '先頭付与
    If Left$(fileExt, 1) <> "." Then fileExt = "." & fileExt
    '末尾除去
    If Right$(fileNm, Len(fileExt)) = fileExt Then fileNm = Left$(fileNm, Len(fileNm) - Len(fileExt))

    With stream
        .Type = 2
        .Charset = charsetNm
        .Open
        .WriteText contents
        .SaveToFile filePath & fileNm & fileExt, 2 '既存ファイル上書き設定
        .Close
    End With
    Set stream = Nothing

    'BOMなしの場合、先頭nバイト(BOM)除去で作り直す
    If Not bomFlg And 0 < bomLen Then
        '===== 読込 =====
        Set binStream = CreateObject("ADODB.Stream")
        With binStream
            .Type = 1
            .Open
            .LoadFromFile filePath & fileNm & fileExt
            .Position = bomLen
            byteData = .Read
            .Close
        End With
        '===== 書込 =====
        With binStream
            .Type = 1
            .Open
            .Write byteData
            .SaveToFile filePath & fileNm & fileExt, 2
            .Close
        End With
        Set binStream = Nothing
    End If
End Sub
'------------------------------------------------------------
'概要 : 文字コード名のBOMバイト数を取得する
'引数 : charsetNm : 文字コード名(UTF-8 / Shift-JIS など)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Function GetBomLen(ByVal charsetNm As String) As Integer
    Select Case UCase$(charsetNm)
        Case "UTF-8"
            GetBomLen = 3
        Case "UTF-16", "UNICODE", "UTF-16LE", "UTF-16BE"
            GetBomLen = 2
        Case "UTF-32"
            GetBomLen = 4
        Case Else
            GetBomLen = 0
    End Select
End Function
'------------------------------------------------------------
'概要 : 罫線を引く(格子)
'引数 : rng : 範囲
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub DrawRuledLine_Grid(rng As Range)
    '●罫線位置区分
    Dim borderType As Variant
    '外
    rng.BorderAround LineStyle:=xlContinuous, Weight:=xlThin
    '内
    For Each borderType In Array(xlInsideHorizontal, xlInsideVertical)
        rng.Borders(borderType).LineStyle = xlContinuous
        rng.Borders(borderType).Weight = xlThin
    Next
End Sub
'------------------------------------------------------------
'概要 : MyBatisGenerator用の日時フォーマットに変換する
'     : (Thu Jan 1 00:00:00 JST 2026)
'引数 : dt : 日時
'戻値 : 英語日時文字列
'備考 : -
'------------------------------------------------------------
Function FormatEnglishDate(dt As Date) As String
    '●曜日/月
    Dim days As Variant: days = Array("Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat")
    Dim months As Variant: months = Array("Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec")
    '各要素セット (Weekdayは1=日曜なのでインデックスは -1)
    Dim w As String: w = days(Weekday(dt, vbSunday) - 1)
    Dim m As String: m = months(Month(dt) - 1)
    Dim d As String: d = Format(Day(dt), "0")
    Dim time As String: time = Format(dt, "hh:nn:ss")
    Dim tz As String: tz = "JST"
    Dim y As String: y = Year(dt)
    '結合
    FormatEnglishDate = w & " " & m & " " & d & " " & time & " " & tz & " " & y
End Function
'------------------------------------------------------------
'概要 : キャメルケースとスネークケースの変換をする
'     : [convertType=UPPER_SNAKE] の場合 "USER_ID"
'     : [convertType=LOWER_SNAKE] の場合 "user_id"
'     : [convertType=LOWER_CAMEL] の場合 "userId"
'     : [convertType=UPPER_CAMEL] の場合 "UserId"
'引数 : str         : 文字列
'     : convertType : 名称変換区分
'戻値 : 文字列
'備考 : -
'------------------------------------------------------------
Private Function ConvertCamelSnakeNm(ByVal str As String, ByVal convertType As String) As String
    Dim wordCol As Collection
    Dim word As Variant
    Dim result As String
    Dim i As Long
    '名前を単語に分解
    Set wordCol = ParseNm(str)
    If wordCol.Count <> 0 Then
        Select Case convertType
            Case U_SNAKE
                For Each word In wordCol
                    If result <> "" Then
                        result = result & "_"
                    End If
                    result = result & UCase$(CStr(word))
                Next
            Case L_SNAKE
                For Each word In wordCol
                    If result <> "" Then
                        result = result & "_"
                    End If
                    result = result & LCase$(CStr(word))
                Next
            Case L_CAMEL
                For i = 1 To wordCol.Count
                    word = LCase$(CStr(wordCol(i)))
                    If i = 1 Then
                        result = result & word
                    Else
                        result = result & UCase$(Left$(word, 1)) & Mid$(word, 2)
                    End If
                Next
            Case U_CAMEL
                For Each word In wordCol
                    word = LCase$(CStr(word))
                    result = result & UCase$(Left$(word, 1)) & Mid$(word, 2)
                Next
        End Select
    End If
    ConvertCamelSnakeNm = result
End Function
'------------------------------------------------------------
'概要 : 文字列を単語ごとに分解する
'     : USER_ID    → USER / ID
'     : userId     → user / Id
'     : UserName01 → User / Name / 01
'     : A_B_C_01   → A / B / C / 01
'引数 : str : 文字列
'戻値 : 単語リスト
'備考 : -
'------------------------------------------------------------
Private Function ParseNm(ByVal str As String) As Collection
    Dim col As New Collection
    Dim i As Long
    Dim ch As String
    Dim preCh As String
    Dim nextCh As String
    Dim crntWord As String
    str = Trim$(str)
    If str <> "" Then
        For i = 1 To Len(str)
            ch = Mid$(str, i, 1)
            If 1 < i Then
                preCh = Mid$(str, i - 1, 1)
            Else
                preCh = ""
            End If
            If i < Len(str) Then
                nextCh = Mid$(str, i + 1, 1)
            Else
                nextCh = ""
            End If
            If ch = "_" Then
                Call AddCol(col, crntWord)
                crntWord = ""
            '小文字 → 大文字
            ElseIf IsLowerCaseChar(preCh) And IsUpperCaseChar(ch) Then
                Call AddCol(col, crntWord)
                crntWord = ch
            '大文字 → 大文字 → 小文字
            ElseIf IsUpperCaseChar(preCh) And IsUpperCaseChar(ch) And IsLowerCaseChar(nextCh) Then
                Call AddCol(col, crntWord)
                crntWord = ch
            Else
                crntWord = crntWord & ch
            End If
        Next
        '最後の単語
        Call AddCol(col, crntWord)
    End If
    Set ParseNm = col
End Function
'------------------------------------------------------------
'概要 : 文字列をリストに追加
'引数 : col : リスト
'     : srt : 文字列
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub AddCol(ByRef col As Collection, ByVal str As String)
    str = Trim$(str)
    If str <> "" Then
        Call col.Add(str)
    End If
End Sub
'------------------------------------------------------------
'概要 : 文字列が全て大文字の英字か判定する
'引数 : str : 文字列
'戻値 : 判定結果(True:のみ / False:以外含む)
'備考 : -
'------------------------------------------------------------
Private Function IsUpperCaseChar(ByVal str As String) As Boolean
    If Len(str) <> 1 Then Exit Function
    IsUpperCaseChar = (str Like "[A-Z]")
End Function
'------------------------------------------------------------
'概要 : 文字列が全て小文字の英字か判定する
'引数 : str : 文字列
'戻値 : 判定結果(True:のみ / False:以外含む)
'備考 : -
'------------------------------------------------------------
Private Function IsLowerCaseChar(ByVal str As String) As Boolean
    If Len(str) <> 1 Then Exit Function
    IsLowerCaseChar = (str Like "[a-z]")
End Function
'------------------------------------------------------------
'概要 : 対応するJDBCタイプ名を取得する
'引数 : colType : 項目型名
'戻値 : JDBCタイプ名
'備考 : -
'------------------------------------------------------------
Private Function GetJdbcType(ByVal colType As String) As String
    Dim result As String
    Select Case colType
        '========== Oracle ==========
        Case "VARCHAR2"
            result = "VARCHAR"
        Case "NVARCHAR2"
            result = "NVARCHAR"
        Case "NUMBER"
            result = "NUMERIC"
        Case "BLOB"
            result = "BLOB"
        Case "RAW"
            result = "VARBINARY"
        '========== PostgreSQL ==========
        Case "character varying", "text"
            result = "VARCHAR"
        Case "character"
            result = "CHAR"
        Case "numeric"
            result = "NUMERIC"
        Case "timestamp with time zone", "timestamp without time zone"
            result = "TIMESTAMP"
        Case "date"
            result = "DATE"
        Case "boolean"
            result = "BIT"
        Case "bytea"
            result = "BINARY"
        '========== SQLServer ==========
        Case "varchar"
            result = "VARCHAR"
        Case "nvarchar"
            result = "NVARCHAR"
        Case "numeric"
            result = "NUMERIC"
        Case "datetime2"
            result = "TIMESTAMP"
        'Case "date"
        '    result = "DATE"
        Case "bit"
            result = "BIT"
        Case "varbinary"
            result = "VARBINARY"
        Case "xml"
            result = "LONGNVARCHAR"
        Case "uniqueidentifier"
            result = "CHAR"
    End Select
    '特殊日時の場合
    If 0 < InStr(colType, "TIMESTAMP") Or colType = "DATE" Or colType = "datetimeoffset" Then result = "TIMESTAMP"
    'その他の場合
    If result = "" Then result = "OTHER"
    GetJdbcType = result
End Function
