'------------------------------------------------------------
'概要 : Sheet[勤務表/祝日]を作成する
'引数 : -
'戻値 : -
'備考 : -
'------------------------------------------------------------
Public Sub CreateSheet_WorkSchedule()
    '処理中描画OFF設定
    Call SettingRenderingProcess(False)

    '●Loopカウンタ [i/j/k]
    Dim i As Long, j As Long, k As Long
    '●開始行/開始列/終了行/終了列
    Dim startRowIdx As Long
    Dim startColIdx As Long
    Dim endRowIdx As Long
    Dim endColIdx As Long
    '●範囲
    Dim rng As Range

    '●新規Sheet/Sheet名
    Dim newSheet As Worksheet
    Dim newSheetNm As String

    '●入力受付Form表示内容
    Dim title As String
    Dim msg As String
    '●[Sheetタイプ]入力値/初期値/入力可能範囲
    Dim inputSheetType As Long
    Dim initSheetType As Double: initSheetType = "1"
    Dim minSheetType As Double: minSheetType = "1"
    Dim maxSheetType As Double: maxSheetType = "2"
    '●[作成年]入力値/初期値/入力可能範囲(1年前～10年後)
    Dim inputYear As Long
    Dim initYear As Double: initYear = Year(Now)
    Dim minYear As Double: minYear = Year(Now) - 1
    Dim maxYear As Double: maxYear = Year(Now) + 10
    '●[時間(勤務/休憩)]入力値/初期値/入力可能範囲(0h～24h)
    Dim inputWhStartTime As String
    Dim inputWhEndTime As String
    Dim inputBtStartTime As String
    Dim inputBtEndTime As String
    Dim initWhStartTime As String: initWhStartTime = "9:00"
    Dim initWhEndTime As String: initWhEndTime = "18:00"
    Dim initBtStartTime As String: initBtStartTime = "12:00"
    Dim initBtEndTime As String: initBtEndTime = "13:00"
    Dim minTime As String: minTime = "0:00"
    Dim maxTime As String: maxTime = "23:59"
    '●[勤務時間数]入力値/初期値/入力可能範囲(0h～24h)
    Dim inputWorkHr As Double
    Dim initWorkHr As Double: initWorkHr = 8
    Dim minHr As Double: minHr = 0
    Dim maxHr As Double: maxHr = 24
    '●[テーマ]入力値/初期値/入力可能範囲
    Dim inputThemeNo As Long
    Dim initThemeNo As Double: initThemeNo = 1
    Dim minThemeNo As Double: minThemeNo = 1
    Dim maxThemeNo As Double: maxThemeNo = 6

    '●祝日辞書(検索用)
    Dim holidayDict As Object: Set holidayDict = CreateObject("Scripting.Dictionary")

    '▼入力受付[Sheetタイプ]
    title = "作成する勤務表タイプを数字で選択して下さい"
    msg = "1：[開始/終了]時間入力タイプ" & vbCrLf & _
          "2：1日の勤務時間数入力タイプ"
    inputSheetType = Int(InputForm_Num(title, msg, initSheetType, minSheetType, maxSheetType))
    'キャンセルの場合
    If inputSheetType = -1 Then Exit Sub
    '▼入力受付[作成年]
    title = "作成する年を入力して下さい"
    msg = ""
    inputYear = Int(InputForm_Num(title, msg, initYear, minYear, maxYear))
    If inputYear = -1 Then Exit Sub
    Select Case inputSheetType
        '===== 時間タイプの場合 =====
        Case "1"
            '▼入力受付[勤務時間(開始/終了)]
            title = "勤務 [開始] 時間を入力して下さい"
            msg = ""
            inputWhStartTime = InputForm_Time(title, msg, initWhStartTime, minTime, maxTime)
            If inputWhStartTime = "-1" Then Exit Sub
            title = "勤務 [終了] 時間を入力して下さい"
            inputWhEndTime = InputForm_Time(title, msg, initWhEndTime, minTime, maxTime)
            If inputWhEndTime = "-1" Then Exit Sub
            '▼入力受付[休憩時間(開始/終了)]
            title = "休憩 [開始] 時間を入力して下さい"
            inputBtStartTime = InputForm_Time(title, msg, initBtStartTime, minTime, maxTime)
            If inputBtStartTime = "-1" Then Exit Sub
            title = "休憩 [終了] 時間を入力して下さい"
            inputBtEndTime = InputForm_Time(title, msg, initBtEndTime, minTime, maxTime)
            If inputBtEndTime = "-1" Then Exit Sub
        '===== 時間数タイプの場合 =====

            '▼入力受付[勤務時間数]
            title = "1日の勤務時間数を入力して下さい"
            msg = ""
            inputWorkHr = InputForm_Num(title, msg, initWorkHr, minHr, maxHr)
            If inputWorkHr = -1 Then Exit Sub
    End Select

    msg = "1：Standard" & vbCrLf & _
          "2：Light" & vbCrLf & _
          "3：Dark" & vbCrLf & _
          "4：Sweets" & vbCrLf & _
          "5：Chiikawa" & vbCrLf & _
          "6：Ocean"
    inputThemeNo = Int(InputForm_Num(title, msg, initThemeNo, minThemeNo, maxThemeNo))
    If inputThemeNo = -1 Then Exit Sub

    '========== Sheet[祝日]作成 ==========
    '(存在する場合、既存セット)
    newSheetNm = "祝日"
    If IsExistsSheet(newSheetNm) Then
        Set newSheet = Sheets(newSheetNm)
    Else
        Set newSheet = CreateSheet_Holiday(newSheetNm, minYear, maxYear)
    End If

    '========== 祝日辞書 作成 ==========
    endRowIdx = newSheet.Cells(newSheet.Rows.Count, "B").End(xlUp).Row
    For Each rng In newSheet.Range("B2:B" & endRowIdx)
        If IsDate(rng.Value) Then holidayDict(CLng(CDate(rng.Value))) = True
    Next

    '========== Sheet[勤務表]作成 ==========
    Select Case inputSheetType
        Case "1"
            Call CreateSheet_WorkSchedule_01(holidayDict, inputYear, inputWhStartTime, inputWhEndTime, inputBtStartTime, inputBtEndTime, inputThemeNo)
        Case "2"
            Call CreateSheet_WorkSchedule_02(holidayDict, inputYear, inputWorkHr, inputThemeNo)
    End Select

    '処理中描画ON設定
    Call SettingRenderingProcess(True)
End Sub
'------------------------------------------------------------
'概要 : 処理中の画面描画を設定する
'引数 : flg(True:ON / False:OFF)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub SettingRenderingProcess(ByVal flg As Boolean)
    Application.ScreenUpdating = flg '画面再描画
    Application.DisplayAlerts = flg  '確認ダイアログ表示
    Application.EnableEvents = flg   'イベント発火
    '再計算モード(セル値が変わるたびに再計算)
    If flg Then
        Application.Calculation = xlCalculationAutomatic '自動
    Else
        Application.Calculation = xlCalculationManual '手動
    End If
End Sub
'------------------------------------------------------------
'概要 : ファイルが存在するか判定する
'引数 : path   : ファイルまでのパス(絶対パス)
'     : fileNm : ファイル名
'     : ext    : 拡張子
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsFile(ByVal path As String, ByVal fileNm As String, ByVal ext As String) As Boolean
    'パス末尾付与
    If Right$(path, 1) <> "\" Then path = path & "\"
    '判定
    IsExistsFile = (Dir(path & fileNm & "." & ext) <> "")
End Function
'------------------------------------------------------------
'概要 : 重複しないファイル名となるまで、末尾に連番を付与する
'     : (同じ名前のファイルが存在しない場合、そのまま返す)
'     : (同じ名前のファイルが存在する場合、2からの連番を末尾に付与し、重複しないファイル名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : path   : ファイルまでのパス
'     : fileNm : ファイル名
'     : ext    : 拡張子
'戻値 : ファイル名(拡張子付き)
'備考 : -
'------------------------------------------------------------
Private Function ChangeUniqueFileNm(ByVal path As String, ByVal fileNm As String, ByVal ext As String) As String
    Dim i As Long
    Dim uniqueNo As String
    '存在しない場合
    If Not IsExistsFile(path, fileNm, ext) Then
        ChangeUniqueFileNm = fileNm & "." & ext
        Exit Function
    End If
    '存在する場合
    For i = 2 To 1000
        '末尾付与
        uniqueNo = " (" & i & ")"
        '存在しない場合
        If Not IsExistsFile(path, fileNm & uniqueNo, ext) Then
            ChangeUniqueFileNm = fileNm & uniqueNo & "." & ext
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : Sheet名のSheetが存在するか判定する
'引数 : wsNm : Sheet名
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsSheet(ByVal wsNm As String) As Boolean
    Dim ws As Worksheet
    For Each ws In Sheets
        If ws.Name = wsNm Then
            IsExistsSheet = True
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : 重複しないSheet名となるまで、末尾に連番を付与する
'     : (同じ名前のSheetが存在しない場合、そのまま返す)
'     : (同じ名前のSheetが存在する場合、2からの連番を末尾に付与し、重複しないSheet名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : wsNm : Sheet名
'戻値 : Sheet名
'備考 : -
'------------------------------------------------------------
Private Function ChangeUniqueSheetNm(ByVal wsNm As String) As String
    Dim i As Long
    Dim uniqueNo As String
    '存在しない場合
    If Not IsExistsSheet(wsNm) Then
        ChangeUniqueSheetNm = wsNm
        Exit Function
    End If
    '存在する場合
    For i = 2 To 1000
        '末尾付与
        uniqueNo = " (" & i & ")"
        '存在しない場合
        If Not IsExistsSheet(wsNm & uniqueNo) Then
            ChangeUniqueSheetNm = wsNm & uniqueNo
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : 列番号から列名(アルファベット)を取得する
'     : (範囲外の数値の場合、空を返す)
'引数 : colIdx : 列番号
'戻値 : 列名(アルファベット)
'備考 : -
'------------------------------------------------------------
Private Function GetColNm(ByVal colIdx As Long) As String
    Dim result As String
    '範囲外の場合
    If colIdx < 1 Or Application.Columns.Count < colIdx Then
        GetColNm = ""
        Exit Function
    End If
    Do While 0 < colIdx
        colIdx = colIdx - 1
        result = Chr$(65 + (colIdx Mod 26)) & result
        colIdx = colIdx \ 26
    Loop
    GetColNm = result
End Function
'------------------------------------------------------------
'概要 : 列名(アルファベット)から列番号を取得する
'     : (範囲外の列名の場合、0を返す)
'引数 : colNm : 列名(アルファベット)
'戻値 : 列番号
'備考 : -
'------------------------------------------------------------
Private Function GetColIdx(ByVal colNm As String) As Long
    Dim i As Long
    Dim result As Long
    Dim baseA As Long: baseA = Asc("A") - 1
    Dim colNmLen As Long: colNmLen = Len(UCase$(colNm))
    Dim ch As Long
    For i = 1 To colNmLen
        ch = Asc(mid$(colNm, i, 1)) - baseA
        '範囲外の場合
        If ch < 1 Or 26 < ch Then
            GetColIdx = 0
            Exit Function
        End If
        result = result * 26 + ch
    Next
    GetColIdx = result
End Function
'------------------------------------------------------------
'概要 : 文字列を桁数と同じ桁数になるまで先頭にchを付け足す
'引数 : str      : 文字列
'     : digitNum : 桁数
'     : ch       : 挿入文字
'戻値 : 文字列
'備考 : -
'------------------------------------------------------------
Private Function AddFrontChara(ByVal str As String, ByVal digitNum As Long, ByVal ch As String) As String
    Do While Len(str) < digitNum
        str = ch & str
    Loop
    AddFrontChara = str
End Function
'------------------------------------------------------------
'概要 : 文字列をCSV出力値に変換する
'     : 1,改行を除去する
'     : 2,ダブルクォーテーションをエスケープする
'引数 : str : 文字列
'戻値 : CSV出力値
'備考 : -
'------------------------------------------------------------
Private Function ConvertCsvVal(ByVal str As String) As String
    '改行 除去
    str = Replace$(str, vbCr, "")
    str = Replace$(str, vbLf, "")
    str = Replace$(str, vbCrLf, "")
    'ダブルクォーテーション エスケープ
    str = Replace$(str, """", """""")
    ConvertCsvVal = str
End Function
'------------------------------------------------------------
'概要 : テーブルスタイル適用[01]
'引数 : ws     : Sheet
'     : rng    : 適用範囲
'     : colIdx : ウィンドウ枠の固定列番号
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub ApplyTableStyle_01(ByRef ws As Worksheet, ByRef rng As Range, ByVal colIdx As Long)
    'テーブルスタイル適用
    Dim tbl As ListObject: Set tbl = ws.ListObjects.Add(xlSrcRange, rng, , xlYes)
    tbl.TableStyle = "TableStyleMedium1"
    'ヘッダー
    With ws.Range(ws.Cells(rng.Row, rng.Column), ws.Cells(rng.Row, rng.Columns.Count))
        .Font.Color = RGB(0, 0, 0)           '#000000 文字色
        .Interior.Color = RGB(221, 235, 247) '#DDEBF7 背景色
        .HorizontalAlignment = xlCenter      '文字揃え
    End With
    '罫線
    rng.Borders(xlInsideHorizontal).Weight = xlHairline
    rng.Borders(xlInsideHorizontal).ColorIndex = xlAutomatic
    'ウィンドウ枠の固定
    ws.Activate
    ws.Cells(rng.Row + 1, rng.Column + colIdx - 1).Select
    ActiveWindow.FreezePanes = True
End Sub
'------------------------------------------------------------
'概要 : 罫線(行境界線)[01]
'     : 境界判定列のセル値が変わる行の境界に罫線を引く
'引数 : ws     : Sheet
'     : rng    : 適用範囲
'     : colIdx : 境界判定列番号
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub ApplyLineStyle_01(ByRef ws As Worksheet, ByRef rng As Range, ByVal colIdx As Long)
    Dim i As Long
    '●名称(前/現在)
    Dim preVal As String
    Dim crntVal As String
    preVal = Left$(CStr(ws.Cells(rng.Row + 1, rng.Column + colIdx - 1)), 4)
    For i = rng.Row + 1 To rng.Row + rng.Rows.Count
        crntVal = Left$(CStr(ws.Cells(i, rng.Column + colIdx - 1)), 4)
        '値が異なる場合
        If preVal <> crntVal Then
            '罫線
            ws.Range(ws.Cells(i, rng.Column), ws.Cells(i, rng.Columns.Count)).Borders(xlEdgeTop).LineStyle = xlDouble
            ws.Range(ws.Cells(i, rng.Column), ws.Cells(i, rng.Columns.Count)).Borders(xlEdgeTop).Weight = xlThick
            preVal = crntVal
        End If
    Next
End Sub

'------------------------------------------------------------
'概要 : 入力受付Form(数値)
'     : (キャンセル選択 or 範囲外エラーの場合 -1を返す)
'引数 : title   : タイトル文
'     : msg     : 説明文
'     : initVal : 初期値
'     : minVal  : 有効範囲(最小)
'     : maxVal  : 有効範囲(最大)
'戻値 : 入力値
'備考 : -
'------------------------------------------------------------
Private Function InputForm_Num(ByVal title As String, ByVal msg As String, ByVal initVal As String, ByVal minVal As Double, ByVal maxVal As Double) As Double
    '●入力値
    Dim inputVal As Variant
    Dim inputValDbl As Double
    '説明文後付け
    If msg = "" Then
        msg = "入力可能範囲（" & minVal & "～" & maxVal & "）"
    Else
        msg = msg & vbCrLf & "入力可能範囲（" & minVal & "～" & maxVal & "）"
    End If
    '入力受付
    inputVal = Application.InputBox(Prompt:=msg, title:=title, Default:=initVal, Type:=1)
    'キャンセルの場合
    If VarType(inputVal) = vbBoolean And inputVal = False Then
        InputForm_Num = -1
        Exit Function
    End If
    '小数点以下有りとして受付
    inputValDbl = CDbl(inputVal)
    '範囲チェック
    If inputValDbl < minVal Or maxVal < inputValDbl Then
        MsgBox "範囲内で入力して下さい（" & minVal & " ～ " & maxVal & "）", vbExclamation
        InputForm_Num = -1
        Exit Function
    End If
    InputForm_Num = inputValDbl
End Function
'------------------------------------------------------------
'概要 : 入力受付Form(時間)
'     : (キャンセル選択 or 範囲外エラーの場合 "-1"を返す)
'引数 : title   : タイトル文
'     : msg     : 説明文
'     : initVal : 初期値
'     : minVal  : 有効範囲(最小)
'     : maxVal  : 有効範囲(最大)
'戻値 : 入力値
'備考 : -
'------------------------------------------------------------
Private Function InputForm_Time(ByVal title As String, ByVal msg As String, ByVal initVal As String, ByVal minVal As String, ByVal maxVal As String) As String
    '●入力値
    Dim inputVal As Variant
    '説明文後付け
    If msg = "" Then
        msg = "入力可能範囲（" & minVal & "～" & maxVal & "）"
    Else
        msg = msg & vbCrLf & "入力可能範囲（" & minVal & "～" & maxVal & "）"
    End If
    '入力受付
    inputVal = Application.InputBox(Prompt:=msg, title:=title, Default:=initVal, Type:=2)
    'キャンセルの場合
    If VarType(inputVal) = vbBoolean And inputVal = False Then
        InputForm_Time = "-1"
        Exit Function
    End If
    inputVal = CStr(inputVal)
    'フォーマットチェック
    If InStr(inputVal, ":") = 0 Or Len(inputVal) < 4 Or 5 < Len(inputVal) Then
        MsgBox "24時間形式で入力して下さい（例：9:00, 13:00）", vbExclamation
        InputForm_Time = "-1"
        Exit Function
    End If
    '範囲チェック
    If CLng(Replace$(inputVal, ":", "")) < CLng(Replace$(minVal, ":", "")) Or CLng(Replace$(maxVal, ":", "")) < CLng(Replace$(inputVal, ":", "")) Then
        MsgBox "範囲内で入力して下さい（" & minVal & " ～ " & maxVal & "）", vbExclamation
        InputForm_Time = "-1"
        Exit Function
    End If
    InputForm_Time = inputVal
End Function
'------------------------------------------------------------
'概要 : Sheet[祝日]を作成する
'引数 : wsNm    : 作成Sheet名
'     : minYear : 作成年(開始)
'     : maxYear : 作成年(終了)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Function CreateSheet_Holiday(ByVal wsNm As String, ByVal minYear As Long, ByVal maxYear As Long) As Worksheet
    Dim i As Long         'Loopカウンタ
    Dim ws As Worksheet   'Sheet
    Dim endRowIdx As Long '終了行
    Dim endColIdx As Long '終了列
    Dim rng As Range      '範囲
    '●祝日Map[日付/祝日名]/行
    Dim holidayMap As New Collection
    Dim holidayArr As Variant
    '●日数
    Dim dateCnt As Integer: dateCnt = 1
    '●前年/今年
    Dim preYear As Long
    Dim crntYear As Long

    'Sheet作成 & Sheet名セット
    Set ws = ThisWorkbook.Sheets.Add(Before:=ThisWorkbook.Worksheets(1))
    ws.Name = wsNm
    '祝日Map 作成
    Set holidayMap = CreateHolidayMap(minYear, maxYear)
    '祝日 書込
    preYear = minYear '現在年
    For i = 1 To holidayMap.Count
        holidayArr = holidayMap(i)
        '月日/祝日名 書込
        ws.Range("B" & i + 1).Value = holidayArr(0)
        ws.Range("C" & i + 1).Value = holidayArr(1)
        crntYear = Year(ws.Range("B" & i + 1).Value2)
        '日数 書込
        If preYear = crntYear Then
            ws.Range("D" & i + 1).Value = CStr(dateCnt)
            dateCnt = dateCnt + 1
        Else
            ws.Range("D" & i + 1).Value = 1
            preYear = Year(ws.Range("B" & i + 1).Value2)
            dateCnt = 2
        End If
    Next
    '年 書込
    ws.Range("A2:A" & holidayMap.Count + 1).FormulaR1C1 = "=RC[1]"
    'ヘッダー 書込
    ws.Range("A1").Value = "年"
    ws.Range("B1").Value = "月日"
    ws.Range("C1").Value = "祝日名"
    ws.Range("D1").Value = "日数"
    '表示形式
    ws.Range("A2:A" & holidayMap.Count + 1).NumberFormatLocal = "yyyy"
    ws.Range("B2:B" & holidayMap.Count + 1).NumberFormatLocal = "m/d"

    '終了行セット
    endRowIdx = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    '表範囲セット
    Set rng = ws.Range("A1:D" & endRowIdx)

    'Sheet設定
    ws.Activate
    'テーブルスタイル/罫線
    Call ApplyTableStyle_01(ws, rng, 1)
    Call ApplyLineStyle_01(ws, rng, 1)
    With ws
        'Tab色
        .Tab.Color = RGB(240, 144, 144) '#F09090
        '列幅
        .Columns("A:B").ColumnWidth = 8
        .Columns("C").ColumnWidth = 12
        .Columns("D").ColumnWidth = 8
        '文字揃え
        .Cells.VerticalAlignment = xlTop
        .Cells.HorizontalAlignment = xlLeft
        .Columns("A:B").HorizontalAlignment = xlCenter
        .Range("C1").HorizontalAlignment = xlCenter
        .Columns("D").HorizontalAlignment = xlCenter
    End With
    '目盛線
    ActiveWindow.DisplayGridlines = False
    'ウィンドウ枠の固定
    ws.Range("A2").Select
    ActiveWindow.FreezePanes = True

    Set CreateSheet_Holiday = ws
End Function
'------------------------------------------------------------
'概要 : 祝日Mapを作成する
'引数 : minYear : 作成年(開始)
'     : maxYear : 作成年(終了)
'戻値 : 祝日Map
'備考 : -
'------------------------------------------------------------
Private Function CreateHolidayMap(ByVal minYear As Long, ByVal maxYear As Long) As Collection
    Dim i As Long
    '●祝日Map/行
    Dim holidayMap As New Collection
    Dim holidayArr As Variant
    '●作成日付(開始/終了)
    Dim startDate As Date: startDate = DateSerial(minYear, 1, 1)
    Dim endDate As Date: endDate = DateSerial(maxYear, 12, 31)
    '●年/日
    Dim y As Long
    Dim d As Date
    '========== 法定祝日 追加 ==========
    For y = Year(startDate) To Year(endDate)
        Call AddRowToHolidayMap(holidayMap, DateSerial(y, 1, 1), "元日")
        Call AddRowToHolidayMap(holidayMap, GetDateOfWeekForMonth(y, 1, 2, vbMonday), "成人の日")
        Call AddRowToHolidayMap(holidayMap, DateSerial(y, 2, 11), "建国記念の日")
        Call AddRowToHolidayMap(holidayMap, DateSerial(y, 2, 23), "天皇誕生日")
        Call AddRowToHolidayMap(holidayMap, GetDateOfVernalEquinox(y), "春分の日")
        Call AddRowToHolidayMap(holidayMap, DateSerial(y, 4, 29), "昭和の日")
        Call AddRowToHolidayMap(holidayMap, DateSerial(y, 5, 3), "憲法記念日")
        Call AddRowToHolidayMap(holidayMap, DateSerial(y, 5, 4), "みどりの日")
        Call AddRowToHolidayMap(holidayMap, DateSerial(y, 5, 5), "こどもの日")
        Call AddRowToHolidayMap(holidayMap, GetDateOfWeekForMonth(y, 7, 3, vbMonday), "海の日")
        Call AddRowToHolidayMap(holidayMap, DateSerial(y, 8, 11), "山の日")
        Call AddRowToHolidayMap(holidayMap, GetDateOfWeekForMonth(y, 9, 3, vbMonday), "敬老の日")
        Call AddRowToHolidayMap(holidayMap, GetDateOfAutumnalEquinox(y), "秋分の日")
        Call AddRowToHolidayMap(holidayMap, GetDateOfWeekForMonth(y, 10, 2, vbMonday), "スポーツの日")
        Call AddRowToHolidayMap(holidayMap, DateSerial(y, 11, 3), "文化の日")
        Call AddRowToHolidayMap(holidayMap, DateSerial(y, 11, 23), "勤労感謝の日")
    Next
    '========== 振替休日 追加 ==========
    '(祝日が日曜日の場合、翌平日を休日とする)
    For i = 1 To holidayMap.Count
        holidayArr = holidayMap(i)
        d = holidayArr(0) '法定祝日
        '法定祝日が日曜日の場合、翌日セット
        If weekday(d, vbSunday) = vbSunday Then
            d = d + 1
            '翌日も祝日の場合、翌平日セット
            Do While IsExistsHolidayOnHolidayMap(holidayMap, d)
                d = d + 1
            Loop
            Call AddRowToHolidayMap(holidayMap, d, "振替休日")
        End If
    Next
    '========== 国民の祝日 追加 ==========
    '(祝日に挟まれた平日は休日とする)
    Dim holidayArr1 As Variant, holidayArr2 As Variant
    '昇順ソート
    Call SortHolidayMap(holidayMap)
    For i = 1 To holidayMap.Count - 1
        holidayArr1 = holidayMap(i)
        holidayArr2 = holidayMap(i + 1)
        '1日だけ間が空いている場合
        If holidayArr2(0) - holidayArr1(0) = 2 Then
            Call AddRowToHolidayMap(holidayMap, holidayArr1(0) + 1, "国民の祝日")
        End If
    Next
    '昇順ソート(再)
    Call SortHolidayMap(holidayMap)

    Set CreateHolidayMap = holidayMap
End Function
'------------------------------------------------------------
'概要 : 祝日Mapに行を追加する(重複しない場合)
'引数 : holidayMap : 祝日Map
'     : dt         : 祝日
'     : nm         : 祝日名
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub AddRowToHolidayMap(ByRef holidayMap As Collection, ByVal dt As Date, ByVal nm As String)
    '重複しない日付の場合
    If Not IsExistsHolidayOnHolidayMap(holidayMap, dt) Then holidayMap.Add Array(dt, nm)
End Sub
'------------------------------------------------------------
'概要 : 祝日Mapに存在する日付か判定する
'引数 : holidayMap : 祝日Map
'     : dt         : 日付
'戻値 : 判定結果(True=する / False=しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsHolidayOnHolidayMap(ByRef holidayMap As Collection, ByVal dt As Date) As Boolean
    Dim i As Long
    For i = 1 To holidayMap.Count
        If holidayMap(i)(0) = dt Then
            IsExistsHolidayOnHolidayMap = True
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : y年のm月の第n w曜日の日付を取得する
'引数 : y : 年
'     : m : 月
'     : n : 第n
'     : w : 曜日
'戻値 : 日付
'備考 : -
'------------------------------------------------------------
Private Function GetDateOfWeekForMonth(ByVal y As Long, ByVal m As Long, ByVal n As Long, ByVal w As VbDayOfWeek) As Date
    Dim d As Date: d = DateSerial(y, m, 1)
    Do While weekday(d, vbSunday) <> w
        d = d + 1
    Loop
    GetDateOfWeekForMonth = d + (n - 1) * 7
End Function
'------------------------------------------------------------
'概要 : 春分の日を取得する(簡易近似式)
'引数 : y : 年
'戻値 : 日付(推測日)
'備考 : 春分の日および秋分の日は、国立天文台の計算結果に基づき､官報で告示された日とする
'------------------------------------------------------------
Private Function GetDateOfVernalEquinox(ByVal y As Long) As Date
    Dim d As Integer
    Select Case y
        Case 2000 To 2099
            d = Int(20.8431 + 0.242194 * (y - 1980) - Int((y - 1980) / 4))
        Case Else
            d = 20
    End Select
    GetDateOfVernalEquinox = DateSerial(y, 3, d)
End Function
'------------------------------------------------------------
'概要 : 秋分の日を取得する(簡易近似式)
'引数 : y : 年
'戻値 : 日付(推測日)
'備考 : 春分の日および秋分の日は、国立天文台の計算結果に基づき､官報で告示された日とする
'------------------------------------------------------------
Private Function GetDateOfAutumnalEquinox(ByVal y As Long) As Date
    Dim d As Integer
    Select Case y
        Case 2000 To 2099
            d = Int(23.2488 + 0.242194 * (y - 1980) - Int((y - 1980) / 4))
        Case Else
            d = 23
    End Select
    GetDateOfAutumnalEquinox = DateSerial(y, 9, d)
End Function
'------------------------------------------------------------
'概要 : 昇順ソート(祝日Map用)
'引数 : holidayMap : 祝日Map
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub SortHolidayMap(ByRef holidayMap As Collection)
    Dim i As Long, j As Long
    Dim tmp As Variant
    '●祝日Map行
    Dim holidayArr As Variant
    'Collectionから配列に変換
    ReDim holidayArr(1 To holidayMap.Count)
    For i = 1 To holidayMap.Count
        holidayArr(i) = holidayMap(i)
    Next
    'バブルソート(日付で昇順)
    For i = 1 To UBound(holidayArr) - 1
        For j = i + 1 To UBound(holidayArr)
            '次行の方が小さい場合、入替
            If holidayArr(j)(0) < holidayArr(i)(0) Then
                tmp = holidayArr(i)
                holidayArr(i) = holidayArr(j)
                holidayArr(j) = tmp
            End If
        Next
    Next
    '初期化
    Set holidayMap = New Collection
    '配列からCollectionに変換
    For i = 1 To UBound(holidayArr)
        holidayMap.Add holidayArr(i)
    Next
End Sub

'概要 : CSVファイル作成
'     : 対象月の勤務表をCsvファイルで、このExcelファイルと同じ場所に作成する
'引数 : createMonth : 作成月
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub CreateCsv(ByVal createMonth As Long)
    '処理中描画OFF設定
    Call SettingRenderingProcess(False)

    '●Loopカウンタ
    Dim i As Long
    '●列
    Dim colIdx As Long: colIdx = 1
    If 2 <= createMonth Then colIdx = colIdx + ((createMonth - 1) * 16)

    '●Excelファイル名
    Dim wbNm As String
    '●ファイルパス/ファイル名/拡張子
    Dim filePath As String: filePath = ThisWorkbook.path & "\"
    Dim fileNm As String
    Dim fileExt As String
    '●CSV内容/1行/文字列
    Dim csvBody As String
    Dim line As String
    Dim str As Variant
    '●コピー元範囲
    Dim sourceRng As Range

    '●項目一覧
    Dim header As Variant: header = Array("日付", "勤務種別", "[実]勤務開始", "[実]勤務終了", "[実]休憩開始", "[実]休憩終了", "[実]時間数", "[実]合計日数", "[実]合計時間数", "[予]勤務開始", "[予]勤務終了", "[予]休憩開始", "[予]休憩終了", "[予]時間数", "[予]合計日数", "[予]合計時間数", "備考", "出社日数", "リモート日数", "出社率")

    'Excelファイル名/ファイルパス/拡張子 セット
    wbNm = ThisWorkbook.Name
    filePath = ThisWorkbook.path & "\"
    fileExt = "csv"

    '========== 新規 [ファイル名] 作成 ==========
    '拡張子除去
    fileNm = Left$(wbNm, InStrRev(wbNm, ".") - 1)
    fileNm = "勤務表_" & ActiveSheet.Name & "年" & createMonth & "月_" & mid$(fileNm, InStrRev(fileNm, "_") + 1)
    fileNm = ChangeUniqueFileNm(filePath, fileNm, fileExt)
    '=============================================

    'コピー元 取得
    Set sourceRng = ActiveSheet.Range(GetColNm(colIdx) & "1:" & GetColNm(14 + colIdx) & 40)

    'ヘッダー作成
    line = ""
    For Each str In header
        line = line & """" & CStr(str) & ""","
    Next
    '末尾除去
    line = Left$(line, Len(line) - 1)
    csvBody = line & vbCrLf

    'データ作成
    For i = 1 To 31
        line = ""
        line = line & """" & sourceRng.Range("B" & i + 9).Value & ""","                      '日付
        line = line & """" & sourceRng.Range("D" & i + 9).text & ""","                       '勤務種別
        line = line & """" & sourceRng.Range("E" & i + 9).text & ""","                       '[実]勤務開始
        line = line & """" & sourceRng.Range("F" & i + 9).text & ""","                       '[実]勤務終了
        line = line & """" & sourceRng.Range("G" & i + 9).text & ""","                       '[実]休憩開始
        line = line & """" & sourceRng.Range("H" & i + 9).text & ""","                       '[実]休憩終了
        line = line & """" & Replace(sourceRng.Range("I" & i + 9).text, "h", "") & ""","     '[実]時間数
        If i = 1 Then
            line = line & """" & Replace(sourceRng.Range("G5").text, " 日", "") & ""","      '[実]合計日数
            line = line & """" & Replace(sourceRng.Range("H5").text, " h", "") & ""","       '[実]合計時間数
        Else
            line = line & """""," & """"","
        End If
        line = line & """" & sourceRng.Range("J" & i + 9).text & ""","                       '[予]勤務開始
        line = line & """" & sourceRng.Range("K" & i + 9).text & ""","                       '[予]勤務終了
        line = line & """" & sourceRng.Range("L" & i + 9).text & ""","                       '[予]休憩開始
        line = line & """" & sourceRng.Range("M" & i + 9).text & ""","                       '[予]休憩終了
        line = line & """" & Replace(sourceRng.Range("N" & i + 9).text, "h", "") & ""","     '[予]時間数
        If i = 1 Then
            line = line & """" & Replace(sourceRng.Range("J5").text, " 日", "") & ""","      '[予]合計日数
            line = line & """" & Replace(sourceRng.Range("K5").text, " h", "") & ""","       '[予]合計時間数
        Else
            line = line & """""," & """"","
        End If
        line = line & """" & ConvertCsvVal(sourceRng.Range("O" & i + 9).text) & ""","        '備考
        If i = 1 Then
            line = line & """" & sourceRng.Range("E3").text & ""","                          '出社日数
            line = line & """" & sourceRng.Range("E4").text & ""","                          'リモート日数
            line = line & """" & sourceRng.Range("E5").text & """"                           '出社率)
        Else
            line = line & """""," & """""," & """"""
        End If
        csvBody = csvBody & line & vbCrLf
    Next

    'ファイル作成(UTF-8(BOMあり))
    Call CreateCsvUTF8(filePath, fileNm, fileExt, csvBody, True)

    '処理中描画ON設定
    Call SettingRenderingProcess(True)
End Sub
'------------------------------------------------------------
'概要 : CSVファイルを作成する(UTF-8)
'引数 : filePath : ファイルパス
'     : fileNm   : ファイル名
'     : fileExt  : ファイル拡張子
'     : contents : 書込内容
'     : bomFlg   : BOM有無フラグ(True=有 / False=無)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub CreateCsvUTF8(ByVal filePath As String, ByVal fileNm As String, ByVal fileExt As String, ByVal contents As String, ByVal bomFlg As Boolean)
    '●CSV書込用
    Dim stream As Object: Set stream = CreateObject("ADODB.Stream")
    Dim binStream As Object
    Dim byteData() As Byte

    '末尾付与
    If Right$(filePath, 1) <> "\" Then filePath = filePath & "\"
    '先頭付与
    If Left$(fileExt, 1) <> "." Then fileExt = "." & fileExt

    With stream
        .Type = 2
        .Charset = "UTF-8"
        .Open
        .WriteText contents
        .SaveToFile filePath & fileNm, 2 '既存ファイル上書き設定
        .Close
    End With
    Set stream = Nothing

    'BOMなしの場合、先頭3バイト(BOM)除去で作り直す
    If Not bomFlg Then
        '===== 読込 =====
        Set binStream = CreateObject("ADODB.Stream")
        With binStream
            .Type = 1
            .Open
            .LoadFromFile filePath & fileNm
            .Position = 3
            byteData = .Read
            .Close
        End With
        '===== 書込 =====
        With binStream
            .Type = 1
            .Open
            .Write byteData
            .SaveToFile filePath & fileNm, 2
            .Close
        End With
        Set binStream = Nothing
    End If
End Sub

    '=============================================

    'コピー元 取得
    Set sourceRng = ActiveSheet.Range(GetColNm(colIdx) & "1:" & GetColNm(14 + colIdx) & 40)

    '新規 [ファイル/Sheet] 作成 & Sheet名セット
    Set newBook = Workbooks.Add(xlWBATWorksheet)
    Set newSheet = newBook.Sheets(1)
    newSheet.Name = newSheetNm

    '貼り付け
    sourceRng.Copy
    newSheet.Range(GetColNm(colIdx) & 1).PasteSpecial xlPasteAll
    Application.CutCopyMode = False

    '余白 [行/列] 削除
    newSheet.Rows(2).Delete
    newSheet.Rows(5).Delete
    If 2 <= createMonth Then newSheet.Columns("A:" & GetColNm(colIdx - 1)).Delete

    'Sheet設定
    newSheet.Activate
    With newSheet
        '全削除
        .Cells.FormatConditions.Delete      '条件付き書式
        .Cells.Interior.ColorIndex = xlNone '背景色
        .Cells.Borders.LineStyle = xlNone   '罫線
        '背景色
        .Range("J2:K4").Interior.Color = RGB(217, 217, 217) '#D9D9D9
        .Range("J6:N38").Interior.Color = RGB(217, 217, 217)
        For Each rng In .Range("C8:C38")
            If rng.Value2 = "土" Or rng.Value2 = "日" Or rng.Offset(0, 1).Value2 = "祝" Then
                rng.Interior.Color = RGB(217, 217, 217)
            End If
        Next
        '文字色
        .Cells.Font.Color = RGB(0, 0, 0)
        '列幅
        .Columns("A").ColumnWidth = 3.5
        .Columns("B:D").ColumnWidth = 3
        .Columns("E:N").ColumnWidth = 6
        .Columns("O").ColumnWidth = 20
        '罫線(外)
        For Each borderType In Array(xlEdgeTop, xlEdgeRight, xlEdgeBottom, xlEdgeLeft)
            .Range("B6:O38").Borders(borderType).Weight = xlThin
        Next
        '罫線(内)
        .Range("B2:E4").Borders(xlInsideHorizontal).Weight = xlHairline
        .Range("G3:H4").Borders(xlInsideHorizontal).Weight = xlHairline
        .Range("J3:K4").Borders(xlInsideHorizontal).Weight = xlHairline
        .Range("B7:O38").Borders(xlInsideHorizontal).Weight = xlHairline
    End With
    ActiveWindow.DisplayGridlines = False '目盛線
    newSheet.Rows("2:5").Group            'グループ化
    newSheet.Range("A8").Select           'ウィンドウ枠の固定
    ActiveWindow.FreezePanes = True

    '保存 & 閉じる
    newBook.SaveAs fileName:=ThisWorkbook.path & "\" & newBookNm, FileFormat:=51
    newBook.Close SaveChanges:=False

    '処理中描画ON設定
    Call SettingRenderingProcess(True)
End Sub
