'変数未宣言エラー設定ON
Option Explicit
'------------------------------------------------------------
'概要 : Sheet[勤務表/祝日]を作成する
'引数 : -
'戻値 : -
'備考 : -
'------------------------------------------------------------
Public Sub CreateSheet_WorkSchedule()
    '処理中描画OFF設定
    RenderingProcess False

    '●Loopカウンタ [i/j/k]
    Dim i As Long, j As Long, k As Long
    '●開始行/開始列/終了行/終了列
    Dim rowStartIdx As Long
    Dim colStartIdx As Long
    Dim rowEndIdx As Long
    Dim colEndIdx As Long
    '●範囲
    Dim rng As Range

    '●新規Sheet/Sheet名
    Dim newSheet As Worksheet
    Dim newSheetNm As String

    '●入力受付Form表示内容
    Dim title As String
    Dim msg As String
    '●入力Sheetタイプ/入力可能範囲(1～2)
    Dim inputSheetType As Long
    Dim initSheetType As Double: initSheetType = 1
    Dim minSheetType As Double: minSheetType = 1
    Dim maxSheetType As Double: maxSheetType = 2
    '●入力年/初期値/入力可能年範囲(1年前～10年後)
    Dim inputYear As Long
    Dim initYear As Double: initYear = Year(Now)
    Dim minYear As Double: minYear = Year(Now) - 1
    Dim maxYear As Double: maxYear = Year(Now) + 10
    '●入力勤務時間/初期値/入力休憩時間/入力可能勤務時間範囲(0h～24h)
    Dim inputStartTime1 As String
    Dim inputEndTime1 As String
    Dim inputStartTime2 As String
    Dim inputEndTime2 As String
    Dim initStartTime1 As String: initStartTime1 = "9:00"
    Dim initEndTime1 As String: initEndTime1 = "18:00"
    Dim initStartTime2 As String: initStartTime2 = "12:00"
    Dim initEndTime2 As String: initEndTime2 = "13:00"
    Dim minTime As String: minTime = "0:00"
    Dim maxTime As String: maxTime = "23:59"
    '●入力勤務時間数/初期値/入力可能勤務時間数範囲(0h～24h)
    Dim inputWorkHr As Double
    Dim initWorkHr As Double: initWorkHr = 8
    Dim minHr As Double: minHr = 0
    Dim maxHr As Double: maxHr = 24

    '●祝日辞書(検索用)
    Dim holidayDict As Object: Set holidayDict = CreateObject("Scripting.Dictionary")

    '▼入力受付[Sheetタイプ]
    title = "作成する勤務表タイプを数字で選択して下さい"
    msg = "1：[開始/終了]時間入力タイプ" & vbCrLf & _
          "2：1日の勤務時間数入力タイプ" & vbCrLf & _
          "入力可能範囲（" & minSheetType & "～" & maxSheetType & "）"
    inputSheetType = Int(InputForm_Num(title, msg, initSheetType, minSheetType, maxSheetType))
    'キャンセルの場合
    If inputSheetType = -1 Then Exit Sub
    '▼入力受付[作成年]
    title = "作成する年を入力して下さい"
    msg = "入力可能範囲（" & minYear & "～" & maxYear & "）"
    inputYear = Int(InputForm_Num(title, msg, initYear, minYear, maxYear))
    If inputYear = -1 Then Exit Sub
    Select Case inputSheetType
        '===== 時間タイプの場合 =====
        Case "1"
            '▼入力受付[勤務時間(開始/終了)]
            title = "勤務 [開始] 時間を入力して下さい"
            msg = "入力可能範囲（" & minTime & "～" & maxTime & "）"
            inputStartTime1 = InputForm_Time(title, msg, initStartTime1, minTime, maxTime)
            If inputStartTime1 = "-1" Then Exit Sub
            title = "勤務 [終了] 時間を入力して下さい"
            inputEndTime1 = InputForm_Time(title, msg, initEndTime1, minTime, maxTime)
            If inputEndTime1 = "-1" Then Exit Sub
            '▼入力受付[休憩時間(開始/終了)]
            title = "休憩 [開始] 時間を入力して下さい"
            inputStartTime2 = InputForm_Time(title, msg, initStartTime2, minTime, maxTime)
            If inputStartTime2 = "-1" Then Exit Sub
            title = "休憩 [終了] 時間を入力して下さい"
            inputEndTime2 = InputForm_Time(title, msg, initEndTime2, minTime, maxTime)
            If inputEndTime2 = "-1" Then Exit Sub
        '===== 時間数タイプの場合 =====
        Case "2"
            '▼入力受付[勤務時間数]
            title = "1日の勤務時間数を入力して下さい"
            msg = "入力可能範囲（" & minHr & "～" & maxHr & "）"
            inputWorkHr = InputForm_Num(title, msg, initWorkHr, minHr, maxHr)
            If inputWorkHr = -1 Then Exit Sub
    End Select

    '========== Sheet[祝日]作成 ==========
    newSheetNm = "祝日"
    '存在する場合、既存セット
    If IsExistsSheet(newSheetNm) Then
        Set newSheet = Sheets(newSheetNm)
    Else
        Set newSheet = CreateHolidaySheet(newSheetNm, minYear, maxYear)
    End If

    '========== 祝日辞書 作成 ==========
    rowEndIdx = newSheet.Cells(newSheet.Rows.count, "B").End(xlUp).row
    For Each rng In newSheet.Range("B2:B" & rowEndIdx)
        If IsDate(rng.Value) Then holidayDict(CLng(CDate(rng.Value))) = True
    Next

    '========== Sheet[勤務表]作成 ==========
    Select Case inputSheetType
        Case "1"
            CreateWorkScheduleSheet01 holidayDict, inputYear, inputStartTime1, inputEndTime1, inputStartTime2, inputEndTime2
        Case "2"
            CreateWorkScheduleSheet02 holidayDict, inputYear, inputWorkHr
    End Select

    '処理中描画ON設定
    RenderingProcess True
End Sub
'------------------------------------------------------------
'概要 : 処理中の画面描画を設定する
'引数 : flg(True:ON / False:OFF)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub RenderingProcess(ByVal flg As Boolean)
    Application.ScreenUpdating = flg
    Application.DisplayAlerts = flg
    Application.EnableEvents = flg
    If flg Then
        Application.Calculation = xlCalculationAutomatic
    Else
        Application.Calculation = xlCalculationManual
    End If
End Sub
'------------------------------------------------------------
'概要 : Sheet名のSheetが存在するか判定する
'引数 : wsNm : Sheet名
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Function IsExistsSheet(ByVal wsNm As String) As Boolean
    Dim ws As Worksheet
    For Each ws In Sheets
        If ws.name = wsNm Then
            IsExistsSheet = True
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : Sheet名を重複しない名前に変換する
'     : (同じ名前のSheetが存在しない場合、そのまま返す)
'     : (同じ名前のSheetが存在する場合、2からの連番を末尾に付与し、重複しないSheet名で返す)
'     : "AAA" → "AAA (2)"
'引数 : wsNm : Sheet名
'戻値 : Sheet名
'備考 : -
'------------------------------------------------------------
Function ChangeToUniqueSheetNm(ByVal wsNm As String) As String
    Dim i As Long
    Dim uniqueNo As String
    '存在しない場合
    If Not IsExistsSheet(wsNm) Then
        ChangeToUniqueSheetNm = wsNm
        Exit Function
    End If
    '存在する場合
    For i = 2 To 100
        '末尾付与
        uniqueNo = " (" & i & ")"
        '存在しない場合
        If Not IsExistsSheet(wsNm & uniqueNo) Then
            ChangeToUniqueSheetNm = wsNm & uniqueNo
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : 列番号から列名(アルファベット)を取得する
'     : (範囲外の数値の場合、空を返す)
'引数 : colIdx : 列番号
'戻値 : 列名(アルファベット)
'備考 : -
'------------------------------------------------------------
Private Function GetColNm(ByVal colIdx As Long) As String
    Dim result As String
    Dim n As Long
    If colIdx < 1 Or Application.Columns.count < colIdx Then
        GetColNm = ""
        Exit Function
    End If
    n = colIdx
    Do While 0 < n
        n = n - 1
        result = Chr$(65 + (n Mod 26)) & result
        n = n \ 26
    Loop
    GetColNm = result
End Function
'------------------------------------------------------------
'概要 : 列名(アルファベット)から列番号を取得する
'     : (範囲外の列名の場合、0を返す)
'引数 : colNm : 列名
'戻値 : 列番号
'備考 : -
'------------------------------------------------------------
Private Function GetColIdx(ByVal colNm As String) As Long
    Dim i As Long
    Dim ch As Long
    Dim result As Long
    Dim baseA As Long
    Dim length As Long
    colNm = Trim$(UCase$(colNm))
    length = Len(colNm)
    baseA = Asc("A") - 1
    result = 0
    For i = 1 To length
        ch = Asc(mid$(colNm, i, 1)) - baseA
        If ch < 1 Or 26 < ch Then
            GetColIdx = 0
            Exit Function
        End If
        result = result * 26 + ch
    Next i
    GetColIdx = result
End Function
'------------------------------------------------------------
'概要 : テーブルスタイル適用
'引数 : ws     : Sheet
'     : rng    : 適用範囲
'     : colIdx : ウィンドウ固定列番号
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub ApplyTableStyle01(ByRef ws As Worksheet, ByRef rng As Range, ByRef colIdx As Long)
    'テーブルスタイル適用
    Dim tbl As ListObject: Set tbl = ws.ListObjects.Add(xlSrcRange, rng, , xlYes)
    tbl.TableStyle = "TableStyleMedium4"
    'ヘッダー
    With ws.Range(ws.Cells(rng.row, rng.Column), ws.Cells(rng.row, rng.Columns.count))
        .Font.Color = RGB(0, 0, 0)           '#000000 文字色
        .Interior.Color = RGB(221, 235, 247) '#DDEBF7 背景色
        .HorizontalAlignment = xlCenter      '文字揃え
    End With
    '罫線
    rng.Borders(xlInsideHorizontal).Weight = xlHairline
    rng.Borders(xlInsideHorizontal).ColorIndex = xlAutomatic
    'ウィンドウ枠の固定
    ws.Activate
    ActiveWindow.FreezePanes = False '既存解除
    ws.Cells(rng.row + 1, rng.Column + colIdx - 1).Select
    ActiveWindow.FreezePanes = True
End Sub
'------------------------------------------------------------
'概要 : 罫線01(行境界線)
'引数 : ws     : Sheet
'     : rng    : 適用範囲
'     : colIdx : 境界判定列番号
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub ApplyLineStyle01(ByRef ws As Worksheet, ByRef rng As Range, ByVal colIdx As Long)
    Dim i As Long
    '●名称(前/現在)
    Dim preNm As String
    Dim crntNm As String
    preNm = Left(CStr(ws.Cells(rng.row + 1, rng.Column + colIdx - 1)), 4)
    For i = rng.row + 1 To rng.row + rng.Rows.count
        crntNm = Left(CStr(ws.Cells(i, rng.Column + colIdx - 1)), 4)
        If preNm <> crntNm Then
            ws.Range(ws.Cells(i, rng.Column), ws.Cells(i, rng.Columns.count)).Borders(xlEdgeTop).LineStyle = xlDouble
            ws.Range(ws.Cells(i, rng.Column), ws.Cells(i, rng.Columns.count)).Borders(xlEdgeTop).Weight = xlThick
            preNm = crntNm
        End If
    Next
End Sub
'------------------------------------------------------------
'概要 : 入力Form表示(数値)
'引数 : title   : タイトル文
'     : msg     : 説明文
'     : initVal : 初期値
'     : minVal  : 有効範囲(最小)
'     : maxVal  : 有効範囲(最大)
'戻値 : 入力値
'備考 : キャンセル選択 or 範囲外エラーの場合 -1を返す
'------------------------------------------------------------
Public Function InputForm_Num(ByVal title As String, ByVal msg As String, ByVal initVal As String, ByVal minVal As Double, ByVal maxVal As Double) As Double
    '●入力値
    Dim inputVal As Variant
    Dim inputValDbl As Double
    '入力受付
    inputVal = Application.InputBox(Prompt:=msg, title:=title, Default:=initVal, Type:=1)
    'キャンセルの場合
    If VarType(inputVal) = vbBoolean And inputVal = False Then
        InputForm_Num = -1
        Exit Function
    End If
    '小数点以下有りで受付
    inputValDbl = CDbl(inputVal)
    '範囲チェック
    If inputValDbl < minVal Or maxVal < inputValDbl Then
        MsgBox "範囲内で入力して下さい（" & minVal & " ～ " & maxVal & "）", vbExclamation
        InputForm_Num = -1
        Exit Function
    End If
    InputForm_Num = inputValDbl
End Function
'------------------------------------------------------------
'概要 : 入力Form表示(時間)
'引数 : title   : タイトル文
'     : msg     : 説明文
'     : initVal : 初期値
'     : minVal  : 有効範囲(最小)
'     : maxVal  : 有効範囲(最大)
'戻値 : 入力値
'備考 : キャンセル選択 or 範囲外エラーの場合 -1を返す
'------------------------------------------------------------
Public Function InputForm_Time(ByVal title As String, ByVal msg As String, ByVal initVal As String, ByVal minVal As String, ByVal maxVal As String) As String
    '●入力値
    Dim inputVal As Variant
    '入力受付
    inputVal = Application.InputBox(Prompt:=msg, title:=title, Default:=initVal, Type:=2)
    'キャンセルの場合
    If VarType(inputVal) = vbBoolean And inputVal = False Then
        InputForm_Time = "-1"
        Exit Function
    End If
    inputVal = CStr(inputVal)
    'フォーマットチェック
    If InStr(inputVal, ":") = 0 Or Len(inputVal) < 4 Or 5 < Len(inputVal) Then
        MsgBox "24時間形式で入力して下さい（例：9:00, 13:00）", vbExclamation
        InputForm_Time = "-1"
        Exit Function
    End If
    '範囲チェック
    If CLng(Replace(inputVal, ":", "")) < CLng(Replace(minVal, ":", "")) Or CLng(Replace(maxVal, ":", "")) < CLng(Replace(inputVal, ":", "")) Then
        MsgBox "範囲内で入力して下さい（" & minVal & " ～ " & maxVal & "）", vbExclamation
        InputForm_Time = "-1"
        Exit Function
    End If
    InputForm_Time = inputVal
End Function
'------------------------------------------------------------
'概要 : Sheet[祝日]を作成する
'引数 : wsNm    : 作成Sheet名
'     : minYear : 祝日作成年(開始)
'     : maxYear : 祝日作成年(終了)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Function CreateHolidaySheet(ByVal wsNm As String, ByVal minYear As Long, ByVal maxYear As Long) As Worksheet
    Dim i As Long
    Dim ws As Worksheet    'Sheet
    Dim rowEndIdx As Long  '終了行番号
    Dim colEndIdx As Long  '終了列番号
    Dim rng As Range       '範囲
    '●祝日Map(日付/祝日名)/行
    Dim holidayMap As New Collection
    Dim holidayArr As Variant
    '●日数
    Dim dateCnt As Integer: dateCnt = 1
    '●前年/今年(罫線用)
    Dim preYear As Long
    Dim crntYear As Long

    'Sheet作成 & Sheet名セット
    Set ws = ThisWorkbook.Sheets.Add(Before:=ThisWorkbook.Worksheets(1))
    ws.name = wsNm
    '祝日Map 作成
    Set holidayMap = CreateHolidayMap(minYear, maxYear)
    '祝日 書込
    preYear = minYear '現在年
    For i = 1 To holidayMap.count
        holidayArr = holidayMap(i)
        '月日/祝日名 書込
        ws.Cells(i + 1, "B").Value = holidayArr(0)
        ws.Cells(i + 1, "C").Value = holidayArr(1)
        crntYear = Year(ws.Cells(i + 1, "B").Value2)
        '日数 書込
        If preYear = crntYear Then
            ws.Cells(i + 1, "D").Value = CStr(dateCnt)
            dateCnt = dateCnt + 1
        Else
            ws.Cells(i + 1, "D").Value = 1
            preYear = Year(ws.Cells(i + 1, "B").Value2)
            dateCnt = 2
        End If
    Next
    'A列 書込
    ws.Range("A2:A" & holidayMap.count + 1).FormulaR1C1 = "=RC[1]"
    'その他 書込
    With ws
        .Cells(1, "A").Value = "年"
        .Cells(1, "B").Value = "月日"
        .Cells(1, "C").Value = "祝日名"
        .Cells(1, "D").Value = "日数"
    End With
    '表示形式
    ws.Range("A2:A" & holidayMap.count + 1).NumberFormatLocal = "yyyy"
    ws.Range("B2:B" & holidayMap.count + 1).NumberFormatLocal = "m/d"

    '終了行セット
    rowEndIdx = ws.Cells(ws.Rows.count, "A").End(xlUp).row
    '表範囲セット
    Set rng = Range(ws.Cells(1, "A"), ws.Cells(rowEndIdx, "D"))

    'Sheet設定
    ws.Activate
    ActiveWindow.DisplayGridlines = False '目盛線
    ws.Cells(2, "A").Select               'ウィンドウ枠の固定
    ActiveWindow.FreezePanes = True
    ApplyTableStyle01 ws, rng, 1          'テーブル
    ApplyLineStyle01 ws, rng, 1           '罫線
    With ws
        .Tab.Color = RGB(&HF0, &H90, &H90)                     '#F09090 Tab色
        .Columns("A:B").ColumnWidth = 8                        '列幅
        .Columns("C").ColumnWidth = 12
        .Columns("D").ColumnWidth = 8
        .Cells.VerticalAlignment = xlTop                       '文字揃え
        .Cells.HorizontalAlignment = xlLeft
        .Columns("A:B").HorizontalAlignment = xlCenter
        .Cells(1, "C").HorizontalAlignment = xlCenter
        .Columns("D").HorizontalAlignment = xlCenter
    End With
    Set CreateHolidaySheet = ws
End Function
'------------------------------------------------------------
'概要 : 祝日Mapを作成する
'引数 : minYear    : 祝日作成年(開始)
'     : maxYear    : 祝日作成年(終了)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Function CreateHolidayMap(ByVal minYear As Long, ByVal maxYear As Long) As Collection
    Dim i As Long, j As Long, k As Long
    '●祝日Map/行
    Dim holidayMap As New Collection
    Dim holidayArr As Variant
    '●祝日作成日付(開始/終了)
    Dim startDate As Date: startDate = DateSerial(minYear, 1, 1)
    Dim endDate As Date: endDate = DateSerial(maxYear, 12, 31)
    '●年/日
    Dim y As Long
    Dim d As Date
    '========== 法定祝日 追加 ==========
    For y = Year(startDate) To Year(endDate)
        AddHolidayMap holidayMap, DateSerial(y, 1, 1), "元日"
        AddHolidayMap holidayMap, GetDateOfWeekForMonth(y, 1, 2, vbMonday), "成人の日"
        AddHolidayMap holidayMap, DateSerial(y, 2, 11), "建国記念の日"
        AddHolidayMap holidayMap, DateSerial(y, 2, 23), "天皇誕生日"
        AddHolidayMap holidayMap, GetDateOfVernalEquinox(y), "春分の日"
        AddHolidayMap holidayMap, DateSerial(y, 4, 29), "昭和の日"
        AddHolidayMap holidayMap, DateSerial(y, 5, 3), "憲法記念日"
        AddHolidayMap holidayMap, DateSerial(y, 5, 4), "みどりの日"
        AddHolidayMap holidayMap, DateSerial(y, 5, 5), "こどもの日"
        AddHolidayMap holidayMap, GetDateOfWeekForMonth(y, 7, 3, vbMonday), "海の日"
        AddHolidayMap holidayMap, DateSerial(y, 8, 11), "山の日"
        AddHolidayMap holidayMap, GetDateOfWeekForMonth(y, 9, 3, vbMonday), "敬老の日"
        AddHolidayMap holidayMap, GetDateOfAutumnalEquinox(y), "秋分の日"
        AddHolidayMap holidayMap, GetDateOfWeekForMonth(y, 10, 2, vbMonday), "スポーツの日"
        AddHolidayMap holidayMap, DateSerial(y, 11, 3), "文化の日"
        AddHolidayMap holidayMap, DateSerial(y, 11, 23), "勤労感謝の日"
    Next
    '========== 振替休日 追加 ==========
    '(祝日が日曜日の場合、翌平日を休日とする)
    For i = 1 To holidayMap.count
        holidayArr = holidayMap(i)
        d = holidayArr(0) '法定祝日
        '法定祝日が日曜日の場合、翌日セット
        If weekday(d, vbSunday) = vbSunday Then
            d = d + 1
            '翌日も祝日の場合、翌平日セット
            Do While IsExistsHoliday(holidayMap, d)
                d = d + 1
            Loop
            AddHolidayMap holidayMap, d, "振替休日"
        End If
    Next
    '========== 国民の祝日 追加 ==========
    '(祝日に挟まれた平日は休日とする)
    Dim holidayArr1 As Variant, holidayArr2 As Variant
    '昇順ソート
    SortHolidayMap holidayMap
    For i = 1 To holidayMap.count - 1
        holidayArr1 = holidayMap(i)
        holidayArr2 = holidayMap(i + 1)
        '1日だけ間が空いている場合
        If holidayArr2(0) - holidayArr1(0) = 2 Then
            AddHolidayMap holidayMap, holidayArr1(0) + 1, "国民の祝日"
        End If
    Next
    '昇順ソート(再)
    SortHolidayMap holidayMap
    Set CreateHolidayMap = holidayMap
End Function
'------------------------------------------------------------
'概要 : 祝日Mapに行を追加する
'引数 : holidayMap : 祝日Map
'     : dt         : 祝日
'     : nm         : 祝日名
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub AddHolidayMap(ByRef holidayMap As Collection, ByVal dt As Date, ByVal nm As String)
    '重複しない日付の場合
    If Not IsExistsHoliday(holidayMap, dt) Then holidayMap.Add Array(dt, nm)
End Sub
'------------------------------------------------------------
'概要 : 祝日Mapに存在する日付か判定する
'引数 : holidayMap : 祝日Map
'     : checkDate  : チェック日付
'戻値 : 判定結果(True=する / False=しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsHoliday(ByRef holidayMap As Collection, ByVal checkDate As Date) As Boolean
    Dim i As Long
    For i = 1 To holidayMap.count
        If holidayMap(i)(0) = checkDate Then
            IsExistsHoliday = True
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : y年のm月の第n w曜日の日付を取得する
'引数 : y : 年
'     : m : 月
'     : n : 第n
'     : w : 曜日
'戻値 : 日付
'備考 : -
'------------------------------------------------------------
Private Function GetDateOfWeekForMonth(ByVal y As Long, ByVal m As Long, ByVal n As Long, ByVal w As VbDayOfWeek) As Date
    Dim d As Date: d = DateSerial(y, m, 1)
    Do While weekday(d, vbSunday) <> w
        d = d + 1
    Loop
    GetDateOfWeekForMonth = d + (n - 1) * 7
End Function
'------------------------------------------------------------
'概要 : 春分の日を取得する(簡易近似式)
'引数 : y : 年
'戻値 : 日付(推測日)
'備考 : 春分の日および秋分の日は、国立天文台の計算結果に基づき､官報で告示された日とする
'------------------------------------------------------------
Private Function GetDateOfVernalEquinox(ByVal y As Long) As Date
    Dim d As Integer
    Select Case y
        Case 1900 To 1979
            d = Int(20.8357 + 0.242194 * (y - 1980) - Int((y - 1983) / 4))
        Case 1980 To 2099
            d = Int(20.8431 + 0.242194 * (y - 1980) - Int((y - 1980) / 4))
        Case Else
            d = 20
    End Select
    GetDateOfVernalEquinox = DateSerial(y, 3, d)
End Function
'------------------------------------------------------------
'概要 : 秋分の日を取得する(簡易近似式)
'引数 : y : 年
'戻値 : 日付(推測日)
'備考 : 春分の日および秋分の日は、国立天文台の計算結果に基づき､官報で告示された日とする
'------------------------------------------------------------
Private Function GetDateOfAutumnalEquinox(ByVal y As Long) As Date
    Dim d As Integer
    Select Case y
        Case 1900 To 1979
            d = Int(23.2588 + 0.242194 * (y - 1980) - Int((y - 1983) / 4))
        Case 1980 To 2099
            d = Int(23.2488 + 0.242194 * (y - 1980) - Int((y - 1980) / 4))
        Case Else
            d = 23
    End Select
    GetDateOfAutumnalEquinox = DateSerial(y, 9, d)
End Function
'------------------------------------------------------------
'概要 : 昇順ソート(祝日Map用)
'引数 : holidayMap : 祝日Map
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub SortHolidayMap(ByRef holidayMap As Collection)
    Dim i As Long, j As Long
    Dim tmp As Variant
    '●祝日Map行
    Dim holidayArr As Variant
    'Collectionから配列に変換
    ReDim holidayArr(1 To holidayMap.count)
    For i = 1 To holidayMap.count
        holidayArr(i) = holidayMap(i)
    Next
    'バブルソート(日付で昇順)
    For i = 1 To UBound(holidayArr) - 1
        For j = i + 1 To UBound(holidayArr)
            '次行の方が小さい場合、入替
            If holidayArr(j)(0) < holidayArr(i)(0) Then
                tmp = holidayArr(i)
                holidayArr(i) = holidayArr(j)
                holidayArr(j) = tmp
            End If
        Next
    Next
    '初期化
    Set holidayMap = New Collection
    '配列からCollectionに変換
    For i = 1 To UBound(holidayArr)
        holidayMap.Add holidayArr(i)
    Next
End Sub

'     : inputWorkHr : 入力勤務時間
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub CreateWorkScheduleSheet02(ByVal holidayDict As Object, ByVal inputYear As Long, ByVal inputWorkHr As Double)
    Dim ws As Worksheet      'Sheet
    Dim wsNm As String       'Sheet名
    Dim i As Long, j As Long
    Dim rowStartIdx As Long  '開始行番号
    Dim rng As Range         '範囲
    Dim tmpDate1 As Date
    Dim tmpDate2 As Date
    Dim borderType As Variant
    '●一括セット用
    Dim balkSetWeek(1 To 1, 1 To 31) As Variant '曜日
    Dim balkSetDate(1 To 1, 1 To 31) As Variant '日付
    Dim balkSetPlan(1 To 1, 1 To 31) As Variant '予
    Dim balkSetAct(1 To 1, 1 To 31) As Variant  '実
    Dim balkSetType(1 To 1, 1 To 31) As Variant '出リ

    'Sheet名 作成
    wsNm = ChangeToUniqueSheetNm(CStr(inputYear))
    'Sheet作成 & Sheet名セット
    Set ws = ThisWorkbook.Sheets.Add(Before:=ThisWorkbook.Worksheets(1))
    ws.name = wsNm
    '12か月分Loop(書込)
    For i = 1 To 12
        rowStartIdx = i * 12
        With ws
            .Cells(rowStartIdx + 3, "A").Value = "予"
            .Cells(rowStartIdx + 4, "A").Value = "実"
            .Cells(rowStartIdx + 5, "A").Value = "出"
            .Cells(rowStartIdx + 6, "A").Value = "備"
            .Cells(rowStartIdx, "C").Value = DateSerial(inputYear, i, 1)
            .Cells(rowStartIdx + 1, "AI").Value = "予定(日)"
            .Cells(rowStartIdx + 2, "AI").Value = "予定(時)"
            .Cells(rowStartIdx + 3, "AI").Value = "実績(日)"
            .Cells(rowStartIdx + 4, "AI").Value = "実績(時)"
            .Cells(rowStartIdx + 5, "AI").Value = "出社"
            .Cells(rowStartIdx + 6, "AI").Value = "リモート"
            .Cells(rowStartIdx + 7, "AI").Value = "出社率"
            .Cells(rowStartIdx + 1, "AJ").Formula = "=COUNT(C" & rowStartIdx + 3 & ":AG" & rowStartIdx + 3 & ")"
            .Cells(rowStartIdx + 2, "AJ").Formula = "=LET(v,SUM(C" & rowStartIdx + 3 & ":AG" & rowStartIdx + 3 & "),IF(v=INT(v),TEXT(v,""0"")&"" h"",TEXT(v,""0.###"")&"" h""))"
            .Cells(rowStartIdx + 3, "AJ").Formula = "=COUNT(C" & rowStartIdx + 4 & ":AG" & rowStartIdx + 4 & ")"
            .Cells(rowStartIdx + 4, "AJ").Formula = "=LET(v,SUM(C" & rowStartIdx + 4 & ":AG" & rowStartIdx + 4 & "),IF(v=INT(v),TEXT(v,""0"")&"" h"",TEXT(v,""0.###"")&"" h""))"
            .Cells(rowStartIdx + 5, "AJ").Formula = "=COUNTIF(C" & rowStartIdx + 5 & ":AG" & rowStartIdx + 5 & ",""出"")"
            .Cells(rowStartIdx + 6, "AJ").Formula = "=COUNTIF(C" & rowStartIdx + 5 & ":AG" & rowStartIdx + 5 & ",""リ"")"
            .Cells(rowStartIdx + 7, "AJ").Formula = "=AJ" & rowStartIdx + 5 & "/(AJ" & rowStartIdx + 5 & "+AJ" & rowStartIdx + 6 & ")"
            .Cells(rowStartIdx + 2, "C").Formula = "=C" & rowStartIdx
            '31日分Loop
            For j = 1 To 31
                '初期化
                balkSetWeek(1, j) = ""
                balkSetDate(1, j) = ""
                balkSetPlan(1, j) = ""
                balkSetAct(1, j) = ""
                balkSetType(1, j) = ""
                balkSetWeek(1, j) = "=TEXT(" & GetColNm(j + 2) & rowStartIdx + 2 & ",""aaa"")"
                balkSetDate(1, j) = "=IF(" & GetColNm(j + 2) & rowStartIdx + 2 & "="""","""",IF(" & GetColNm(j + 2) & rowStartIdx + 2 & "+1<=EOMONTH($C$" & rowStartIdx & ",0)," & GetColNm(j + 2) & rowStartIdx + 2 & "+1,""""))"
                '平日の場合、[勤務時間/出リ] 書込
                tmpDate1 = DateSerial(inputYear, i, j)
                If Month(tmpDate1) = i And weekday(tmpDate1, vbMonday) <= 5 Then
                    '12/29-1/3でない場合
                    tmpDate2 = Month(tmpDate1) & "-" & Day(tmpDate1)
                    If tmpDate2 <> "12-29" And tmpDate2 <> "12-30" And tmpDate2 <> "12-31" And tmpDate2 <> "1-1" And tmpDate2 <> "1-2" And tmpDate2 <> "1-3" Then
                        '祝日でない場合
                        If Not holidayDict.exists(tmpDate1) Then
                            balkSetPlan(1, j) = inputWorkHr
                            balkSetAct(1, j) = inputWorkHr
                            balkSetType(1, j) = "リ"
                        End If
                    End If
                End If
            Next
            .Cells(rowStartIdx + 1, "C").Resize(1, 31).Formula = balkSetWeek
            .Cells(rowStartIdx + 2, "D").Resize(1, 30).Formula = balkSetDate
            .Cells(rowStartIdx + 3, "C").Resize(1, 31).Value = balkSetPlan
            .Cells(rowStartIdx + 4, "C").Resize(1, 31).Value = balkSetAct
            .Cells(rowStartIdx + 5, "C").Resize(1, 31).Value = balkSetType
        End With
    Next

    'Sheet設定
    ws.Activate
    ActiveWindow.DisplayGridlines = False '目盛線
    ws.Cells.VerticalAlignment = xlTop    '文字揃え
    ws.Cells.HorizontalAlignment = xlLeft
    '12か月分Loop
    For i = 1 To 12
        rowStartIdx = i * 12
        With ws
            .Cells(rowStartIdx, "C").Font.Bold = True '太字
            .Cells(rowStartIdx + 3, "A").Interior.Color = RGB(217, 217, 217)                             '#D9D9D9 背景色
            .Cells(rowStartIdx, "C").Interior.Color = RGB(218, 242, 208)                                 '#DAF2D0
            .Range("C" & rowStartIdx + 3 & ":AG" & rowStartIdx + 3).Interior.Color = RGB(217, 217, 217)  '#D9D9D9
            .Range("AI" & rowStartIdx + 1 & ":AJ" & rowStartIdx + 2).Interior.Color = RGB(217, 217, 217) '#D9D9D9
            .Columns("A").ColumnWidth = 3   '列幅
            .Columns("B").ColumnWidth = 0.5
            If Len(mid(inputWorkHr, InStr(inputWorkHr, ".") + 1)) < 2 Then
                .Columns("C:AG").ColumnWidth = 3
            Else
                .Columns("C:AG").ColumnWidth = 4
            End If
            .Columns("AH").ColumnWidth = 0.5
            .Columns("AI:AJ").ColumnWidth = 8
            .Range("C" & rowStartIdx & ":AG" & rowStartIdx).Merge Across:=True 'セル結合
            .Range("C" & rowStartIdx + 6 & ":Q" & rowStartIdx + 10).Merge
            .Range("S" & rowStartIdx + 6 & ":AG" & rowStartIdx + 10).Merge
            .Range("A" & rowStartIdx + 3 & ":A" & rowStartIdx + 6).HorizontalAlignment = xlCenter
            .Range("C" & rowStartIdx & ":AG" & rowStartIdx + 5).HorizontalAlignment = xlCenter
            .Range("AI" & rowStartIdx + 1 & ":AI" & rowStartIdx + 7).HorizontalAlignment = xlCenter
            .Range("AJ" & rowStartIdx + 1 & ":AJ" & rowStartIdx + 7).HorizontalAlignment = xlRight
            .Range("C" & rowStartIdx + 6 & ":Q" & rowStartIdx + 10).VerticalAlignment = xlTop
            .Range("S" & rowStartIdx + 6 & ":AG" & rowStartIdx + 10).VerticalAlignment = xlTop
            If i = 1 Then .Range("C" & rowStartIdx).NumberFormatLocal = "m"" 月 (""ggge""年)""" '表示形式
            If i <> 1 Then .Range("C" & rowStartIdx).NumberFormatLocal = "m"" 月"""
            .Range("C" & rowStartIdx + 2 & ":AG" & rowStartIdx + 2).NumberFormatLocal = "d"
            .Range("AJ" & rowStartIdx + 1).NumberFormatLocal = "0"" 日"""
            .Range("AJ" & rowStartIdx + 3).NumberFormatLocal = "0"" 日"""
            .Range("AJ" & rowStartIdx + 5).NumberFormatLocal = "0"" 日"""
            .Range("AJ" & rowStartIdx + 6).NumberFormatLocal = "0"" 日"""
            .Range("AJ" & rowStartIdx + 7).NumberFormatLocal = "0 %"
            For j = 1 To 2 '備考枠線
                If j = 1 Then Set rng = .Range("C" & rowStartIdx + 6 & ":Q" & rowStartIdx + 10)
                If j = 2 Then Set rng = .Range("S" & rowStartIdx + 6 & ":AG" & rowStartIdx + 10)
                For Each borderType In Array(xlEdgeTop, xlEdgeBottom, xlEdgeLeft, xlEdgeRight, xlInsideHorizontal, xlInsideVertical)
                    rng.Borders(borderType).Weight = xlThin
                    rng.Borders(borderType).Color = RGB(191, 191, 191) '#BFBFBF
                Next
            Next
            .Range("A" & rowStartIdx + 1 & ":A" & rowStartIdx + 11).EntireRow.Group 'グループ化
            Set rng = Range(.Cells(rowStartIdx + 1, "C"), .Cells(rowStartIdx + 5, "AG"))                                                                  '条件付き書式(背景色)
            rng.FormatConditions.Add(Type:=xlExpression, Formula1:="=C$" & rowStartIdx + 2 & "=""""").Interior.Color = RGB(217, 217, 217)                 '#D9D9D9
            Set rng = Range(.Cells(rowStartIdx + 1, "C"), .Cells(rowStartIdx + 1, "AG"))
            rng.FormatConditions.Add(Type:=xlExpression, Formula1:="=COUNTIF(祝日!$B:$B,C" & rowStartIdx + 2 & ")>0").Interior.Color = RGB(240, 144, 144) '#F09090
            Set rng = Range(.Cells(rowStartIdx + 1, "C"), .Cells(rowStartIdx + 1, "AG"))
            rng.FormatConditions.Add(Type:=xlExpression, Formula1:="=WEEKDAY(C" & rowStartIdx + 2 & ",2)=7").Interior.Color = RGB(251, 226, 213)          '#FBE2D5
            Set rng = Range(.Cells(rowStartIdx + 1, "C"), .Cells(rowStartIdx + 1, "AG"))
            rng.FormatConditions.Add(Type:=xlExpression, Formula1:="=WEEKDAY(C" & rowStartIdx + 2 & ",2)=6").Interior.Color = RGB(218, 233, 248)          '#DAE9F8
        End With
    Next

    '行 削除
    ws.Rows("1:11").Delete
    '削除状態更新
    ActiveSheet.UsedRange
End Sub
