'------------------------------------------------------------
'概要 : Sheet[入力]を作成する
'引数 : -
'戻値 : -
'備考 : -

Public Sub CreateSheet_InputCsvContents()

    Call SettingRenderingProcess(False)

    Dim i As Long       'Loopカウンタ
    Dim ws As Worksheet 'Sheet
    Dim rng As Range    '範囲
    '●図形/サイズ(横/縦)
    Dim shp As Shape
    Dim shpWidth As Double: shpWidth = 4
    Dim shpHeight As Double: shpHeight = 1.5

    '新規Sheet作成 & 新規Sheet名セット
    Set ws = ThisWorkbook.Sheets.Add(Before:=ThisWorkbook.Worksheets(1))
    ws.Name = ChangeUniqueSheetNm("INPUT")

    '書込
    With ws
        .Range("A1").Value = "FILE名"
        .Range("A2").Value = "CSV/TSV"
        .Range("A3").Value = "HEADER"
        .Range("A4").Value = "HEAD論/理"
        .Range("A9").Value = "No"
        .Range("A10:A109").Formula2 = "=ROW()-9"
        .Range("B1").Value = "ファイル名"
        '=== ドロップダウン ==========
        .Range("B2").Validation.Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, Operator:=xlBetween, Formula1:="CSV,TSV"
        .Range("B2").Value = "CSV"
        .Range("B3").Validation.Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, Operator:=xlBetween, Formula1:="有,無"
        .Range("B3").Value = "有"
        .Range("B4").Validation.Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, Operator:=xlBetween, Formula1:="論理,物理"
        .Range("B4").Value = "論理"
        .Range("D4:BA4").Validation.Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, Operator:=xlBetween, Formula1:="Y,N"
        .Range("D4:BA4").Value = "Y"
        '=============================
        .Range("C1").Value = "No"
        .Range("C2").Value = "項目名(論理)"
        .Range("C3").Value = "項目名(物理)"
        .Range("C4").Value = "ダブクォ付与"
        .Range("C5").Value = "備考1"
        .Range("C6").Value = "備考2"
        .Range("C7").Value = "備考3"
        .Range("C8").Value = "備考4"
        .Range("C9").Value = "備考5"
        .Range("D1:BA1").Formula2 = "=COLUMN()-3"
        For i = 1 To 50
            .Range("C2").Offset(0, i).Value = "項目名" & i
            .Range("C3").Offset(0, i).Value = "COL_NM_" & i
        Next
    End With

    'Sheet設定
    ws.Activate
    With ws
        'フォント
        .Cells.Font.Name = "HGｺﾞｼｯｸM"
        '背景色
        .Range("A1:A4").Interior.Color = RGB(255, 192, 0)      '#FFC000
        .Range("B1:B4").Interior.Color = RGB(255, 242, 204)    '#FFF2CC
        .Range("A5:B8").Interior.Color = RGB(128, 128, 128)    '#808080
        .Range("A9:C9").Interior.Color = RGB(155, 194, 230)    '#9BC2E6
        .Range("C1:C8").Interior.Color = RGB(155, 194, 230)    '#9BC2E6
        .Range("A10:A109").Interior.Color = RGB(217, 217, 217) '#D9D9D9
        .Range("B10:C109").Interior.Color = RGB(191, 191, 191) '#BFBFBF
        .Range("D1:BA1").Interior.Color = RGB(155, 194, 230)   '#9BC2E6
        .Range("D2:BA4").Interior.Color = RGB(189, 215, 238)   '#BDD7EE
        .Range("D5:BA9").Interior.Color = RGB(221, 235, 247)   '#DDEBF7
        '文字揃え
        .Cells.VerticalAlignment = xlTop
        .Cells.HorizontalAlignment = xlLeft
        .Columns("A").HorizontalAlignment = xlCenter
        .Range("B1:C9").HorizontalAlignment = xlCenter
        .Range("B1").HorizontalAlignment = xlLeft
        .Range("D1:BA4").HorizontalAlignment = xlCenter
        '列幅
        .Columns("A").ColumnWidth = 10
        .Columns("B").ColumnWidth = 20
        .Columns("C").ColumnWidth = 13
        .Columns("D:BA").ColumnWidth = 15
        'セル結合
        .Range("B10:C109").Merge Across:=True
        'グループ化
        .Columns("B:C").Group
        .Rows("5:9").Group
        '折り返して表示
        .Range("D5:BA9").WrapText = True
        .Range("B10:C109").WrapText = True
    End With
    '表示倍率
    ActiveWindow.Zoom = 85
    '目盛線
    ActiveWindow.DisplayGridlines = False
    'ウィンドウ枠の固定
    ws.Range("D10").Select
    ActiveWindow.FreezePanes = True

    '条件付き書式
    ws.Range("D10:BA109").FormatConditions.Add(Type:=xlExpression, Formula1:="=ISODD(ROW())").Interior.Color = RGB(242, 242, 242)
    '罫線
    ws.Range("C5:BA9").Borders(xlInsideHorizontal).Weight = xlHairline
    ws.Range("C5:BA9").Borders(xlInsideHorizontal).ColorIndex = xlAutomatic
    ws.Range("A9:BA109").Borders(xlInsideHorizontal).Weight = xlHairline
    ws.Range("A9:BA109").Borders(xlInsideHorizontal).ColorIndex = xlAutomatic

    'マクロボタン作成
    Set shp = ws.Shapes.AddShape(msoShapeBevel, 0, 0, Application.CentimetersToPoints(shpWidth), Application.CentimetersToPoints(shpHeight))
    With shp
        .Fill.ForeColor.RGB = RGB(91, 155, 213)                            '#5B9BD5 塗りつぶし色
        .line.ForeColor.RGB = RGB(91, 155, 213)                            '#5B9BD5 枠線色
        .TextFrame2.TextRange.Text = "作成"                                '表示文字列
        .TextFrame2.TextRange.Font.NameFarEast = "游ゴシック"              'フォント(JP)
        .TextFrame2.TextRange.Font.Size = 16                               '文字サイズ
        .TextFrame2.TextRange.Font.Bold = True                             '太字
        .TextFrame2.TextRange.Font.Fill.ForeColor.RGB = RGB(255, 255, 255) '文字色
        .TextFrame2.VerticalAnchor = msoAnchorMiddle                       '文字揃え(横)
        .TextFrame2.TextRange.ParagraphFormat.Alignment = msoAlignCenter   '文字揃え(縦)
        .OnAction = "Btn_CreateCsvFile"                                    'マクロ登録
        .Top = ws.Range("B5").Top + 5                                      '配置場所(縦)
        .Left = ws.Range("B5").Left + 6                                    '配置場所(横)

'------------------------------------------------------------
'概要 : CSV or TSV ファイル作成
'     : (このExcelファイルと同じ場所に作成する)
'引数 : -
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub Btn_CreateCsvFile()
    '処理中描画OFF設定
    Call SettingRenderingProcess(False)

    Dim i As Long, j As Long                  'Loopカウンタ
    Dim ws As Worksheet: Set ws = ActiveSheet 'Sheet
    Dim rowIdx As Long                        '行
    Dim colIdx As Long                        '列
    Dim endRowIdx As Long                     '終了行
    Dim endColIdx As Long                     '終了列
    Dim rng As Range                          '範囲
    '●CSV/TSVフラグ(True=CSV/False=TSV)
    Dim csvTsvFlg As Boolean
    '●Header有無フラグ(True=有/False=無)
    Dim headerFlg As Boolean
    '●論理/物理フラグ(True=論理/False=物理)
    Dim physiLogiFlg As Boolean
    '●ファイルパス/ファイル名/拡張子
    Dim filePath As String: filePath = ThisWorkbook.path & "\"
    Dim fileNm As String
    Dim fileExt As String
    '●CSV内容/1行/文字列/区切り文字
    Dim csvBody As String
    Dim line As String
    Dim str As Variant
    Dim delimiter As String
    '●コピー元範囲
    Dim sourceRng As Range
    '●ヘッダー項目リスト/ヘッダー項目/ダブルクォート設定値リスト
    Dim headerValCol As New Collection
    Dim headerVal As Variant
    Dim dblQtValList As Variant
    '●メッセージ内容
    Dim msg As String

    '=== ▼入力 & 存在チェック[CSV/TSV / HEADER / HEAD論/物] ==========
    '=== [CSV/TSV]が未選択の場合 ==========
    If LenB(ws.Range("B2").Value2) = 0 Then
        msg = "(B2) [CSV/TSV] を選択して下さい"
        Call MsgBox(msg, vbCritical, "")
        Exit Sub
    End If
    '=== [CSV/TSV]が未選択の場合 ==========
    If LenB(ws.Range("B3").Value2) = 0 Then
        msg = "(B3) [HEADER] を選択して下さい"
        Call MsgBox(msg, vbCritical, "")
        Exit Sub
    End If
    '=== [CSV/TSV]が未選択の場合 ==========
    If LenB(ws.Range("B4").Value2) = 0 Then
        msg = "(B4) [HEAD論/物] を選択して下さい"
        Call MsgBox(msg, vbCritical, "")
        Exit Sub
    End If

    '=== 設定値 セット ====================
    'CSV/TSVフラグ/拡張子/区切り文字
    str = ws.Range("B2").Value2
    If str = "CSV" Then
        csvTsvFlg = True
        fileExt = "csv"
        delimiter = ","
    Else
        csvTsvFlg = False
        fileExt = "tsv"
        delimiter = vbTab
    End If
    'ファイル名 (未入力の場合 Sheet名をセット)
    fileNm = Trim$(RemoveForbiddenCharsFileNm(ws.Range("B1").Value2))
    If LenB(fileNm) = 0 Then
        fileNm = ChangeUniqueFileNm(filePath, ws.Name, fileExt)
    Else
        fileNm = ChangeUniqueFileNm(filePath, fileNm, fileExt)
    End If
    'ヘッダー有無フラグ
    headerFlg = IIf(ws.Range("B3").Value2 = "有", True, False)
    '論理/物理フラグ
    physiLogiFlg = IIf(ws.Range("B4").Value2 = "論理", True, False)
    '======================================

    '終了(行/列) セット
    endRowIdx = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    endColIdx = ws.Cells(2, ws.Columns.Count).End(xlToLeft).Column

    'ダブルクォート有無行 取得
    dblQtValList = ws.Range(ws.Range("D4"), ws.Cells(4, endColIdx)).Value2

    '========== ヘッダー作成 ==========
    If headerFlg Then
        'ヘッダー行 取得
        If physiLogiFlg Then
            Set rng = ws.Range(ws.Range("D2"), ws.Cells(2, endColIdx))
        Else
            Set rng = ws.Range(ws.Range("D3"), ws.Cells(3, endColIdx))
        End If
        For colIdx = 1 To rng.Columns.Count
            headerValCol.Add (CStr(rng.Cells(1, colIdx).Text))
        Next
        '作成
        For colIdx = 1 To rng.Columns.Count
            'ダブルクォート囲み有無
            If CStr(dblQtValList(1, colIdx)) = "Y" Then
                line = line & """" & CStr(headerValCol(colIdx)) & """" & delimiter
            Else
                line = line & CStr(headerValCol(colIdx)) & delimiter
            End If
        Next
        '末尾除去
        line = Left$(line, Len(line) - Len(delimiter))
        csvBody = csvBody & line & vbCrLf
    End If

    'データ範囲 取得
    Set sourceRng = ws.Range(ws.Range("D10"), ws.Cells(endRowIdx, endColIdx))

    '========== データ作成 ==========
    For rowIdx = 1 To sourceRng.Rows.Count
        'ステータスバー
        Application.StatusBar = "ファイル内容 作成中... " & rowIdx & " / " & sourceRng.Rows.Count
        line = ""
        For colIdx = 1 To sourceRng.Columns.Count
            'CSV用文字列変換
            str = ConvertCsvVal(CStr(sourceRng.Cells(rowIdx, colIdx).Text))
            'ダブルクォート囲み有無
            If CStr(dblQtValList(1, colIdx)) = "Y" Then
                line = line & """" & str & """" & delimiter
            Else
                line = line & str & delimiter
            End If
        Next
        '末尾除去
        line = Left$(line, Len(line) - Len(delimiter))
        csvBody = csvBody & line & vbCrLf
    Next

    'ファイル作成(UTF-8(BOMあり))
    Call CreateTxtFile(filePath, fileNm, fileExt, csvBody, "UTF-8", True)

    'ステータスバー初期化
    Application.StatusBar = False

    '処理中描画ON設定
    Call SettingRenderingProcess(True)
End Sub
'------------------------------------------------------------
'概要 : 処理中の画面描画を設定する
'引数 : flg(True:ON / False:OFF)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub SettingRenderingProcess(ByVal flg As Boolean)
    Application.ScreenUpdating = flg '画面再描画
    Application.DisplayAlerts = flg  '確認ダイアログ表示
    Application.EnableEvents = flg   'イベント発火
    '再計算モード(セル値が変わるたびに再計算)
    If flg Then
        Application.Calculation = xlCalculationAutomatic '自動
    Else
        Application.Calculation = xlCalculationManual '手動
    End If
End Sub
'------------------------------------------------------------
'概要 : ファイルが存在するか判定する
'引数 : path   : ファイルまでのパス(絶対パス)
'     : fileNm : ファイル名
'     : ext    : 拡張子
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsFile(ByVal path As String, ByVal fileNm As String, ByVal ext As String) As Boolean
    'パス末尾付与
    If Right$(path, 1) <> "\" Then path = path & "\"
    '判定
    IsExistsFile = (Dir(path & fileNm & "." & ext) <> "")
End Function
'------------------------------------------------------------
'概要 : 重複しないファイル名となるまで、末尾に連番を付与する
'     : (同じ名前のファイルが存在しない場合、そのまま返す)
'     : (同じ名前のファイルが存在する場合、2からの連番を末尾に付与し、重複しないファイル名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : path   : ファイルまでのパス
'     : fileNm : ファイル名
'     : ext    : 拡張子
'戻値 : ファイル名(拡張子付き)
'備考 : -
'------------------------------------------------------------
Private Function ChangeUniqueFileNm(ByVal path As String, ByVal fileNm As String, ByVal ext As String) As String
    Dim i As Long
    Dim uniqueNo As String
    For i = 2 To 1000
        '存在しない場合
        If Not IsExistsFile(path, fileNm & uniqueNo, ext) Then
            ChangeUniqueFileNm = fileNm & uniqueNo & "." & ext
            Exit Function
        End If
        '末尾付与
        uniqueNo = " (" & i & ")"
    Next
End Function
'------------------------------------------------------------
'概要 : Sheet名のSheetが存在するか判定する
'引数 : wsNm : Sheet名
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsSheet(ByVal wsNm As String) As Boolean
    Dim ws As Worksheet
    For Each ws In Sheets
        '存在する場合
        If ws.Name = wsNm Then
            IsExistsSheet = True
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : 重複しないSheet名となるまで、末尾に連番を付与する
'     : (同じ名前のSheetが存在しない場合、そのまま返す)
'     : (同じ名前のSheetが存在する場合、2からの連番を末尾に付与し、重複しないSheet名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : wsNm : Sheet名
'戻値 : Sheet名
'備考 : -
'------------------------------------------------------------
Private Function ChangeUniqueSheetNm(ByVal wsNm As String) As String
    Dim i As Long
    Dim uniqueNo As String
    For i = 2 To 1000
        '存在しない場合
        If Not IsExistsSheet(wsNm & uniqueNo) Then
            ChangeUniqueSheetNm = wsNm & uniqueNo
            Exit Function
        End If
        '末尾付与
        uniqueNo = " (" & i & ")"
    Next
End Function
'------------------------------------------------------------
'概要 : 列番号から列名(アルファベット)を取得する
'     : (範囲外の数値の場合、空を返す)
'引数 : colIdx : 列番号
'戻値 : 列名(アルファベット)
'備考 : -
'------------------------------------------------------------
Private Function GetColNm(ByVal colIdx As Long) As String
    Dim result As String
    '範囲外の場合
    If colIdx < 1 Or Application.Columns.Count < colIdx Then
        GetColNm = ""
        Exit Function
    End If
    Do While 0 < colIdx
        colIdx = colIdx - 1
        result = Chr$(65 + (colIdx Mod 26)) & result
        colIdx = colIdx \ 26
    Loop
    GetColNm = result
End Function
'------------------------------------------------------------
'概要 : 列名(アルファベット)から列番号を取得する
'     : (範囲外の列名の場合、0を返す)
'引数 : colNm : 列名(アルファベット)
'戻値 : 列番号
'備考 : -
'------------------------------------------------------------
Private Function GetColIdx(ByVal colNm As String) As Long
    Dim i As Long
    Dim result As Long
    Dim baseA As Long: baseA = Asc("A") - 1
    Dim colNmLen As Long: colNmLen = Len(UCase$(colNm))
    Dim ch As Long
    For i = 1 To colNmLen
        ch = Asc(Mid$(colNm, i, 1)) - baseA
        '範囲外の場合
        If ch < 1 Or 26 < ch Then
            GetColIdx = 0
            Exit Function
        End If
        result = result * 26 + ch
    Next
    GetColIdx = result
End Function
'------------------------------------------------------------
'概要 : 文字列をCSV出力値に変換する
'     : 1,改行を除去
'     : 2,タブを文字列に変換
'     : 3,ダブルクォーテーションをエスケープ
'引数 : str : 文字列
'戻値 : CSV出力値
'備考 : -
'------------------------------------------------------------
Private Function ConvertCsvVal(ByVal str As String) As String
    '改行
    str = Replace$(str, vbCr, "")
    str = Replace$(str, vbLf, "")
    str = Replace$(str, vbCrLf, "")
    'タブ
    str = Replace$(str, vbTab, "[TAB]")
    'ダブルクォーテーション
    str = Replace$(str, """", """""")
    ConvertCsvVal = str
End Function
 '------------------------------------------------------------
'概要 : ファイル名から使用不可文字を除去する
'引数 : fileNm : ファイル名
'戻値 : ファイル名
'備考 : -
'------------------------------------------------------------
Private Function RemoveForbiddenCharsFileNm(ByVal fileNm As String) As String
    Dim i As Long
    Dim str As Variant
    Dim forbiddenChars As Variant: forbiddenChars = Array("\", "/", ":", "*", "?", """", "<", ">", "|")
    For Each str In forbiddenChars
        fileNm = Replace(fileNm, CStr(str), "")
    Next
    RemoveForbiddenCharsFileNm = fileNm
End Function
'------------------------------------------------------------
'概要 : テキストファイルを作成する(UTF-8)
'引数 : filePath  : ファイルパス
'     : fileNm    : ファイル名
'     : fileExt   : ファイル拡張子
'     : contents  : 書込内容
'     : charsetNm : 文字コード名(UTF-8 / Shift-JIS など)
'     : bomFlg    : BOM有無フラグ(True=有 / False=無)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub CreateTxtFile(ByVal filePath As String, ByVal fileNm As String, ByVal fileExt As String, ByVal contents As String, ByVal charsetNm As String, ByVal bomFlg As Boolean)
    '●ファイル書込用
    Dim stream As Object: Set stream = CreateObject("ADODB.Stream")
    Dim binStream As Object
    Dim byteData() As Byte
    Dim bomLen As Integer: bomLen = GetBomLen(charsetNm)

    '末尾付与
    If Right$(filePath, 1) <> "\" Then filePath = filePath & "\"
    '先頭付与
    If Left$(fileExt, 1) <> "." Then fileExt = "." & fileExt

    With stream
        .Type = 2
        .Charset = charsetNm
        .Open
        .WriteText contents
        .SaveToFile filePath & fileNm, 2 '既存ファイル上書き設定
        .Close
    End With
    Set stream = Nothing

    'BOMなしの場合、先頭nバイト(BOM)除去で作り直す
    If Not bomFlg And 0 < bomLen Then
        '===== 読込 =====
        Set binStream = CreateObject("ADODB.Stream")
        With binStream
            .Type = 1
            .Open
            .LoadFromFile filePath & fileNm
            .Position = bomLen
            byteData = .Read
            .Close
        End With
        '===== 書込 =====
        With binStream
            .Type = 1
            .Open
            .Write byteData
            .SaveToFile filePath & fileNm, 2
            .Close
        End With
        Set binStream = Nothing
    End If
End Sub
'------------------------------------------------------------
'概要 : 文字コード名のBOMバイト数を取得する
'引数 : charsetNm : 文字コード名(UTF-8 / Shift-JIS など)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Function GetBomLen(ByVal charsetNm As String) As Integer
    Select Case UCase$(charsetNm)
        Case "UTF-8"
            GetBomLen = 3
        Case "UTF-16", "UNICODE", "UTF-16LE", "UTF-16BE"
            GetBomLen = 2
        Case "UTF-32"
            GetBomLen = 4
        Case Else
            GetBomLen = 0
    End Select
End Function
