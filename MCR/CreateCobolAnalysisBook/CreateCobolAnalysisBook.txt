'------------------------------------------------------------
'概要 : COBOLの3種ファイル[.jcl/.cob/.cpy]の情報整理マクロ
'     : 情報まとめSheetを複数作成する
'     : 各Sheetの作成可否はフラグで設定
'引数 : -
'戻値 : -
'備考 : ※※※ 未完成(特殊パターン未対応など) ※※※
'------------------------------------------------------------
Public Sub CreateCobolAnalysisBook()
    '処理中描画OFF設定
    RenderingProcess False

    '==================== ユーザー設定 ====================
    '●Sheet作成フラグ(True:する / False:しない)
    Dim createSheet00Flg As Boolean: createSheet00Flg = True
    Dim createSheet01Flg As Boolean: createSheet01Flg = True
    Dim createSheet02Flg As Boolean: createSheet02Flg = True
    Dim createSheet03Flg As Boolean: createSheet03Flg = True
    Dim createSheet04Flg As Boolean: createSheet04Flg = True
    Dim createSheet05Flg As Boolean: createSheet05Flg = True
    Dim createSheet06Flg As Boolean: createSheet06Flg = True
    Dim createSheet07Flg As Boolean: createSheet07Flg = True
    '●Sheet名
    sheetNm00 = "00_README"
    sheetNm01 = "01_JCL_STEP一覧"
    sheetNm02 = "02_JCL_STEP_DD一覧"
    sheetNm03 = "03_PGM一覧"
    sheetNm04 = "04_PGM_IO一覧"
    sheetNm05 = "05_PGM_CALL一覧"
    sheetNm06 = "06_PGM_COPY一覧"
    sheetNm07 = "07_CPY_項目一覧"
    '●フォルダパス(末尾 "\" 必要)
    jclFolderPath = ThisWorkbook.Path & "\COBOL\JCL\"
    pgmFolderPath = ThisWorkbook.Path & "\COBOL\PGM\"
    cpyFolderPath = ThisWorkbook.Path & "\COBOL\CPY\"
    '======================================================

    '==================== 初期化 ====================
    '●タイムスタンプ
    timestamp = Format(Now, "YYYYMMDDHHMMSS")
    '●Const
    EMP = ""
    HYPHEN = "-"
    '●ファイル種別
    FILETYPE_JCL = "JCL"
    FILETYPE_PGM = "PGM"
    FILETYPE_CPY = "CPY"
    '●ファイル名リスト
    Set jclFileNmList = GetFileNmList(FILETYPE_JCL)
    Set pgmFileNmList = GetFileNmList(FILETYPE_PGM)
    Set cpyFileNmList = GetFileNmList(FILETYPE_CPY)
    '================================================

    'Sheet作成
    If createSheet00Flg Then CreateSheet00 sheetNm00
    If createSheet01Flg Then CreateSheet01 sheetNm01
    If createSheet02Flg Then CreateSheet02 sheetNm02
    If createSheet03Flg Then CreateSheet03 sheetNm03
    If createSheet04Flg Then CreateSheet04 sheetNm04
    If createSheet05Flg Then CreateSheet05 sheetNm05
    If createSheet06Flg Then CreateSheet06 sheetNm06
    If createSheet07Flg Then CreateSheet07 sheetNm07

    'ステータスバー初期化
    Application.StatusBar = False
    '処理中描画ON設定
    RenderingProcess True
End Sub
'------------------------------------------------------------
'概要 : [00] COBOL Sheetを作成する
'引数 : wsNm : 作成Sheet名
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub CreateSheet00(ByVal wsNm As String)
    '●Sheet
    Dim ws As Worksheet

    Application.StatusBar = "[" & wsNm & "] 書込中..."

    '書込初期化
    InitWriteContentsCol 1
    'Sheet作成
    Set ws = CreateNewSheet(wsNm)

    '内容
    writeContentsCol.Add Array("【解析対象】")
    writeContentsCol.Add Array("JCLフォルダパス", jclFolderPath)
    writeContentsCol.Add Array("PGMフォルダパス", pgmFolderPath)
    writeContentsCol.Add Array("CPYフォルダパス", cpyFolderPath)
    writeContentsCol.Add Array("JCLファイル数", jclFileNmList.count)
    writeContentsCol.Add Array("PGMファイル数", pgmFileNmList.count)
    writeContentsCol.Add Array("CPYファイル数", cpyFileNmList.count)
    writeContentsCol.Add Array("")
    writeContentsCol.Add Array("【各Sheet説明】")
    writeContentsCol.Add Array(sheetNm01, "JCL内のSTEP一覧")
    writeContentsCol.Add Array(sheetNm02, "JCL内のSTEP-DD一覧")
    writeContentsCol.Add Array(sheetNm03, "PGM一覧")
    writeContentsCol.Add Array(sheetNm04, "PGM内のOPENファイル一覧")
    writeContentsCol.Add Array(sheetNm05, "PGM内のCALL行一覧")
    writeContentsCol.Add Array(sheetNm06, "PGM内のCOPY行一覧")
    writeContentsCol.Add Array(sheetNm07, "CPY内の項目一覧")
    '一括書込
    BulkWriteCells ws, writeRowIdx, 1
    'Sheet設定
    ws.Activate
    ActiveWindow.DisplayGridlines = False '目盛線
    With ws
        .Columns("A").ColumnWidth = 30 '列幅
        .Columns("B").ColumnWidth = 30
    End With
End Sub
'------------------------------------------------------------
'概要 : [01] COBOL Sheetを作成する
'引数 : wsNm : 作成Sheet名
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub CreateSheet01(ByVal wsNm As String)
    '●Sheet
    Dim ws As Worksheet
    '●ファイル内容/ファイル名/行データ/行番号
    Dim fileContents As New Collection
    Dim fileNm As Variant
    Dim line As Variant
    Dim lineNo As String
    '●JCL/JOB/STEP/EXEC PGM/COND/PARM/TIME/REGION/DYNAMNBR/ファイル有無
    Dim jcl As String
    Dim job As String
    Dim step As String
    Dim execPgm As String
    Dim cond As String
    Dim parm As String
    Dim time As String
    Dim region As String
    Dim dynamnbr As String
    Dim existsFile As String
    '●EXEC PGM行有無
    Dim execPgmFlg As Boolean

    Dim rowEndIdx As Long  '終了行番号
    Dim colEndIdx As Long  '終了列番号
    Dim rng As Range       '範囲
    Dim fileCnt As Long    'ファイル数
    Dim allFileCnt As Long '全ファイル数

    '書込 初期化
    InitWriteContentsCol 1
    'Sheet作成
    Set ws = CreateNewSheet(wsNm)
    'ヘッダー
    writeContentsCol.Add Array("JCLファイル", "JOB", "STEP", "EXEC PGM", "COND", "PARM", "TIME", "REGION", "DYNAMNBR", "PGMファイル有無", "行番号", "元", "備考")
    '終了列セット
    colEndIdx = GetMaxColSize(writeContentsCol)

    'データ行
    fileCnt = 1
    allFileCnt = jclFileNmList.count
    For Each fileNm In jclFileNmList
        Application.StatusBar = "[" & wsNm & "] 書込中...(" & fileCnt & "/" & allFileCnt & ")"
        fileCnt = fileCnt + 1
        '初期化
        execPgmFlg = False
        job = ""
        'ファイル内容 取得
        Set fileContents = GetFileContents(FILETYPE_JCL, fileNm)
        '拡張子除去
        jcl = GetFileBaseNm(fileNm)
        '全行Loop
        For Each line In fileContents
            'JOB取得
            If job = "" Then
                job = GetJob(line)
                If job <> "" Then
                    '行番号 取得
                    lineNo = ConvertEmpToHyphen(GetLineNo(FILETYPE_JCL, line))
                    writeContentsCol.Add Array(jcl, job, HYPHEN, HYPHEN, HYPHEN, HYPHEN, HYPHEN, HYPHEN, HYPHEN, HYPHEN, lineNo, line)
                End If
            End If
            'EXEC PGM行の場合
            If IsExecPgmLine(line) Then
                execPgmFlg = True
                '値取得
                step = ConvertEmpToHyphen(GetStep(line))
                execPgm = GetExecPgm(line)
                cond = ConvertEmpToHyphen(GPGetVal(line, "COND="))
                parm = ConvertEmpToHyphen(GPGetVal(line, "PARM="))
                time = ConvertEmpToHyphen(GPGetVal(line, "TIME="))
                region = ConvertEmpToHyphen(GPGetVal(line, "REGION="))
                dynamnbr = ConvertEmpToHyphen(GPGetVal(line, "DYNAMNBR="))
                existsFile = ConvertFlgToMark(IsExistsFile(FILETYPE_PGM, line))
                lineNo = ConvertEmpToHyphen(GetLineNo(FILETYPE_JCL, line))
                writeContentsCol.Add Array(jcl, job, step, execPgm, cond, parm, time, region, dynamnbr, existsFile, lineNo, line)
            End If
        Next
        'EXEC PGM無しの場合
        If Not execPgmFlg Then writeContentsCol.Add Array(jcl, job, "無し", HYPHEN, HYPHEN, HYPHEN, HYPHEN, HYPHEN, HYPHEN, HYPHEN, HYPHEN, HYPHEN)
        '一括書込
        BulkWriteCells ws, writeRowIdx, 1
        '書込 初期化
        InitWriteContentsCol writeRowIdx + writeContentsCol.count
        '応答なし 対策
        DoEvents
    Next

    '終了行セット
    rowEndIdx = ws.Cells(ws.Rows.count, "A").End(xlUp).row
    '表範囲セット
    Set rng = Range(ws.Cells(1, 1), ws.Cells(rowEndIdx, colEndIdx))

    'Sheet共通設定/個別設定
    CommonSheetSetting ws
    ApplyTableStyle01 ws, rng, 5 'テーブル
    ApplyLineStyle01 ws, rng, 1 '罫線
    With ws
        .Columns("A").ColumnWidth = 15 '列幅
        .Columns("B").ColumnWidth = 10
        .Columns("C").ColumnWidth = 10
        .Columns("D").ColumnWidth = 13
        .Columns("E").ColumnWidth = 18
        .Columns("F").ColumnWidth = 18
        .Columns("G").ColumnWidth = 18
        .Columns("H").ColumnWidth = 18
        .Columns("I").ColumnWidth = 18
        .Columns("J").ColumnWidth = 18
        .Columns("K").ColumnWidth = 10
        .Columns("L").ColumnWidth = 90
        .Columns("M").ColumnWidth = 50
        .Range("G2:I" & rowEndIdx).HorizontalAlignment = xlRight '文字揃え
        .Range("J2:J" & rowEndIdx).HorizontalAlignment = xlCenter
        .Range("K2:K" & rowEndIdx).HorizontalAlignment = xlRight
    End With
End Sub

    Dim colEndIdx As Long  '終了列番号

Private Sub BulkWriteCells(ByRef ws As Worksheet, ByVal rowStartIdx As Long, ByVal colStartIdx As Long)
    '●書込内容(配列)

    '●リスト内の最大項目数
    Dim maxColSize As Long: maxColSize = GetMaxColSize(writeContentsCol)

'------------------------------------------------------------
'概要 : リスト内の最大項目数を取得する
'     : col(1) [AAA, BBB]
'     : col(2) [AAA, BBB, CCC]
'     : col(3) [AAA]
'     :   → 2行目の項目数3を返す
'引数 : col : リスト
'戻値 : リスト内の最大項目数
'備考 : -
'------------------------------------------------------------
Private Function GetMaxColSize(ByRef col As Collection) As Long
    Dim i As Long
    Dim maxColSize As Long
    For i = 1 To col.count
        maxColSize = Application.Max(maxColSize, UBound(col(i)) + 1)
    Next
    GetMaxColSize = maxColSize
End Function
'------------------------------------------------------------
'概要 : Collectionから配列へ変換する
'引数 : col        : リスト
'     : maxColSize : リスト内の最大項目数
'戻値 : 配列
'備考 : -
'------------------------------------------------------------
Private Function ConvertCollectionToArray(ByRef col As Collection, ByVal maxColSize As Long) As Variant
    '●配列
    Dim arr As Variant: ReDim arr(1 To col.count, 1 To maxColSize)
    '●行番号/列番号
    Dim rowIdx As Long
    Dim colIdx As Long
    For rowIdx = 1 To col.count
        '配列の場合
        If IsArray(col(rowIdx)) Then
            For colIdx = 1 To UBound(col(rowIdx)) + 1
                arr(rowIdx, colIdx) = col(rowIdx)(colIdx - 1)
            Next
        Else
            arr(rowIdx, 1) = col(rowIdx)
        End If
    Next
    ConvertCollectionToArray = arr
End Function
'------------------------------------------------------------
'概要 : ファイルが存在するか判定する
'引数 : fileType : ファイル種別(JCL/PGM/CPY)
'     : fileNm   : ファイル名(拡張子無しの場合、ファイル種別に応じて付与する)
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsFile(ByVal fileType As String, ByVal fileNm As String) As Boolean
    '●ファイルパス
    Dim filePath As String
    'シングルクォート除去
    If 0 < InStr(fileNm, "'") Then fileNm = Replace(fileNm, "'", "")
    '設定
    Select Case fileType
        Case FILETYPE_JCL
            If InStr(fileNm, ".") = 0 Then fileNm = fileNm & ".jcl"
            filePath = jclFolderPath & fileNm
        Case FILETYPE_PGM
            If InStr(fileNm, ".") = 0 Then fileNm = fileNm & ".cob"
            filePath = jclFolderPath & fileNm
        Case FILETYPE_CPY
            If InStr(fileNm, ".") = 0 Then fileNm = fileNm & ".cpy"
            filePath = jclFolderPath & fileNm
    End Select
    '存在しない場合
    If Dir(filePath) = "" Then
        IsExistsFile = False
    Else
        IsExistsFile = True
    End If
End Function
'------------------------------------------------------------
'概要 : ファイル名一覧を取得する(ファイル種別の拡張子ファイルのみ取得する)
'引数 : fileType : ファイル種別(JCL/PGM/CPY)
'戻値 : ファイル名リスト
'備考 : -
'------------------------------------------------------------
Private Function GetFileNmList(ByVal fileType As String) As Collection
    Dim col As New Collection
    '●フォルダパス/ファイル名/拡張子
    Dim folderPath As String
    Dim fileNm As String
    Dim ext As String
    '設定
    Select Case fileType
        Case FILETYPE_JCL
            folderPath = jclFolderPath
            ext = ".jcl"
        Case FILETYPE_PGM
            folderPath = pgmFolderPath
            ext = ".cob"
        Case FILETYPE_CPY
            folderPath = cpyFolderPath
            ext = ".cpy"
    End Select
    '全ファイル名 取得
    fileNm = Dir(folderPath & "*" & ext)
    Do While fileNm <> ""
        col.Add fileNm
        fileNm = Dir()
    Loop
    Set GetFileNmList = col
End Function
'------------------------------------------------------------
'概要 : ファイル名から拡張子を除去する
'引数 : fileNm : ファイル名
'戻値 : 拡張子除去済みファイル名
'備考 : -
'------------------------------------------------------------
Private Function GetFileBaseNm(ByVal fileNm As String) As String
    Dim pos As Long: pos = InStr(fileNm, ".")
    If pos = 0 Then
        GetFileBaseNm = fileNm
    Else
        GetFileBaseNm = Left(fileNm, pos - 1)
    End If
End Function
'------------------------------------------------------------
'概要 : ファイル内容を取得する(改行は全てLFに変換する)
'引数 : fileType : ファイル種別(JCL/PGM/CPY)
'     : fileNm   : ファイル名(拡張子無しの場合、ファイル種別に応じて付与する)
'戻値 : ファイル内容行リスト
'備考 : -
'------------------------------------------------------------
Private Function GetFileContents(ByVal fileType As String, ByVal fileNm As String) As Collection
    Dim col As Collection
    '●フォルダパス/ファイルパス
    Dim folderPath As String
    Dim filePath As String
    '●ファイル読込用
    Dim fileNo As Long: fileNo = FreeFile
    '●ファイル内容/全行データ/行データ
    Dim fileContents As String
    Dim lines As String
    Dim line As String
    '設定
    Select Case fileType
        Case FILETYPE_JCL
            folderPath = jclFolderPath
            If InStr(fileNm, ".") = 0 Then fileNm = fileNm & ".jcl"
        Case FILETYPE_PGM
            folderPath = pgmFolderPath
            If InStr(fileNm, ".") = 0 Then fileNm = fileNm & ".pgm"
        Case FILETYPE_CPY
            folderPath = cpyFolderPath
            If InStr(fileNm, ".") = 0 Then fileNm = fileNm & ".cpy"
    End Select
    '存在する場合
    If Dir(folderPath & fileNm) <> "" Then
        'ファイル内容 取得
        Open folderPath & fileNm For Binary As #fileNo
        fileContents = space(LOF(fileNo))
        Get #fileNo, , fileContents
        Close #fileNo
        '改行統一(LFに統一)
        fileContents = Replace(fileContents, vbCrLf, vbLf)
        fileContents = Replace(fileContents, vbCr, vbLf)
        '行分割
        lines = Split(fileContents, vbLf)
        '結果セット
        For Each line In lines
            col.Add line
        Next
    End If
    Set GetFileContents = col
End Function

'------------------------------------------------------------
'概要 : 真偽値をマークに変換する
'引数 : flg : 真偽値(True/False)
'戻値 : マーク
'備考 : -

'------------------------------------------------------------
'概要 : コメント行か判定する
'引数 : fileType : ファイル種別(JCL/PGM/CPY)
'     : line     : 行データ
'戻値 : 判定結果(True:である / False:でない)
'備考 : -
'------------------------------------------------------------
Private Function IsCmntLine(ByVal fileType As String, ByVal line As String) As Boolean
    Dim result As Boolean: result = False
    '判定
    Select Case fileType
        Case FILETYPE_JCL
            If mid(line, 15, 3) = "//*" Then result = True
        Case FILETYPE_PGM, FILETYPE_CPY
            If mid(line, 7, 1) = "*" Or mid(line, 7, 1) = "/" Then result = True
    End Select
    IsCmntLine = result
End Function
'------------------------------------------------------------
'概要 : 新しい行か判定する(簡易)
'引数 : line : 行データ
'戻値 : 判定結果(True:である / False:でない)
'備考 : -
'------------------------------------------------------------
Private Function IsNewStmt(ByVal line As String) As Boolean
    Dim matches As New Collection
    Dim match As Variant
    matches.Add (" MOVE ")
    matches.Add (" READ ")
    matches.Add (" WRITE ")
    matches.Add (" CALL ")
    matches.Add (" IF ")
    matches.Add (" PERFORM ")
    For Each match In matches
        If 0 < InStr(line, match) Then
            IsNewStmt = True
            Exit Function
        End If
    Next
End Function

'------------------------------------------------------------
'概要 : OPEN行一覧を取得する
'引数 : fileContents : ファイル内容行リスト
'戻値 : 対象行リスト
'備考 : -
'------------------------------------------------------------
Private Function GetOpenLineList(ByRef fileContents As Collection) As Collection
    Dim i As Long
    Dim str As String
    Dim line As String
    Dim nextLine As String
    Dim col As New Collection
    Dim match1 As String: match1 = "OPENINPUT"
    Dim match2 As String: match2 = "OPENOUTPUT"
    Dim match3 As String: match3 = "OPENI-O"
    Dim match4 As String: match4 = "OPENEXTEND"
    '●桁
    Dim posIn As Long
    Dim posOut As Long
    Dim posIo As Long
    Dim posEx As Long
    For i = 1 To fileContents.count
        line = Left(CStr(fileContents(i)), 72)
        'コメント行でない
        If Not IsCmntLine(FILETYPE_PGM, line) Then
            str = Replace(line, " ", "")
            posIn = InStr(str, match1)
            posOut = InStr(str, match2)
            posIo = InStr(str, match3)
            posEx = InStr(str, match4)
            '有りの場合
            If 0 < poin Or 0 < poout Or 0 < poio Or 0 < poex Then
                '継続行判定
                Do While Right(Trim(line), 1) <> "." And i < fileContents.count
                    i = i + 1
                    nextLine = Left(CStr(fileContents(i)), 72)
                    'コメント行でない & 継続行でない場合
                    If Not IsCmntLine(FILETYPE_PGM, line) And Not IsNewStmt(fileContents(i)) Then
                        line = Trim(line) & " " & Trim(mid(nextLine, 8))
                    Else
                        Exit Do
                    End If
                Loop
            End If
        End If
    Next
    Set GetOpenLineList = col
End Function

'------------------------------------------------------------
'概要 : CPYファイル内容を取得する(コメント行など除外)
'引数 : fileContents : ファイル内容行リスト
'戻値 : 対象行リスト
'備考 : -
'------------------------------------------------------------
Private Function GetCpyFileLineList(ByRef fileContents As Collection) As Collection
    Dim i As Long
    Dim str As String
    Dim line As String
    Dim nextLine As String
    Dim col As New Collection
    Dim pos1 As Long
    Dim pos2 As Long
    Dim lv88Flg As Boolean
    For i = 1 To fileContents.count
        line = Left(CStr(fileContents(i)), 72)
        '72桁までスペース埋め
        line = line & space(72 - Len(line))
        'コメント行でない & 空行でない場合
        If Not IsCmntLine(FILETYPE_CPY, line) And Trim(mid(line, 8)) <> "" Then
            'Lv88フラグ セット
            lv88Flg = (GetCopyLv(line) = "88")
            Do While i < fileContents.count
                nextLine = Left(CStr(fileContents(i + 1)), 72)
                '72桁までスペース埋め
                nextLine = nextLine & space(72 - Len(nextLine))
                'コメント行でない & 空行でない場合
                If Not IsCmntLine(FILETYPE_CPY, nextLine) And Trim(mid(nextLine, 8)) <> "" Then
                    '継続行の場合
                    If mid(nextLine, 7, 1) = "-" Then
                        'シングルクォートが奇数個の場合(VALUE値途中終了の場合)
                        If UBound(Split(nextLine, "'")) Mod 2 <> 0 Then
                            '以降 切り抜き
                            line = line & mid(nextLine, InStr(nextLine, "'") + 1)
                        'シングルクォート固定値のみの場合(継続VALUE値有りの場合)
                        ElseIf Left(Trim(mid(nextLine, 8)), 1) = "'" And (Right(Trim(mid(nextLine, 8)), 1) = "'" Or Right(Trim(mid(nextLine, 8)), 1) = ".") Then
                            pos1 = InStr(nextLine, "'") '1個目
                            pos1 = InStr(mid(nextLine, pos1 + 1), "'") '1個目
                            line = line & mid(nextLine, pos1 + 1, pos2)
                        Else
                            line = line & mid(nextLine, 8)
                        End If
                    'Lv88の継続行の場合
                    ElseIf lv88Flg And GetCopyLv(nextLine) = "" Then
                        line = line & mid(nextLine, 8)
                    Else
                        Exit Do
                    End If
                End If
                i = i + 1
            Loop
            col.Add line
        End If
    Next
    GetCpyFileLineList = col
End Function

'引数 : line  : 行データ
'     : match : 検索文字列
'戻値 : 値
'備考 : -
'------------------------------------------------------------
Private Function GPGetVal(ByVal line As String, ByVal match As String) As String
    Dim str As String: str = ""
    Dim pos As Long: pos = InStr(line, match)
    '有りの場合
    If 0 < pos Then
        str = Trim(mid(line, pos + Len(match)))
        '"'" 始まりの場合
        If Left(str, 1) = "'" Then
            '切り抜き
            srt = "'" & Left(str, InStr(mid(str, 2), "'") + 1)
        '"(" 始まりの場合
        ElseIf Left(str, 1) = "(" Then
            srt = Left(str, InStr(mid(str, 2), ")") + 1)
        Else
            '手前 切り抜き
            pos = InStr(str, ",")
            If 0 < pos Then str = Left(str, pos - 1)
            pos = InStr(str, ".")
            If 0 < pos Then str = Left(str, pos - 1)
            pos = InStr(str, " ")
            If 0 < pos Then str = Left(str, pos - 1)
        End If
    End If
    GPGetVal = str
End Function

        '以降 切り抜き
        vals = Trim(mid(line, pos + Len(match)))
        If vals <> "" Then
            '除去
            If Right(vals, 1) = "." Then vals = Left(vals, Len(vals) - 1)
            '固定値の場合
            If Left(vals, 1) = "'" Then
                Do While vals <> ""
                    '終了シングルクォートまで切り抜き
                    str = Left(vals.InStr(mid(vals, 2), "'") + 1)
                    pos = InStr(vals, str)
                    vals = Trim(mid(vals, pos + Len(str)))
                    col.Add str
                Loop
            Else
                If 0 < InStr(vals, " ") Then vals = Left(vals, InStr(vals, " "))
                col.Add vals
            End If
        End If
    End If
    '1行文字列に変換
    For Each colStr In col
        result = result & colStr & " "
    Next
    '末尾空白除去
    If result <> "" Then result = Left(result, Len(result) - 1)
    GetCopyValue = result
End Function
