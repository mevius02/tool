'------------------------------------------------------------
'概要 : Sheet[目次]を作成する
'引数 : -
'戻値 : -
'備考 : -
'------------------------------------------------------------
Public Sub CreateSheet_TableOfContents()
    '処理中描画OFF設定
    Call SettingRenderingProcess(False)

    'Book/Sheet/Sheet(TOP)/行/範囲
    Dim wb As Workbook: Set wb = ThisWorkbook
    Dim ws As Worksheet
    Dim wsTop As Worksheet
    Dim wsNmTop As String: wsNmTop = "目次"
    Dim rowIdx As Long: rowIdx = 4
    Dim rng As Range
    '●項目No
    Dim colNo As Long: colNo = 1

    '新規Sheet作成 & 新規Sheet名セット
    Set wsTop = ThisWorkbook.Sheets.Add(Before:=ThisWorkbook.Worksheets(1))
    wsNmTop = ChangeUniqueSheetNm(wsNmTop)
    wsTop.Name = wsNmTop

    '書込
    wsTop.Range("B2").value = "目次"

    '全Sheet Loop
    For Each ws In wb.Worksheets
        '自分以外の場合
        If ws.Name <> wsNmTop Then
            wsTop.Cells(rowIdx, "B").value = colNo & ","
            'セル場所 セット
            Set rng = wsTop.Cells(rowIdx, "C")
            'Sheet名 書込
            rng.value = ws.Name
            'ハイパーリンク セット
            Call wsTop.Hyperlinks.Add(Anchor:=rng, Address:="", SubAddress:="'" & ws.Name & "'!A1", TextToDisplay:=ws.Name)
            'インクリメント
            colNo = colNo + 1
            rowIdx = rowIdx + 1
        End If
    Next

    'Sheet設定
    With wsTop
        '文字揃え
        .Cells.VerticalAlignment = xlTop
        .Cells.HorizontalAlignment = xlLeft
        '列幅
        .Columns.ColumnWidth = 2
        .Columns("B").ColumnWidth = 4
        .Columns("C").ColumnWidth = 50
    End With
    wsTop.Activate
    '表示倍率
    ActiveWindow.Zoom = 85
    '目盛線
    ActiveWindow.DisplayGridlines = False

    '処理中描画ON設定
    Call SettingRenderingProcess(True)
End Sub

    'Loopカウンタ/終了行/書込行
    Dim i As Long
    Dim endRowIdx As Long
    Dim wtriteRowIdx As Long: wtriteRowIdx = 2

    '●TBLNo/項目No
    Dim tblNo As Long: tblNo = 1
    Dim colNo As Long
    '●テーブル名(論理/物理)
    Dim logiTblNm As String
    Dim physiTblNm As String
    '●項目定義データ
    Dim colInfo As Variant
    '●PK/項目名(論理/物理)/型/桁/NULL不可/初期値
    Dim pk As String
    Dim logiColNm As String
    Dim physiColNm As String
    Dim colType As String
    Dim colLen As Variant
    Dim notNull As String
    Dim initVal As String
    '●一括書込用
    Dim balkDatas As Variant

    '●A5M2緑色
    Dim a5m2Color As Long: a5m2Color = RGB(135, 231, 173) '#87E7AD

    '新規Sheet作成 & 新規Sheet名セット

    wsNmInfo = ChangeUniqueSheetNm(wsNmInfo)
    wsInfo.Name = wsNmInfo

    '========== 書込 ==========
    wsInfo.Range("A1").value = "TBL-No"
    wsInfo.Range("B1").value = "TBL(論理)"
    wsInfo.Range("C1").value = "TBL(物理)"
    wsInfo.Range("D1").value = "COL-No"
    wsInfo.Range("E1").value = "PK"
    wsInfo.Range("F1").value = "COL(論理)"
    wsInfo.Range("G1").value = "COL(物理)"
    wsInfo.Range("H1").value = "型"
    wsInfo.Range("I1").value = "桁"
    wsInfo.Range("J1").value = "NULL不可"
    wsInfo.Range("K1").value = "初期値"
    '全Sheet Loop
    For Each ws In wb.Worksheets
        '自分以外 & TBL定義Sheetの場合
        If ws.Name <> wsNmInfo And Trim$(ws.Range("A1").Value2) = "DB" Then
            '終了行 セット
            endRowIdx = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
            'データ有の場合
            If 7 <= endRowIdx Then
                '項目定義 取得
                colInfo = ws.Range(ws.Range("B7"), ws.Cells(endRowIdx, "H")).Value2
                '一括書込変数 再設定
                ReDim balkDatas(1 To endRowIdx - 6, 1 To 11)
                '全行Loop
                colNo = 1
                For i = 1 To UBound(colInfo, 1)
                    '一括セット
                    balkDatas(i, 1) = tblNo                        'TBL-No
                    balkDatas(i, 2) = Trim$(ws.Range("B3").Value2) 'TBL名(論理)
                    balkDatas(i, 3) = Trim$(ws.Range("B4").Value2) 'TBL名(物理)
                    balkDatas(i, 4) = colNo                        'COL-No
                    balkDatas(i, 5) = Trim$(colInfo(i, 1))         'PK
                    balkDatas(i, 6) = Trim$(colInfo(i, 2))         '項目名(論理)
                    balkDatas(i, 7) = Trim$(colInfo(i, 3))         '項目名(物理)
                    balkDatas(i, 8) = Trim$(colInfo(i, 4))         '型
                    balkDatas(i, 9) = Trim$(colInfo(i, 5))         '桁
                    balkDatas(i, 10) = Trim$(colInfo(i, 6))        'NULL不可
                    balkDatas(i, 11) = Trim$(colInfo(i, 7))        '初期値
                    colNo = colNo + 1
                Next
                '書込
                wsInfo.Cells(wtriteRowIdx, "A").Resize(UBound(colInfo, 1), 11).value = balkDatas
                tblNo = tblNo + 1
                wtriteRowIdx = wtriteRowIdx + UBound(colInfo, 1)
            End If
        End If
    Next

    '========== Sheet設定 ==========
    With wsInfo
        'Tab色
        .Tab.Color = a5m2Color
        '背景色
        .Range("A1:K1").Interior.Color = a5m2Color
        '太字
        .Range("A1:K1").Font.Bold = True

        .Columns("I").HorizontalAlignment = xlRight

        .Columns("B:C").ColumnWidth = 30
        .Columns("D").ColumnWidth = 10

        .Columns("F:G").ColumnWidth = 30

'------------------------------------------------------------
'概要 : Sheet名のSheetが存在するか判定する
'引数 : wsNm : Sheet名
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsSheet(ByVal wsNm As String) As Boolean
    Dim ws As Worksheet
    For Each ws In Sheets
        '存在する場合
        If ws.Name = wsNm Then
            IsExistsSheet = True
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : フォルダが存在するか判定する
'引数 : path     : フォルダまでのパス(絶対パス)
'     : folderNm : フォルダ名
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsFolder(ByVal path As String, ByVal folderNm As String) As Boolean
    '末尾付与
    If Right$(path, 1) <> "\" Then path = path & "\"
    '判定(Dirだけだと同名ファイルにもヒットするためGetAttrで再確認)
    If Dir(path & folderNm, vbDirectory) = "" Then
        IsExistsFolder = False
    Else
        IsExistsFolder = (GetAttr(path & folderNm) And vbDirectory) = vbDirectory
    End If
End Function
'------------------------------------------------------------
'概要 : 重複しないフォルダ名となるまで、末尾に連番を付与する
'     : (同じ名前のフォルダが存在しない場合、そのまま返す)
'     : (同じ名前のフォルダが存在する場合、2からの連番を末尾に付与し、重複しないフォルダ名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : path     : フォルダまでのパス
'     : folderNm : フォルダ名
'戻値 : フォルダ名
'備考 : -
'------------------------------------------------------------
Private Function ChangeUniqueFolderNm(ByVal path As String, ByVal folderNm As String) As String
    Dim i As Long
    Dim uniqueNo As String
    '存在する場合
    For i = 2 To 1000
        '存在しない場合
        If Not IsExistsFolder(path, folderNm & uniqueNo) Then
            ChangeUniqueFolderNm = folderNm & uniqueNo
            Exit Function
        End If
        '末尾付与
        uniqueNo = " (" & i & ")"
    Next
End Function
'------------------------------------------------------------
'概要 : 指定パスとなるフォルダを、存在しない階層も含めて全て作成する
'     : (途中でエラーが発生した場合、今回新規作成した分だけを削除してロールバックする)
'引数 : path : 作成したいフォルダの絶対パス
'戻値 : 判定結果(True:全て作成できた(元々あった場合も含む) / False:途中でエラー発生、ロールバック済み)
'備考 : -
'------------------------------------------------------------
Private Function CreateFolderPath(ByVal path As String) As Boolean
    On Error GoTo ErrHandler
    Dim i As Long
    Dim parts() As String
    Dim crntPath As String
    '作成フォルダパス(ロールバック用)
    Dim cratPaths() As String
    Dim cratCnt As Long
    '最大分だけ確保
    ReDim cratPaths(UBound(Split(path, "\")))
    cratCnt = 0
    '末尾除去
    If Right$(path, 1) = "\" Then path = Left$(path, Len(path) - 1)
    parts = Split(path, "\")
    crntPath = parts(0)
    For i = 1 To UBound(parts)
        If Not IsExistsFolder(crntPath, parts(i)) Then
            MkDir crntPath & "\" & parts(i)
            '作成フォルダパス 追加
            cratPaths(cratCnt) = crntPath & "\" & parts(i)
            cratCnt = cratCnt + 1
        End If
        crntPath = crntPath & "\" & parts(i)
    Next
    CreateFolderPath = True
    Exit Function
ErrHandler:
    'ロールバック(逆順で削除)
    For i = cratCnt - 1 To 0 Step -1
        '削除中の追加エラーは無視して続行
        On Error Resume Next
        RmDir cratPaths(i)
        On Error GoTo 0
    Next
    CreateFolderPath = False
End Function

'概要 : ファイルが存在するか判定する
'引数 : path   : ファイルまでのパス(絶対パス)
'     : fileNm : ファイル名
'     : ext    : 拡張子
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsFile(ByVal path As String, ByVal fileNm As String, ByVal ext As String) As Boolean
    '末尾付与 & 先頭付与
    If Right$(path, 1) <> "\" Then path = path & "\"
    If Left$(ext, 1) <> "." Then ext = "." & ext
    '判定
    IsExistsFile = (Dir(path & fileNm & ext) <> "")
End Function
'------------------------------------------------------------
'概要 : 重複しないファイル名となるまで、末尾に連番を付与する
'     : (同じ名前のファイルが存在しない場合、そのまま返す)
'     : (同じ名前のファイルが存在する場合、2からの連番を末尾に付与し、重複しないファイル名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : path   : ファイルまでのパス
'     : fileNm : ファイル名
'     : ext    : 拡張子
'戻値 : ファイル名(拡張子付き)
'備考 : -
'------------------------------------------------------------
Private Function ChangeUniqueFileNm(ByVal path As String, ByVal fileNm As String, ByVal ext As String) As String
    Dim i As Long
    Dim uniqueNo As String
    For i = 2 To 1000
        '存在しない場合
        If Not IsExistsFile(path, fileNm & uniqueNo, ext) Then
            ChangeUniqueFileNm = fileNm & uniqueNo & "." & ext
            Exit Function
        End If
        '末尾付与
        uniqueNo = " (" & i & ")"
    Next
End Function
'------------------------------------------------------------
'概要 : 重複しないSheet名となるまで、末尾に連番を付与する
'     : (同じ名前のSheetが存在しない場合、そのまま返す)
'     : (同じ名前のSheetが存在する場合、2からの連番を末尾に付与し、重複しないSheet名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : wsNm : Sheet名
'戻値 : Sheet名
'備考 : -
'------------------------------------------------------------
Private Function ChangeUniqueSheetNm(ByVal wsNm As String) As String
    Dim i As Long
    Dim uniqueNo As String
    For i = 2 To 1000
        '存在しない場合
        If Not IsExistsSheet(wsNm & uniqueNo) Then
            ChangeUniqueSheetNm = wsNm & uniqueNo
            Exit Function
        End If
        '末尾付与
        uniqueNo = " (" & i & ")"
    Next
End Function
'------------------------------------------------------------
'概要 : 列番号から列名(アルファベット)を取得する
'     : (範囲外の数値の場合、空を返す)
'引数 : colIdx : 列番号
'戻値 : 列名(アルファベット)
'備考 : -
'------------------------------------------------------------
Private Function GetColNm(ByVal colIdx As Long) As String
    Dim result As String
    '範囲外の場合
    If colIdx < 1 Or Application.Columns.Count < colIdx Then
        GetColNm = ""
        Exit Function
    End If
    Do While 0 < colIdx
        colIdx = colIdx - 1
        result = Chr$(65 + (colIdx Mod 26)) & result
        colIdx = colIdx \ 26
    Loop
    GetColNm = result
End Function
'------------------------------------------------------------
'概要 : 列名(アルファベット)から列番号を取得する
'     : (範囲外の列名の場合、0を返す)
'引数 : colNm : 列名(アルファベット)
'戻値 : 列番号
'備考 : -
'------------------------------------------------------------
Private Function GetColIdx(ByVal colNm As String) As Long
    Dim i As Long
    Dim result As Long
    Dim baseA As Long: baseA = Asc("A") - 1
    Dim colNmLen As Long: colNmLen = Len(UCase$(colNm))
    Dim ch As Long
    For i = 1 To colNmLen
        ch = Asc(Mid$(colNm, i, 1)) - baseA
        '範囲外の場合
        If ch < 1 Or 26 < ch Then
            GetColIdx = 0
            Exit Function
        End If
        result = result * 26 + ch
    Next
    GetColIdx = result
End Function

'------------------------------------------------------------
'概要 : 罫線(行境界線)[01]
'     : 境界判定列のセル値が変わる行の境界に罫線を引く
'引数 : ws     : Sheet
'     : rng    : 適用範囲
'     : colIdx : 境界判定列番号
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub ApplyLineStyle_01(ByRef ws As Worksheet, ByRef rng As Range, ByVal colIdx As Long)
    Dim i As Long
    '●名称(前/現在)
    Dim preVal As String
    Dim crntVal As String
    preVal = Left$(CStr(ws.Cells(rng.Row + 1, rng.Column + colIdx - 1)), 4)
    For i = rng.Row + 1 To rng.Row + rng.Rows.Count
        crntVal = Left$(CStr(ws.Cells(i, rng.Column + colIdx - 1)), 4)
        '値が異なる場合
        If preVal <> crntVal Then
            '罫線
            ws.Range(ws.Cells(i, rng.Column), ws.Cells(i, rng.Columns.Count)).Borders(xlEdgeTop).LineStyle = xlDouble
            ws.Range(ws.Cells(i, rng.Column), ws.Cells(i, rng.Columns.Count)).Borders(xlEdgeTop).Weight = xlThick
            preVal = crntVal
        End If
    Next
End Sub
'------------------------------------------------------------
'概要 : 型の候補を書込する
'引数 : -
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub Btn_SetColumnDataType()
    '処理中描画OFF設定
    Call SettingRenderingProcess(False)

    'Sheet
    Dim ws As Worksheet: Set ws = ActiveSheet
    Dim rowIdx As Long    '行
    Dim endRowIdx As Long '終了行
    '●DB名
    Dim dbNm As String: dbNm = ws.Range("B1").Value2
    '●入力値範囲データ(データ分類/日本語有無/桁数/小数桁数/時間有無/時差修正要否)
    Dim inputRngData As Variant
    '●型候補リスト
    Dim dataTypeList As Variant
    '●エラーメッセージ
    Dim errMsg As String

    '終了行 セット
    endRowIdx = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    '入力チェック(DB)
    If LenB(dbNm) = 0 Then
        Call MsgBox("[B1] にDBを入力して下さい", vbExclamation, "ERROR")
        Exit Sub
    End If

    '初期化
    ws.Range("O7:Q" & endRowIdx).value = ""
    ws.Range("T7:T" & endRowIdx).value = ""
    '全行Loop
    For rowIdx = 7 To endRowIdx
        '入力値範囲データ 取得(COL詳細)
        inputRngData = ws.Range(ws.Cells(rowIdx, "I"), ws.Cells(rowIdx, "N")).Value2
        '入力チェック
        errMsg = Valid_SetDataType(dbNm, inputRngData)
        'エラー有りの場合
        If 0 < LenB(errMsg) Then
            ws.Cells(rowIdx, "T").value = errMsg
        Else
            '型候補 取得
            dataTypeList = GetColDataTypes(dbNm, inputRngData)
            '書込
            ws.Cells(rowIdx, "O").value = CStr(dataTypeList(0))
            ws.Cells(rowIdx, "P").value = CStr(dataTypeList(1))
            ws.Cells(rowIdx, "Q").value = CStr(dataTypeList(2))
        End If
    Next

    '処理中描画ON設定
    Call SettingRenderingProcess(True)
End Sub
'------------------------------------------------------------
'概要 : 入力チェック(型候補セット用)
'引数 : dbNm         : DB名(Oracle / PostgreSQL / SQLServer)
'     : inputRngData : 入力値範囲データ(データ分類/日本語有無/桁数/小数桁数/時間有無/時差修正要否)
'戻値 : エラーメッセージ
'備考 : -
'------------------------------------------------------------
Private Function Valid_SetDataType(ByVal dbNm As String, ByVal inputRngData As Variant) As String
    '●データ分類
    Dim dataType As String: dataType = CStr(inputRngData(1, 1))
    '●エラーメッセージ/冒頭メッセージ
    Dim errMsg As String
    Dim errAtartMsg As String: errAtartMsg = "[データ分類]が""" & dataType & """の場合"

    '[データ分類]未入力
    If LenB(dataType) = 0 Then
        Valid_SetDataType = "[データ分類]を入力して下さい"
        Exit Function
    End If

    '***** 文字列 *****
    If dataType = "文字列" Then
        '[日本語有無]未入力
        If LenB(inputRngData(1, 2)) = 0 Then
            errMsg = errMsg & errAtartMsg & "[日本語有無]を指定して下さい" & vbCrLf
        End If
        '[桁数]未入力 or 数値以外 or 0
        If LenB(inputRngData(1, 3)) = 0 Then
            errMsg = errMsg & errAtartMsg & "[桁数]を入力して下さい" & vbCrLf
        ElseIf Not IsNumeric(inputRngData(1, 3)) Then
            errMsg = errMsg & "[桁数]は数値を入力して下さい" & vbCrLf
        ElseIf CLng(inputRngData(1, 3)) = 0 Then
            errMsg = errMsg & errAtartMsg & "[桁数]に1以上を入力して下さい" & vbCrLf
        End If
        '[小数桁数]入力有
        If 0 < LenB(inputRngData(1, 4)) And inputRngData(1, 4) <> "0" Then
            errMsg = errMsg & errAtartMsg & "[小数桁数]は指定できません" & vbCrLf
        End If
        '[時間有無]入力有
        If 0 < LenB(inputRngData(1, 5)) And inputRngData(1, 5) <> "無" Then
            errMsg = errMsg & errAtartMsg & "[時間有無]は指定できません" & vbCrLf
        End If
        '[時差修正要否]入力有
        If 0 < LenB(inputRngData(1, 6)) And inputRngData(1, 6) <> "否" Then
            errMsg = errMsg & errAtartMsg & "[時差修正要否]は指定できません" & vbCrLf
        End If
    End If

    '***** 数値 *****
    If dataType = "数値" Then
        '[日本語有無]入力有
        If 0 < LenB(inputRngData(1, 2)) And inputRngData(1, 2) <> "無" Then
            errMsg = errMsg & errAtartMsg & "[日本語有無]は指定できません" & vbCrLf
        End If
        '[桁数]未入力 or 数値以外 or 0 or 16以上
        If LenB(inputRngData(1, 3)) = 0 Then
            errMsg = errMsg & errAtartMsg & "[桁数]を入力して下さい" & vbCrLf
        ElseIf Not IsNumeric(inputRngData(1, 3)) Then
            errMsg = errMsg & "[桁数]は数値を入力して下さい" & vbCrLf
        ElseIf CLng(inputRngData(1, 3)) = 0 Then
            errMsg = errMsg & errAtartMsg & "[桁数]に1以上を入力して下さい" & vbCrLf
        ElseIf 15 < CLng(inputRngData(1, 3)) Then
            errMsg = errMsg & errAtartMsg & "[桁数]は15以下で入力して下さい" & vbCrLf
        End If
        '[小数桁数]未入力 or 数値以外 or 6以上
        If LenB(inputRngData(1, 4)) = 0 Then
            errMsg = errMsg & errAtartMsg & "[小数桁数]に0以上を入力して下さい" & vbCrLf
        ElseIf Not IsNumeric(inputRngData(1, 4)) Then
            errMsg = errMsg & "[小数桁数]は数値を入力して下さい" & vbCrLf
        ElseIf 5 < CLng(inputRngData(1, 4)) Then
            errMsg = errMsg & "[小数桁数]は5以下で入力して下さい" & vbCrLf
        End If
        '[小数桁数/全体桁数] 逆転
        If 0 < LenB(inputRngData(1, 3)) And 0 < LenB(inputRngData(1, 4)) Then
            If IsNumeric(inputRngData(1, 3)) And IsNumeric(inputRngData(1, 4)) Then
                If CLng(inputRngData(1, 3)) <= CLng(inputRngData(1, 4)) Then
                    errMsg = errMsg & "[桁数]は[小数桁数]より大きい数値を入力して下さい" & vbCrLf
                End If
            End If
        End If
        '[時間有無]入力有
        If 0 < LenB(inputRngData(1, 5)) And inputRngData(1, 5) <> "無" Then
            errMsg = errMsg & errAtartMsg & "[時間有無]は指定できません" & vbCrLf
        End If
        '[時差修正要否]入力有
        If 0 < LenB(inputRngData(1, 6)) And inputRngData(1, 6) <> "否" Then
            errMsg = errMsg & errAtartMsg & "[時差修正要否]は指定できません" & vbCrLf
        End If
    End If

    '***** 日時 *****
    If dataType = "日時" Then
        '[日本語有無]入力有
        If 0 < LenB(inputRngData(1, 2)) And inputRngData(1, 2) <> "無" Then
            errMsg = errMsg & errAtartMsg & "[日本語有無]は指定できません" & vbCrLf
        End If
        '[桁数]未入力/数値以外/範囲外
        If LenB(inputRngData(1, 3)) = 0 Then
            errMsg = errMsg & errAtartMsg & "[桁数]を入力して下さい" & vbCrLf
        ElseIf Not IsNumeric(inputRngData(1, 3)) Then
            errMsg = errMsg & "[桁数]は数値を入力して下さい" & vbCrLf
        ElseIf dbNm = ORACLE And 9 < CLng(inputRngData(1, 3)) Then
            errMsg = errMsg & ORACLE & "の場合[桁数]は0～9で入力して下さい" & vbCrLf
        ElseIf dbNm = POSTGRESQL And 6 < CLng(inputRngData(1, 3)) Then
            errMsg = errMsg & POSTGRESQL & "の場合[桁数]は0～6で入力して下さい" & vbCrLf
        ElseIf dbNm = SQLSERVER And 7 < CLng(inputRngData(1, 3)) Then
            errMsg = errMsg & SQLSERVER & "の場合[桁数]は0～7で入力して下さい" & vbCrLf
        End If
        '[小数桁数]入力有
        If 0 < LenB(inputRngData(1, 4)) And inputRngData(1, 4) <> "0" Then
            errMsg = errMsg & errAtartMsg & "[小数桁数]は指定できません" & vbCrLf
        End If
        '[時間有無]未入力
        If LenB(inputRngData(1, 5)) = 0 Then
            errMsg = errMsg & errAtartMsg & "[時間有無]を指定して下さい" & vbCrLf
        End If
        '[時差修正要否]未入力
        If LenB(inputRngData(1, 6)) = 0 Then
            errMsg = errMsg & errAtartMsg & "[時差修正要否]を指定して下さい" & vbCrLf
        End If
    End If

    '***** 真偽値 / バイナリ / JSON / XML / UUID *****
    If dataType = "真偽値" Or dataType = "バイナリ" Or dataType = "JSON" Or dataType = "XML" Or dataType = "UUID" Then
        '[日本語有無]入力有
        If 0 < LenB(inputRngData(1, 2)) And inputRngData(1, 2) <> "無" Then
            errMsg = errMsg & errAtartMsg & "[日本語有無]は指定できません" & vbCrLf
        End If
        '[桁数]未入力 or 数値以外 or 0
        If 0 < LenB(inputRngData(1, 3)) And inputRngData(1, 3) <> "0" Then
            errMsg = errMsg & errAtartMsg & "[桁数]は指定できません" & vbCrLf
        End If
        '[小数桁数]入力有
        If 0 < LenB(inputRngData(1, 4)) And inputRngData(1, 4) <> "0" Then
            errMsg = errMsg & errAtartMsg & "[小数桁数]は指定できません" & vbCrLf
        End If
        '[時間有無]入力有
        If 0 < LenB(inputRngData(1, 5)) And inputRngData(1, 5) <> "無" Then
            errMsg = errMsg & errAtartMsg & "[時間有無]は指定できません" & vbCrLf
        End If
        '[時差修正要否]入力有
        If 0 < LenB(inputRngData(1, 6)) And inputRngData(1, 6) <> "否" Then
            errMsg = errMsg & errAtartMsg & "[時差修正要否]は指定できません" & vbCrLf
        End If
    End If

    '末尾除去
    If 0 < LenB(errMsg) Then errMsg = Left$(errMsg, Len(errMsg) - Len(vbCrLf))
    Valid_SetDataType = errMsg
End Function
'------------------------------------------------------------
'概要 : 型候補リストを取得する
'引数 : dbNm         : DB名(Oracle / PostgreSQL / SQLServer)
'     : inputRngData : 入力値範囲データ(データ分類/日本語有無/桁数/小数桁数/時間有無/時差修正要否)
'戻値 : 型候補リスト
'備考 : -
'------------------------------------------------------------
Private Function GetColDataTypes(ByVal dbNm As String, ByVal inputRngData As Variant) As Variant
    '●データ分類/日本語有無/桁数/小数桁数/時間有無/時差修正要否
    Dim dataType As String: dataType = CStr(inputRngData(1, 1))
    Dim jpFlg As String: jpFlg = CStr(inputRngData(1, 2))
    Dim maxLen As String: maxLen = CStr(inputRngData(1, 3))
    Dim dcmlLen As String: dcmlLen = CStr(inputRngData(1, 4))
    Dim timeFlg As String: timeFlg = CStr(inputRngData(1, 5))
    Dim timeZoneFlg As String: timeZoneFlg = CStr(inputRngData(1, 6))
    '判定
    GetColDataTypes = Array("", "", "")
    Select Case dbNm
        Case ORACLE
            Select Case dataType
                Case "文字列"
                    If 4000 <= CLng(maxLen) Then
                        If jpFlg = "有" Then
                            GetColDataTypes = Array("CLOB", "NCLOB", "VARCHAR2(" & maxLen & ")")
                        Else
                            GetColDataTypes = Array("CLOB", "VARCHAR2(" & maxLen & ")", "")
                        End If
                    ElseIf jpFlg = "有" Then
                        GetColDataTypes = Array("NVARCHAR2(" & maxLen & ")", "VARCHAR2(" & maxLen & ")", "")
                    Else
                        GetColDataTypes = Array("VARCHAR2(" & maxLen & ")", "NVARCHAR2(" & maxLen & ")", "")
                    End If
                Case "数値"
                    GetColDataTypes = Array("NUMBER(" & (CLng(maxLen) + CLng(dcmlLen)) & "," & dcmlLen & ")", "BINARY_DOUBLE", "BINARY_FLOAT")
                Case "日時"
                    If timeZoneFlg = "要" Then
                        GetColDataTypes = Array("TIMESTAMP WITH TIME ZONE(" & maxLen & ")", "TIMESTAMP(" & maxLen & ")", "")
                    ElseIf timeFlg = "有" Then
                        GetColDataTypes = Array("TIMESTAMP(" & maxLen & ")", "DATE", "")
                    Else
                        GetColDataTypes = Array("DATE", "TIMESTAMP(" & maxLen & ")", "")
                    End If
                Case "真偽値"
                    GetColDataTypes = Array("NUMBER(1)", "CHAR(1)", "")
                Case "バイナリ"
                    GetColDataTypes = Array("BLOB", "RAW", "")
                Case "JSON"
                    GetColDataTypes = Array("JSON", "CLOB", "")
                Case "XML"
                    GetColDataTypes = Array("XMLTYPE", "CLOB", "")
                Case "UUID"
                    GetColDataTypes = Array("RAW(16)", "VARCHAR2(36)", "")
            End Select
        Case POSTGRESQL
            Select Case dataType
                Case "文字列"
                    If 4000 <= CLng(maxLen) Then
                        GetColDataTypes = Array("text", "varchar(" & maxLen & ")", "char(" & maxLen & ")")
                    Else
                        GetColDataTypes = Array("varchar(" & maxLen & ")", "text", "char(" & maxLen & ")")
                    End If
                Case "数値"
                    GetColDataTypes = Array("numeric(" & (CLng(maxLen) + CLng(dcmlLen)) & "," & dcmlLen & ")", "double precision", "real")
                Case "日時"
                    If timeZoneFlg = "要" Then
                        GetColDataTypes = Array("timestamp with time zone(" & maxLen & ")", "timestamp(" & maxLen & ")", "")
                    ElseIf timeFlg = "有" Then
                        GetColDataTypes = Array("timestamp(" & maxLen & ")", "date", "")
                    Else
                        GetColDataTypes = Array("date", "timestamp(" & maxLen & ")", "")
                    End If
                Case "真偽値"
                    GetColDataTypes = Array("boolean", "smallint", "")
                Case "バイナリ"
                    GetColDataTypes = Array("bytea", "", "")
                Case "JSON"
                    GetColDataTypes = Array("jsonb", "json", "")
                Case "XML"
                    GetColDataTypes = Array("xml", "", "")
                Case "UUID"
                    GetColDataTypes = Array("uuid", "varchar(36)", "")
            End Select
        Case SQLSERVER
            Select Case dataType
                Case "文字列"
                    If 4000 <= CLng(maxLen) Then
                        If jpFlg = "有" Then
                            GetColDataTypes = Array("nvarchar(max)", "varchar(max)", "")
                        Else
                            GetColDataTypes = Array("varchar(max)", "nvarchar(max)", "")
                        End If
                    Else
                        If jpFlg = "有" Then
                            GetColDataTypes = Array("nvarchar(" & maxLen & ")", "varchar(" & maxLen & ")", "")
                        Else
                            GetColDataTypes = Array("varchar(" & maxLen & ")", "nvarchar(" & maxLen & ")", "")
                        End If
                    End If
                Case "数値"
                    GetColDataTypes = Array("numeric(" & (CLng(maxLen) + CLng(dcmlLen)) & "," & dcmlLen & ")", "float", "real")
                Case "日時"
                    If timeZoneFlg = "要" Then
                        GetColDataTypes = Array("datetimeoffset(" & maxLen & ")", "datetime2(" & maxLen & ")", "")
                    ElseIf timeFlg = "有" Then
                        GetColDataTypes = Array("datetime2(" & maxLen & ")", "date", "")
                    Else
                        GetColDataTypes = Array("date", "datetime2(" & maxLen & ")", "")
                    End If
                Case "真偽値"
                    GetColDataTypes = Array("bit", "tinyint", "")
                Case "バイナリ"
                    GetColDataTypes = Array("varbinary", "binary", "")
                Case "JSON"
                    GetColDataTypes = Array("nvarchar(max)", "varchar(max)", "")
                Case "XML"
                    GetColDataTypes = Array("xml", "nvarchar(max)", "")
                Case "UUID"
                    GetColDataTypes = Array("uniqueidentifier", "nvarchar(36)", "")
            End Select
    End Select
End Function

    Dim setCrntDateDict As Object: Set setCrntDateDict = GetDbSetCrntDateDict

    '●DB名/スキーマ名/テーブル名(論理/物理)
    Dim dbNm As String: dbNm = ws.Range("B1").Value2
    Dim schemaNm As String: schemaNm = Trim$(ws.Range("B2").Value2)
    Dim logiTblNm As String: logiTblNm = Trim$(ws.Range("B3").Value2)
    Dim physiTblNm As String: physiTblNm = Trim$(ws.Range("B4").Value2)

    '●項目定義データ
    Dim colInfo As Variant
    '●PK/項目名(論理/物理)/型/桁/NULL不可/初期値/正数部/小数部
    Dim pk As String
    Dim logiColNm As String
    Dim physiColNm As String
    Dim colType As String
    Dim colLen As Variant
    Dim notNull As String
    Dim initVal As String
    Dim intPart As Long
    Dim dcmlPart As Long

    '●内容/PK行/行
    Dim contents As String
    Dim contents2 As String
    Dim pkLine As String
    Dim line1 As String
    Dim line2 As String
    '●TBLオプション
    Dim tblOp As String
    '●追加フラグ
    Dim addFlg As Boolean: addFlg = False

    '●メッセージ
    Dim msg As String
    '●日付スタンプ
    Dim datestamp As String: datestamp = Format(Now, "yyyymmdd")
    '●ファイルObject
    Dim fso As Object: Set fso = CreateObject("Scripting.FileSystemObject")
    '●作成ファイルPath/ファイル名/作成ファイル
    Dim cratFilePath As String: cratFilePath = ThisWorkbook.path & "\"
    Dim cratFileNm As String: cratFileNm = "DDL-" & ws.Name
    Dim cratFile As Object

    '終了行 セット
    endRowIdx = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    '項目定義 取得
    colInfo = ws.Range(ws.Range("B7"), ws.Cells(endRowIdx, "H")).Value2

    '========== ▼入力チェック(TBL定義) ==========
    '[DB]未選択
    If LenB(dbNm) = 0 Then
        Call MsgBox("(B1) [DB] を選択して下さい", vbCritical, "ERROR")
        Exit Sub
    '[SCHEMA]未入力
    ElseIf LenB(schemaNm) = 0 Then
        Call MsgBox("(B2) [SCHEMA] を入力して下さい", vbCritical, "ERROR")
        Exit Sub
    '[TBL物理名]未入力 / 不正物理名
    ElseIf LenB(physiTblNm) = 0 Then
        Call MsgBox("(B4) [TBL(物理名)] を入力して下さい", vbCritical, "ERROR")
        Exit Sub
    ElseIf IsErrPhysicalNm(dbNm, physiTblNm) Then
        Call MsgBox("TBL(物理名) [" & physiTblNm & "]" & vbCrLf & "半角英数字 & 半角アンダーバーで入力して下さい" & vbCrLf & "(先頭数字もNG)", vbCritical, "ERROR")
        Exit Sub
    'データなし
    ElseIf endRowIdx < 7 Then
        Call MsgBox("出力データがありません", vbCritical, "ERROR")
        Exit Sub
    End If

    '========== ▼入力チェック(項目定義) ==========
    '全行Loop
    For i = 1 To UBound(colInfo, 1)
        msg = i & "行目："
        '項目定義 セット(COL物理名/型/桁/NULL不可/初期値)
        pk = Trim$(colInfo(i, 1))
        logiColNm = Trim$(colInfo(i, 2))
        physiColNm = Trim$(colInfo(i, 3))
        colType = Trim$(colInfo(i, 4))
        colLen = Trim$(colInfo(i, 5))
        notNull = Trim$(colInfo(i, 6))
        initVal = Trim$(colInfo(i, 7))
        '型末尾のかっこ 除去
        If 0 < InStr(colType, "(") And 0 < InStr(colType, ")") Then
            bgnIdx = InStr(colType, "(") - 1
            endIdx = InStr(colType, ")") + 1
            colType = Trim$(Left$(colType, bgnIdx) & Mid$(colType, endIdx))
        End If
        '桁のかっこ 除去
        If 0 < InStr(colLen, "(") And 0 < InStr(colLen, ")") Then
            colLen = Replace(colLen, "(", "")
            colLen = Replace(colLen, ")", "")
        End If
        '未入力の場合(COL物理名/型)
        If LenB(physiColNm) = 0 Then
            Call MsgBox(msg & "COL物理名" & vbCrLf & "半角英数字 & 半角アンダーバーで入力して下さい" & vbCrLf & "(先頭数字もNG)", vbCritical, "ERROR")
            Exit Sub
        ElseIf LenB(colType) = 0 Then
            Call MsgBox(msg & "型" & "を入力して下さい", vbCritical, "ERROR")
            Exit Sub
        '不正な項目名の場合(COL物理名)
        ElseIf IsErrPhysicalNm(dbNm, physiColNm) Then
            Call MsgBox(msg & physiColNm & vbCrLf & "半角英数字 & 半角アンダーバーで入力して下さい" & vbCrLf & "(先頭数字もNG)", vbCritical, "ERROR")
            Exit Sub
        '予約語の場合(COL物理名)
        ElseIf dbNm = ORACLE And dbReservedWordDict_Ora.exists(UCase$(physiColNm)) Then
            Call MsgBox(msg & physiColNm & vbCrLf & "この項目名は予約語の為、登録できません", vbCritical, "ERROR")
            Exit Sub
        ElseIf dbNm = POSTGRESQL And dbReservedWordDict_Pos.exists(UCase$(physiColNm)) Then
            Call MsgBox(msg & physiColNm & vbCrLf & "この項目名は予約語の為、登録できません", vbCritical, "ERROR")
            Exit Sub
        ElseIf dbNm = SQLSERVER And dbReservedWordDict_Sql.exists(UCase$(physiColNm)) Then
            Call MsgBox(msg & physiColNm & vbCrLf & "この項目名は予約語の為、登録できません", vbCritical, "ERROR")
            Exit Sub
        End If
        '[文字列]型の場合
        If colTypeDict_Str.exists(UCase$(colType)) Then
            '未入力の場合
            If LenB(colLen) = 0 Then
                Call MsgBox(msg & "文字列型の場合、桁を1以上で入力して下さい", vbCritical, "ERROR")
                Exit Sub
            '小数桁有りの場合
            ElseIf 0 < InStr(colLen, ",") Or 0 < InStr(colLen, ".") Then
                Call MsgBox(msg & "文字列型の場合、桁に小数は登録できません", vbCritical, "ERROR")
                Exit Sub
            '数値以外の場合
            ElseIf Not IsNumeric(colLen) Then
                Call MsgBox(msg & "桁は数値で入力して下さい", vbCritical, "ERROR")
                Exit Sub
            '0の場合
            ElseIf CLng(colLen) = 0 Then
                Call MsgBox(msg & "文字列型の場合、桁に0以上を入力して下さい", vbCritical, "ERROR")
                Exit Sub
            'NULL不可 & 初期値=NULL の場合
            ElseIf notNull = "Y" And initVal = "NULL" Then
                Call MsgBox(msg & "NULL不可の場合、初期値にNULLを登録することはできません", vbCritical, "ERROR")
                Exit Sub
            End If
        '[数値]型の場合
        ElseIf colTypeDict_Num.exists(UCase$(colType)) Then
            '未入力の場合
            If LenB(colLen) = 0 Then
                Call MsgBox(msg & "数値型の場合、桁を入力して下さい", vbCritical, "ERROR")
                Exit Sub
            '小数点有りの場合
            ElseIf 0 < InStr(colLen, ".") Then
                Call MsgBox(msg & "桁に小数点は登録できません" & vbCrLf & "例：(10,0)", vbCritical, "ERROR")
                Exit Sub
            '小数桁無しの場合
            ElseIf InStr(colLen, ",") = 0 Then
                Call MsgBox(msg & "数値型の場合、桁に小数桁を0以上で入力して下さい" & vbCrLf & "例：(10,0)", vbCritical, "ERROR")
                Exit Sub
            End If
            '正数桁/小数桁 セット
            intPart = CLng(Split(colLen, ",")(0))
            dcmlPart = CLng(Split(colLen, ",")(1))
            '範囲外の場合
            If 15 < intPart Then
                Call MsgBox(msg & "数値型の場合、正数桁は15以下で入力して下さい", vbCritical, "ERROR")
                Exit Sub
            ElseIf 5 < dcmlPart Then
                Call MsgBox(msg & "数値型の場合、小数桁は5以下で入力して下さい", vbCritical, "ERROR")
                Exit Sub
            '逆転の場合
            ElseIf intPart <= dcmlPart Then
                Call MsgBox(msg & "正数桁は小数桁より大きい数値を入力して下さい", vbCritical, "ERROR")
                Exit Sub
            'NULL不可 & 初期値=NULL の場合
            ElseIf notNull = "Y" And initVal = "NULL" Then
                Call MsgBox(msg & "NULL不可の場合、初期値にNULLを登録することはできません", vbCritical, "ERROR")
                Exit Sub
            '数値以外の場合
            ElseIf notNull <> "Y" And Not IsNumeric(initVal) Then
                Call MsgBox(msg & "数値型の場合、初期値に数値以外は登録できません", vbCritical, "ERROR")
                Exit Sub
            End If
        '[日付]型の場合
        ElseIf colTypeDict_Date.exists(UCase$(colType)) Then
            '時間無しの型の場合
            If colType = "DATE" Then
                '入力有の場合
                If 0 < LenB(colLen) Then
                    If Not IsNumeric(colLen) Then
                        Call MsgBox(msg & "日付型(DATE)の場合、桁は登録できません", vbCritical, "ERROR")
                        Exit Sub
                    ElseIf CLng(colLen) <> 0 Then
                        Call MsgBox(msg & "日付型(DATE)の場合、桁は登録できません", vbCritical, "ERROR")
                        Exit Sub
                    End If
                End If
                '時間有りの現在日時セットの場合(SQLServer用)
                If dbNm = SQLSERVER And UCase$(initVal) <> "GETDATE()" And UCase$(initVal) <> "CURRENT_DATE" And setCrntDateDict.exists(UCase$(initVal)) Then
                    Call MsgBox(msg & "SQLServerのDATE型の場合、初期値に時間付きの現在日時記述は登録できません", vbCritical, "ERROR")
                    Exit Sub
                End If
            Else
                '未入力の場合
                If LenB(colLen) = 0 Then
                    Call MsgBox(msg & "日付型の場合、桁にミリ秒桁を0以上で入力して下さい", vbCritical, "ERROR")
                    Exit Sub
                '小数桁有りの場合
                ElseIf 0 < InStr(colLen, ",") Or 0 < InStr(colLen, ".") Then
                    Call MsgBox(msg & "日付型の場合、桁に小数は登録できません", vbCritical, "ERROR")
                    Exit Sub
                '数値以外の場合
                ElseIf Not IsNumeric(colLen) Then
                    Call MsgBox(msg & "桁は数値で入力して下さい", vbCritical, "ERROR")
                    Exit Sub
                '範囲外の場合
                ElseIf dbNm = ORACLE And 9 < CLng(colLen) Then
                    Call MsgBox(msg & "日付型の場合、桁は9以下で入力して下さい", vbCritical, "ERROR")
                    Exit Sub
                ElseIf dbNm = POSTGRESQL And 6 < CLng(colLen) Then
                    Call MsgBox(msg & "日付型の場合、桁は6以下で入力して下さい", vbCritical, "ERROR")
                    Exit Sub
                ElseIf dbNm = SQLSERVER And 7 < CLng(colLen) Then
                    Call MsgBox(msg & "日付型の場合、桁は7以下で入力して下さい", vbCritical, "ERROR")
                    Exit Sub
                End If
            End If
            'NULL不可 & 初期値=NULL の場合
            If notNull = "Y" And UCase$(initVal) = "NULL" Then
                Call MsgBox(msg & "NULL不可の場合、初期値にNULLを登録することはできません", vbCritical, "ERROR")
                Exit Sub
            End If
        End If
    Next

    '========== DDL内容 作成 ==========
    pkLine = "    PRIMARY KEY("
    contents = ""
    contents = contents & "-- ■■■■■■■■■■■■■■■■■■■■■■■■■■■■■■" & vbCrLf
    contents = contents & "--   TBL削除(存在する場合) [" & logiTblNm & " / " & physiTblNm & "]" & vbCrLf
    contents = contents & "-- ■■■■■■■■■■■■■■■■■■■■■■■■■■■■■■" & vbCrLf
    contents = contents & "DROP TABLE IF EXISTS " & schemaNm & "." & physiTblNm & ";" & vbCrLf
    If dbNm = SQLSERVER Then
        contents = contents & "-- ===== OLD ver =====" & vbCrLf
        contents = contents & "-- IF OBJECT_ID('" & schemaNm & "." & physiTblNm & "', 'U') IS NOT NULL DROP TABLE " & schemaNm & "." & physiTblNm & ";" & vbCrLf
    End If
    contents = contents & vbCrLf
    contents = contents & "-- ■■■■■■■■■■■■■■■■■■■■■■■■■■■■■■" & vbCrLf
    contents = contents & "--   TBL作成 [" & logiTblNm & " / " & physiTblNm & "]" & vbCrLf
    contents = contents & "-- ■■■■■■■■■■■■■■■■■■■■■■■■■■■■■■" & vbCrLf

    '=== 見出し ==========
    line1 = "--[項目名(論理)] "
    line2 = "--[項目名(物理)] "
    For i = 1 To UBound(colInfo, 1)
        'COL名(論理/物理)
        logiColNm = Trim$(colInfo(i, 2))
        physiColNm = Trim$(colInfo(i, 3))
        '桁数調整
        Call AppendCharaToString(logiColNm, physiColNm, " ")
        line1 = line1 & i & ":" & logiColNm & " / "
        line2 = line2 & i & ":" & physiColNm & " / "
    Next
    '末尾除去
    line1 = Left$(line1, Len(line1) - 3)
    line2 = Left$(line2, Len(line2) - 3)
    '追加
    contents = contents & line1 & vbCrLf
    contents = contents & line2 & vbCrLf

    '=== 内容 ==========
    contents = contents & "CREATE TABLE " & schemaNm & "." & physiTblNm & "(" & vbCrLf
    For i = 1 To UBound(colInfo, 1)
        '項目定義 セット(COL名(論理/物理)/型/桁/NULL不可/初期値)
        pk = Trim$(colInfo(i, 1))
        logiColNm = Trim$(colInfo(i, 2))
        physiColNm = Trim$(colInfo(i, 3))
        colType = Trim$(colInfo(i, 4))
        colLen = Trim$(colInfo(i, 5))
        notNull = Trim$(colInfo(i, 6))
        initVal = Trim$(colInfo(i, 7))
        '型末尾のかっこ 除去
        If 0 < InStr(colType, "(") And 0 < InStr(colType, ")") Then
            bgnIdx = InStr(colType, "(") - 1
            endIdx = InStr(colType, ")") + 1
            colType = Trim$(Left$(colType, bgnIdx) & Mid$(colType, endIdx))
        End If
        '桁のかっこ 付与
        If 0 < LenB(colLen) Then
            If Left$(colLen, 1) <> "(" Then
                colLen = "(" & colLen
            End If
            If Right$(colLen, 1) <> ")" Then
                colLen = colLen & ")"
            End If
        End If
        '初期値 整形(シングルクォート付与)
        If 0 < LenB(initVal) Then
            '[文字列]型 or [日付]型の場合
            If (colTypeDict_Str.exists(UCase$(colType)) And UCase$(initVal) <> "NULL") Or (colTypeDict_Date.exists(UCase$(colType)) And Not setCrntDateDict.exists(UCase$(initVal)) And UCase$(initVal) <> "NULL") Then
                If Left$(initVal, 1) <> "'" Then
                    initVal = "'" & initVal
                End If
                If Right$(initVal, 1) <> "'" Then
                    initVal = initVal & "'"
                End If
            End If
            initVal = " DEFAULT " & initVal
        End If
        'NULL不可の場合
        If notNull = "Y" Then notNull = " NOT NULL"
        'PKの場合
        If pk = "Y" Then pkLine = pkLine & physiColNm & ", "
        '内容追加
        contents = contents & "    " & physiColNm & " " & colType & colLen & initVal & notNull & "," & vbCrLf
    Next
    '末尾除去 & 付与
    If 16 < Len(pkLine) Then
        pkLine = Left$(pkLine, Len(pkLine) - 2) & ")" & vbCrLf
        contents = contents & vbCrLf & pkLine
    Else
        contents = Left$(contents, Len(contents) - (Len(vbCrLf) + 1)) & vbCrLf
    End If
    contents = contents & ");" & vbCrLf

    '=== TBLオプション ==========
    tblOp = ws.Range("F2").Value2
    If 0 < LenB(tblOp) Then
        contents = contents & "-- ===== TBLオプション =====" & vbCrLf
        contents = contents & tblOp & vbCrLf
    End If

    '=== 論理名 ==========
    contents2 = "" & vbCrLf
    contents2 = contents2 & "-- ==================================================" & vbCrLf
    contents2 = contents2 & "--  論理名 追加 [" & logiTblNm & " / " & physiTblNm & "]" & vbCrLf
    contents2 = contents2 & "-- ==================================================" & vbCrLf
    If 0 < LenB(logiTblNm) Then
        If dbNm = ORACLE Or dbNm = POSTGRESQL Then
            contents2 = contents2 & "COMMENT ON TABLE " & schemaNm & "." & physiTblNm & " IS '" & logiTblNm & "';" & vbCrLf
        ElseIf dbNm = SQLSERVER Then
            contents2 = contents2 & "EXEC sp_addextendedproperty @name=N'MS_Description', @level0type=N'SCHEMA', @level0name=N'" & schemaNm & "', @level1type=N'TABLE', @level1name=N'" & physiTblNm & "', @value=N'" & logiTblNm & "';" & vbCrLf
        End If
        addFlg = True
    End If
    For i = 1 To UBound(colInfo, 1)
        'COL名(論理/物理)
        logiColNm = Trim$(colInfo(i, 2))
        physiColNm = Trim$(colInfo(i, 3))
        If 0 < LenB(logiColNm) Then
            If dbNm = ORACLE Or dbNm = POSTGRESQL Then
                contents2 = contents2 & "COMMENT ON COLUMN " & schemaNm & "." & physiTblNm & "." & physiColNm & " IS '" & logiColNm & "';" & vbCrLf
            ElseIf dbNm = SQLSERVER Then
                contents2 = contents2 & "EXEC sp_addextendedproperty @name=N'MS_Description', @level0type=N'SCHEMA', @level0name=N'" & schemaNm & "', @level1type=N'TABLE', @level1name=N'" & physiTblNm & "', @level2type=N'COLUMN', @level2name=N'" & physiColNm & "',   @value=N'" & logiColNm & "';" & vbCrLf
            End If
        addFlg = True
        End If
    Next
    '追加
    If addFlg Then contents = contents & contents2

    '========== ファイル作成先フォルダ作成(存在しない場合) ==========
    If Not IsExistsFolder(cratFilePath, datestamp) Then
        If Not CreateFolderPath(cratFilePath & datestamp) Then
            msg = "フォルダの作成に失敗しました" & vbCrLf & _
                  "PATH : " & cratFilePath & datestamp
            Call MsgBox(msg, vbCritical, "")
            Exit Sub
        End If
    End If

    '========== ファイル名 作成 & ファイル作成 ==========
    cratFileNm = ChangeUniqueFileNm(cratFilePath & datestamp, IIf(cratFileNm <> "", cratFileNm, ws.Name), "sql")
    Call CreateTxtFile(cratFilePath & datestamp, cratFileNm, "sql", contents, "UTF-8", False)

    '完了メッセージ
    msg = "DDLファイルを作成しました！！" & vbCrLf & _
          "PATH : " & cratFilePath & datestamp & vbCrLf & _
          "FILE名 : " & cratFileNm
    Call MsgBox(msg, vbInformation, "")

    '処理中描画ON設定
    Call SettingRenderingProcess(True)
End Sub
'------------------------------------------------------------
'概要 : テキストファイルを作成する(UTF-8)
'引数 : filePath  : ファイルパス
'     : fileNm    : ファイル名
'     : fileExt   : ファイル拡張子
'     : contents  : 書込内容
'     : charsetNm : 文字コード名(UTF-8 / Shift-JIS など)
'     : bomFlg    : BOM有無フラグ(True=有 / False=無)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub CreateTxtFile(ByVal filePath As String, ByVal fileNm As String, ByVal fileExt As String, ByVal contents As String, ByVal charsetNm As String, ByVal bomFlg As Boolean)
    '●ファイル書込用
    Dim stream As Object: Set stream = CreateObject("ADODB.Stream")
    Dim binStream As Object
    Dim byteData() As Byte
    Dim bomLen As Integer: bomLen = GetBomLen(charsetNm)

    '末尾付与
    If Right$(filePath, 1) <> "\" Then filePath = filePath & "\"
    '先頭付与
    If Left$(fileExt, 1) <> "." Then fileExt = "." & fileExt

    With stream
        .Type = 2
        .Charset = charsetNm
        .Open
        .WriteText contents
        .SaveToFile filePath & fileNm, 2 '既存ファイル上書き設定
        .Close
    End With
    Set stream = Nothing

    'BOMなしの場合、先頭nバイト(BOM)除去で作り直す
    If Not bomFlg And 0 < bomLen Then
        '===== 読込 =====
        Set binStream = CreateObject("ADODB.Stream")
        With binStream
            .Type = 1
            .Open
            .LoadFromFile filePath & fileNm
            .Position = bomLen
            byteData = .Read
            .Close
        End With
        '===== 書込 =====
        With binStream
            .Type = 1
            .Open
            .Write byteData
            .SaveToFile filePath & fileNm, 2
            .Close
        End With
        Set binStream = Nothing
    End If
End Sub
'------------------------------------------------------------
'概要 : 文字コード名のBOMバイト数を取得する
'引数 : charsetNm : 文字コード名(UTF-8 / Shift-JIS など)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Function GetBomLen(ByVal charsetNm As String) As Integer
    Select Case UCase$(charsetNm)
        Case "UTF-8"
            GetBomLen = 3
        Case "UTF-16", "UNICODE", "UTF-16LE", "UTF-16BE"
            GetBomLen = 2
        Case "UTF-32"
            GetBomLen = 4
        Case Else
            GetBomLen = 0
    End Select
End Function
'------------------------------------------------------------
'概要 : 物理名の命名エラーチェック
'引数 : dbNm    : DB名(Oracle / PostgreSQL / SQLServer)
'     : physiNm : 物理名
'戻値 : エラーメッセージ
'備考 : -
'------------------------------------------------------------
Private Function IsErrPhysicalNm(ByVal dbNm As String, ByVal physiNm As String) As Boolean
    Dim i As Long
    Dim c As String
    Dim code As Long
    '空の場合
    If LenB(physiNm) = 0 Then
        IsErrPhysicalNm = True
        Exit Function
    End If
    '先頭数字の場合
    c = Mid$(physiNm, 1, 1)
    code = AscW(c)
    '(0-9)(_)
    If (48 <= code And code <= 57) Or (code = 95) Then
        IsErrPhysicalNm = True
        Exit Function
    End If
    '[半角英数字 / 半角アンダーバー] 以外の場合
    For i = 2 To Len(physiNm)
        c = Mid$(physiNm, i, 1)
        code = AscW(c)
        '(A-Z)(a-z)(0-9)(_)
        If Not ((65 <= code And code <= 90) Or (97 <= code And code <= 122) Or (48 <= code And code <= 57) Or (code = 95)) Then
            IsErrPhysicalNm = True
            Exit Function
        End If
    Next
    IsErrPhysicalNm = False
End Function
'------------------------------------------------------------
'概要 : 2つの文字列の桁数を比較し、大きい方に合わせて、小さい方の文字列の末尾に半角文字を追記し、桁数を合わせる
'引数 : str1  : 文字列1
'     : str2  : 文字列2
'     : addCh : 追記文字(半角)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub AppendCharaToString(ByRef str1 As String, ByRef str2 As String, ByVal addCh As String)
    Dim len1 As Long
    Dim len2 As Long
    Dim diffLen As Long
    '桁数セット(半角=1/全角=2)
    len1 = LenB(StrConv(str1, vbFromUnicode))
    len2 = LenB(StrConv(str2, vbFromUnicode))
    '桁数を比較して、小さい方の末尾に半角スペース（Space関数）を補充
    If len2 < len1 Then
        diffLen = len1 - len2
        str2 = str2 & String(diffLen, addCh)
    ElseIf len1 < len2 Then
        diffLen = len2 - len1
        str1 = str1 & String(diffLen, addCh)
    End If
End Sub
