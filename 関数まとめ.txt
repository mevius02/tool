'------------------------------------------------------------
'概要 : 処理中の画面描画を設定する
'引数 : flg(True:ON / False:OFF)
'戻値 : -
'備考 : -
'------------------------------------------------------------
Private Sub SettingRenderingProcess(ByVal flg As Boolean)
    Application.ScreenUpdating = flg '画面再描画
    Application.DisplayAlerts = flg  '確認ダイアログ表示
    Application.EnableEvents = flg   'イベント発火
    '再計算モード(セル値が変わるたびに再計算)
    If flg Then
        Application.Calculation = xlCalculationAutomatic '自動
    Else
        Application.Calculation = xlCalculationManual '手動
    End If
End Sub
'------------------------------------------------------------
'概要 : Sheet名のSheetが存在するか判定する
'引数 : wsNm : Sheet名
'戻値 : 判定結果(True:する / False:しない)
'備考 : -
'------------------------------------------------------------
Private Function IsExistsSheet(ByVal wsNm As String) As Boolean
    Dim ws As Worksheet
    For Each ws In Sheets
        If ws.Name = wsNm Then
            IsExistsSheet = True
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : Sheet名が重複しないよう、末尾に連番を付与する
'     : (同じ名前のSheetが存在しない場合、そのまま返す)
'     : (同じ名前のSheetが存在する場合、2からの連番を末尾に付与し、重複しないSheet名で返す)
'     : "AAA" → "AAA (2)" → "AAA (3)"
'引数 : wsNm : Sheet名
'戻値 : Sheet名
'備考 : -
'------------------------------------------------------------
Private Function ChangeToUniqueSheetNm(ByVal wsNm As String) As String
    Dim i As Long
    Dim uniqueNo As String
    '存在しない場合
    If Not IsExistsSheet(wsNm) Then
        ChangeToUniqueSheetNm = wsNm
        Exit Function
    End If
    '存在する場合
    For i = 2 To 1000
        '末尾付与
        uniqueNo = " (" & i & ")"
        '存在しない場合
        If Not IsExistsSheet(wsNm & uniqueNo) Then
            ChangeToUniqueSheetNm = wsNm & uniqueNo
            Exit Function
        End If
    Next
End Function
'------------------------------------------------------------
'概要 : 列番号から列名(アルファベット)を取得する
'     : (範囲外の数値の場合、空を返す)
'引数 : colIdx : 列番号
'戻値 : 列名(アルファベット)
'備考 : -
'------------------------------------------------------------
Private Function GetColNm(ByVal colIdx As Long) As String
    Dim result As String
    Dim n As Long
    '範囲外の場合
    If colIdx < 1 Or Application.Columns.Count < colIdx Then
        GetColNm = ""
        Exit Function
    End If
    n = colIdx
    Do While 0 < n
        n = n - 1
        result = Chr$(65 + (n Mod 26)) & result
        n = n \ 26
    Loop
    GetColNm = result
End Function
'------------------------------------------------------------
'概要 : 列名(アルファベット)から列番号を取得する
'     : (範囲外の列名の場合、0を返す)
'引数 : colNm : 列名(アルファベット)
'戻値 : 列番号
'備考 : -
'------------------------------------------------------------
Private Function GetColIdx(ByVal colNm As String) As Long
    Dim i As Long
    Dim result As Long
    Dim baseA As Long
    Dim length As Long
    Dim ch As Long
    baseA = Asc("A") - 1
    colNm = Trim$(UCase$(colNm))
    length = Len(colNm)
    For i = 1 To length
        ch = Asc(Mid$(colNm, i, 1)) - baseA
        '範囲外の場合
        If ch < 1 Or 26 < ch Then
            GetColIdx = 0
            Exit Function
        End If
        result = result * 26 + ch
    Next
    GetColIdx = result
End Function
'------------------------------------------------------------
'概要 : 色コードからRGB値に変換する
'引数 : colorCd : 色コード
'戻値 : RGB値
'備考 : -
'------------------------------------------------------------
Function ConvertHexToRGB(colorCd As String) As Long
    Dim hex As String
    '"#"除去
    hex = Replace(colorCd, "#", "")
    ConvertHexToRGB = RGB(CLng("&H" & Mid(hex, 1, 2)), CLng("&H" & Mid(hex, 3, 2)), CLng("&H" & Mid(hex, 5, 2)))
End Function
