vbOption Explicit

'==================================================
' Sheet3: ファイル割当一覧（JCL ↔ COB 紐付き）
'
' 出力列: No / JCLファイル名 / ジョブ名 / ステップ名 /
'         実行PGM / DDname / DSN / DISP / 対応FD名 / R/W区分
'
' ★ 紐付けできなかったDD → FD名列をオレンジハイライト
'==================================================
Public Sub CreateSheet3_FileAllocation()

    Dim ws     As Worksheet
    Dim rowIdx As Long

    Call InitSheet("Sheet3_ファイル割当一覧", ws)

    ws.Cells(1, 1).Value  = "No"
    ws.Cells(1, 2).Value  = "JCLファイル名"
    ws.Cells(1, 3).Value  = "ジョブ名"
    ws.Cells(1, 4).Value  = "ステップ名"
    ws.Cells(1, 5).Value  = "実行PGM"
    ws.Cells(1, 6).Value  = "DDname"
    ws.Cells(1, 7).Value  = "データセット名(DSN)"
    ws.Cells(1, 8).Value  = "DISP"
    ws.Cells(1, 9).Value  = "対応FD名"
    ws.Cells(1, 10).Value = "R/W区分"
    Call SetHeaderStyle(ws, 10)

    '----------------------------------------------
    ' Pass1: .cobを全解析 → COBテーブル構築
    '----------------------------------------------
    Dim cobProgram()  As String  ' PROGRAM-ID（大文字）
    Dim cobFdName()   As String  ' SELECT name = FD名
    Dim cobAssignTo() As String  ' ASSIGN TO値（大文字）= JCL DDname相当
    Dim cobOpenMode() As String  ' INPUT / OUTPUT / I-O / EXTEND
    Dim cobCount      As Integer
    cobCount = 0
    ReDim cobProgram(0)
    ReDim cobFdName(0)
    ReDim cobAssignTo(0)
    ReDim cobOpenMode(0)

    Dim pgmFolder As String
    pgmFolder = GetPgmFolderPath()
    If Dir(pgmFolder, vbDirectory) = "" Then
        MsgBox "PGMフォルダが見つかりません: " & pgmFolder, vbExclamation
        Exit Sub
    End If

    Dim cobFile As String
    cobFile = Dir(pgmFolder & "*.cob")
    Do While cobFile <> ""
        Call S3_BuildCobTable(pgmFolder & cobFile, _
                               cobProgram, cobFdName, cobAssignTo, cobOpenMode, cobCount)
        cobFile = Dir()
    Loop

    '----------------------------------------------
    ' Pass2: .jclを全解析 → COB情報を紐付けてシートに出力
    '----------------------------------------------
    Dim jclFolder As String
    jclFolder = GetJclFolderPath()
    If Dir(jclFolder, vbDirectory) = "" Then
        MsgBox "JCLフォルダが見つかりません: " & jclFolder, vbExclamation
        Exit Sub
    End If

    rowIdx = 2
    Dim jclFile As String
    jclFile = Dir(jclFolder & "*.jcl")
    Do While jclFile <> ""
        Call S3_ParseJcl(jclFolder & jclFile, jclFile, ws, rowIdx, _
                          cobProgram, cobFdName, cobAssignTo, cobOpenMode, cobCount)
        jclFile = Dir()
    Loop

    ws.Columns("A:E").AutoFit
    ws.Columns("F").ColumnWidth = 12
    ws.Columns("G").ColumnWidth = 40
    ws.Columns("H").ColumnWidth = 22
    ws.Columns("I:J").AutoFit

    If rowIdx > 2 Then
        With ws.Range(ws.Cells(1, 1), ws.Cells(rowIdx - 1, 10)).Borders
            .LineStyle = xlContinuous
            .Color = RGB(180, 180, 180)
        End With
    End If

    ws.Cells(1, 1).Select

    MsgBox "Sheet3（ファイル割当一覧）完了。" & vbCrLf & _
           "件数: " & (rowIdx - 2) & " 件", vbInformation, "COBOL仕様書生成"
End Sub


'##################################################
' Pass1: COBファイル解析 → テーブル構築
'##################################################

'--------------------------------------------------
' 1つの.cobを解析してCOBテーブルに追加
' 収集対象:
'   PROGRAM-ID
'   SELECT name（= FD名）と ASSIGN TO値（= JCL DDname相当）
'   OPEN文のファイル名とモード
'--------------------------------------------------
Private Sub S3_BuildCobTable(filePath As String, _
                               ByRef cobProgram()  As String, _
                               ByRef cobFdName()   As String, _
                               ByRef cobAssignTo() As String, _
                               ByRef cobOpenMode() As String, _
                               ByRef cobCount      As Integer)

    ' このファイル内のSELECT/ASSIGN情報
    Dim selName()   As String
    Dim selAssign() As String
    Dim selCount    As Integer
    ReDim selName(0)
    ReDim selAssign(0)
    selCount = 0

    ' このファイル内のOPEN情報
    Dim opFd()   As String
    Dim opMode() As String
    Dim opCount  As Integer
    ReDim opFd(0)
    ReDim opMode(0)
    opCount = 0

    Dim programId As String
    Dim fileNum   As Integer
    Dim line      As String
    Dim upperLine As String
    programId = ""

    fileNum = FreeFile()
    On Error GoTo FileErr
    Open filePath For Input As #fileNum
    On Error GoTo 0

    Do While Not EOF(fileNum)
        Line Input #fileNum, line

        If IsCobolCommentLine(line) Then GoTo NextCobLine
        If Len(line) > 72 Then line = Left(line, 72)
        upperLine = UCase(line)

        ' PROGRAM-ID
        If programId = "" And InStr(upperLine, "PROGRAM-ID") > 0 Then
            programId = S3_ExtractProgramId(upperLine)
        End If

        ' SELECT ... ASSIGN TO（同一行に両方ある場合のみ対応）
        If InStr(upperLine, "SELECT ") > 0 And InStr(upperLine, "ASSIGN") > 0 Then
            Dim sName As String
            Dim sAsgn As String
            sName = S3_ExtractSelectName(upperLine)
            sAsgn = S3_ExtractAssignTo(upperLine)
            If sName <> "" And sAsgn <> "" Then
                ReDim Preserve selName(selCount)
                ReDim Preserve selAssign(selCount)
                selName(selCount)   = UCase(Trim(sName))
                selAssign(selCount) = UCase(Trim(sAsgn))
                selCount = selCount + 1
            End If
        End If

        ' OPEN文
        If InStr(upperLine, "OPEN ") > 0 Then
            Call S3_ParseOpen(upperLine, opFd, opMode, opCount)
        End If

        NextCobLine:
    Loop
    Close #fileNum

    ' SELECT情報 + OPEN情報 をCOBテーブルに登録
    Dim i As Integer
    For i = 0 To selCount - 1
        Dim oMode As String
        oMode = S3_LookupOpenMode(selName(i), opFd, opMode, opCount)

        ReDim Preserve cobProgram(cobCount)
        ReDim Preserve cobFdName(cobCount)
        ReDim Preserve cobAssignTo(cobCount)
        ReDim Preserve cobOpenMode(cobCount)

        cobProgram(cobCount)  = UCase(Trim(programId))
        cobFdName(cobCount)   = selName(i)
        cobAssignTo(cobCount) = selAssign(i)
        cobOpenMode(cobCount) = oMode
        cobCount = cobCount + 1
    Next i

    Exit Sub
FileErr:
    MsgBox "ファイルを開けません: " & filePath, vbExclamation
End Sub

' PROGRAM-ID抽出
Private Function S3_ExtractProgramId(upperLine As String) As String
    Dim pos    As Integer
    Dim after  As String
    Dim result As String

    pos = InStr(upperLine, "PROGRAM-ID")
    If pos = 0 Then Exit Function

    after = Trim(Mid(upperLine, pos + Len("PROGRAM-ID")))
    If Left(after, 1) = "." Then after = Trim(Mid(after, 2))
    If after = "" Then Exit Function

    result = Split(after, " ")(0)
    If InStr(result, ".") > 0 Then result = Left(result, InStr(result, ".") - 1)

    S3_ExtractProgramId = Trim(result)
End Function

' SELECT name 抽出
' 例: "       SELECT INFILE ASSIGN TO INFILE." → "INFILE"
Private Function S3_ExtractSelectName(upperLine As String) As String
    Dim pos    As Integer
    Dim after  As String
    Dim result As String

    pos = InStr(upperLine, "SELECT ")
    If pos = 0 Then Exit Function

    after = Trim(Mid(upperLine, pos + 7))
    If after = "" Then Exit Function

    result = Split(after, " ")(0)
    Do While Len(result) > 0 And _
       (Right(result, 1) = "." Or Right(result, 1) = ",")
        result = Left(result, Len(result) - 1)
    Loop

    S3_ExtractSelectName = Trim(result)
End Function

' ASSIGN TO値 抽出
' 例: "ASSIGN TO INFILE."   → "INFILE"
'      "ASSIGN TO 'INFILE'" → "INFILE"（クォート除去）
Private Function S3_ExtractAssignTo(upperLine As String) As String
    Dim pos    As Integer
    Dim after  As String
    Dim result As String

    pos = InStr(upperLine, "ASSIGN TO ")
    If pos = 0 Then Exit Function

    after = Trim(Mid(upperLine, pos + 10))
    If after = "" Then Exit Function

    result = Split(after, " ")(0)

    ' クォートを除去
    If Left(result, 1) = "'" Or Left(result, 1) = Chr(34) Then
        result = Mid(result, 2)
        Dim endQ As Integer
        endQ = InStr(result, "'")
        If endQ = 0 Then endQ = InStr(result, Chr(34))
        If endQ > 0 Then result = Left(result, endQ - 1)
    End If

    ' 末尾のピリオド/カンマを除去
    Do While Len(result) > 0 And _
       (Right(result, 1) = "." Or Right(result, 1) = ",")
        result = Left(result, Len(result) - 1)
    Loop

    S3_ExtractAssignTo = Trim(result)
End Function

' OPEN文を解析してopFd/opModeテーブルに追加
' 例: "OPEN INPUT  INFILE1 INFILE2." → 2エントリ（INPUT×2）
Private Sub S3_ParseOpen(upperLine As String, _
                           ByRef opFd()   As String, _
                           ByRef opMode() As String, _
                           ByRef opCount  As Integer)
    Dim pos   As Integer
    Dim after As String
    Dim mode  As String

    pos = InStr(upperLine, "OPEN ")
    If pos = 0 Then Exit Sub

    after = Trim(Mid(upperLine, pos + 5))

    ' モード識別
    If InStr(after, "INPUT") = 1 Then
        mode = "INPUT"
        after = Trim(Mid(after, 7))
    ElseIf InStr(after, "OUTPUT") = 1 Then
        mode = "OUTPUT"
        after = Trim(Mid(after, 8))
    ElseIf InStr(after, "I-O") = 1 Then
        mode = "I-O"
        after = Trim(Mid(after, 5))
    ElseIf InStr(after, "EXTEND") = 1 Then
        mode = "EXTEND"
        after = Trim(Mid(after, 8))
    Else
        Exit Sub
    End If

    ' モードの後に続くファイル名を1つずつ登録
    Dim parts() As String
    parts = Split(after, " ")

    Dim i As Integer
    For i = 0 To UBound(parts)
        Dim fname As String
        fname = Trim(parts(i))
        If fname = "" Then GoTo NextToken

        Do While Len(fname) > 0 And _
           (Right(fname, 1) = "." Or Right(fname, 1) = ",")
            fname = Left(fname, Len(fname) - 1)
        Loop

        If fname <> "" Then
            ReDim Preserve opFd(opCount)
            ReDim Preserve opMode(opCount)
            opFd(opCount)   = UCase(fname)
            opMode(opCount) = mode
            opCount = opCount + 1
        End If

        NextToken:
    Next i
End Sub

' FD名に対応するOPEN区分を取得
Private Function S3_LookupOpenMode(fdName As String, _
                                     opFd()   As String, _
                                     opMode() As String, _
                                     opCount  As Integer) As String
    Dim i      As Integer
    Dim result As String
    result = ""

    For i = 0 To opCount - 1
        If opFd(i) = UCase(fdName) Then
            If InStr(result, opMode(i)) = 0 Then
                If result <> "" Then result = result & "/"
                result = result & opMode(i)
            End If
        End If
    Next i

    S3_LookupOpenMode = IIf(result = "", "（OPEN文未検出）", result)
End Function


'##################################################
' Pass2: JCLファイル解析 → シート出力
'##################################################

Private Sub S3_ParseJcl(filePath As String, _
                          fileName As String, _
                          ws As Worksheet, _
                          ByRef rowIdx As Long, _
                          cobProgram()  As String, _
                          cobFdName()   As String, _
                          cobAssignTo() As String, _
                          cobOpenMode() As String, _
                          cobCount      As Integer)

    Dim fileNum  As Integer
    Dim line     As String
    Dim jobName  As String
    Dim stepName As String
    Dim pgmName  As String
    Dim no       As Long
    jobName  = ""
    stepName = ""
    pgmName  = ""
    no = 0

    fileNum = FreeFile()
    On Error GoTo FileErr
    Open filePath For Input As #fileNum
    On Error GoTo 0

    Do While Not EOF(fileNum)
        Line Input #fileNum, line

        If Left(line, 2) <> "//" Then GoTo NextJclLine
        If Left(line, 3) = "//*" Then GoTo NextJclLine
        If Len(line) >= 3 Then
            If Mid(line, 3, 1) = " " Then GoTo NextJclLine
        End If

        Dim cardName     As String
        Dim cardVerb     As String
        Dim cardOperands As String
        Call S3_ParseJclLine(line, cardName, cardVerb, cardOperands)

        Select Case UCase(cardVerb)

            Case "JOB"
                jobName = cardName

            Case "EXEC"
                stepName = cardName
                pgmName  = S3_ExtractPgmName(cardOperands)

            Case "DD"
                If cardName = "" Then GoTo NextJclLine

                Dim dsn  As String
                Dim disp As String
                dsn  = S3_ExtractDsn(cardOperands)
                disp = S3_ExtractDisp(cardOperands)

                ' COBテーブルを検索して紐付き
                ' PGMでない場合（PROC呼び出し等）はそのまま"（紐付けなし）"になる
                Dim fdName As String
                Dim rwKbn  As String
                fdName = "（紐付けなし）"
                rwKbn  = "（不明）"

                Dim j As Integer
                For j = 0 To cobCount - 1
                    If cobProgram(j) = UCase(pgmName) And _
                       cobAssignTo(j) = UCase(cardName) Then
                        fdName = cobFdName(j)
                        rwKbn  = S3_OpenModeToRW(cobOpenMode(j))
                        Exit For
                    End If
                Next j

                no = no + 1
                ws.Cells(rowIdx, 1).Value  = no
                ws.Cells(rowIdx, 2).Value  = fileName
                ws.Cells(rowIdx, 3).Value  = jobName
                ws.Cells(rowIdx, 4).Value  = stepName
                ws.Cells(rowIdx, 5).Value  = pgmName
                ws.Cells(rowIdx, 6).Value  = cardName
                ws.Cells(rowIdx, 7).Value  = dsn
                ws.Cells(rowIdx, 8).Value  = disp
                ws.Cells(rowIdx, 9).Value  = fdName
                ws.Cells(rowIdx, 10).Value = rwKbn

                ' 偶数行グレー
                If rowIdx Mod 2 = 0 Then
                    ws.Range(ws.Cells(rowIdx, 1), ws.Cells(rowIdx, 10)) _
                      .Interior.Color = RGB(242, 242, 242)
                End If

                ' 紐付けなし → FD名セルをオレンジハイライト
                If fdName = "（紐付けなし）" Then
                    ws.Cells(rowIdx, 9).Interior.Color = RGB(255, 200, 150)
                End If

                rowIdx = rowIdx + 1
        End Select

        NextJclLine:
    Loop

    Close #fileNum
    Exit Sub
FileErr:
    MsgBox "ファイルを開けません: " & filePath, vbExclamation
End Sub

' JCL1行をname/verb/operandsに分解
Private Sub S3_ParseJclLine(line As String, _
                              ByRef cardName     As String, _
                              ByRef cardVerb     As String, _
                              ByRef cardOperands As String)
    cardName = "": cardVerb = "": cardOperands = ""

    Dim content As String
    content = Mid(line, 3)

    Dim tokens() As String
    tokens = S3_SplitMultiSpace(content)

    Dim cnt As Integer
    cnt = UBound(tokens) + 1
    If cnt = 0 Then Exit Sub
    If cnt >= 1 Then cardName = tokens(0)
    If cnt >= 2 Then cardVerb = tokens(1)

    If cnt >= 3 Then
        Dim pos As Long
        pos = 1
        pos = pos + Len(cardName)
        Do While pos <= Len(content) And Mid(content, pos, 1) = " "
            pos = pos + 1
        Loop
        pos = pos + Len(cardVerb)
        Do While pos <= Len(content) And Mid(content, pos, 1) = " "
            pos = pos + 1
        Loop
        cardOperands = Trim(Mid(content, pos))
    End If
End Sub

' 複数スペース分割
Private Function S3_SplitMultiSpace(str As String) As String()
    Dim parts()  As String
    Dim result() As String
    Dim count    As Integer
    Dim i        As Integer

    parts = Split(str, " ")
    count = 0
    ReDim result(UBound(parts))
    For i = 0 To UBound(parts)
        If parts(i) <> "" Then
            result(count) = parts(i)
            count = count + 1
        End If
    Next i
    If count = 0 Then
        ReDim result(-1)
    Else
        ReDim Preserve result(count - 1)
    End If
    S3_SplitMultiSpace = result
End Function

' EXECカードから実行名を取得
'   PGM=PROG01   → "PROG01"
'   PROC=MYPROC  → "[PROC:MYPROC]"
'   MYPROC       → "[PROC:MYPROC]"（暗黙的PROC）
Private Function S3_ExtractPgmName(operands As String) As String
    Dim upperOp As String
    upperOp = UCase(Trim(operands))

    If upperOp = "" Then
        S3_ExtractPgmName = "（不明）"
        Exit Function
    End If

    Dim pos    As Integer
    Dim after  As String
    Dim result As String

    ' PGM= を優先検索
    pos = InStr(upperOp, "PGM=")
    If pos > 0 Then
        after = Mid(operands, pos + 4)
        pos = InStr(after, ",")
        If pos > 0 Then
            result = Left(after, pos - 1)
        Else
            result = Trim(Split(after, " ")(0))
        End If
        S3_ExtractPgmName = UCase(Trim(result))
        Exit Function
    End If

    ' PROC= または暗黙的PROC
    pos = InStr(upperOp, "PROC=")
    If pos > 0 Then
        after = Mid(operands, pos + 5)
    Else
        after = operands  ' 暗黙的PROC
    End If

    Dim tokens() As String
    tokens = S3_SplitMultiSpace(after)
    If UBound(tokens) >= 0 Then
        result = tokens(0)
        pos = InStr(result, ",")
        If pos > 0 Then result = Left(result, pos - 1)
        S3_ExtractPgmName = "[PROC:" & UCase(Trim(result)) & "]"
    Else
        S3_ExtractPgmName = "（不明）"
    End If
End Function

' DDカードからDSN値を取得
Private Function S3_ExtractDsn(operands As String) As String
    Dim upperOp As String
    upperOp = UCase(operands)

    Dim pos    As Integer
    Dim keyLen As Integer

    pos = InStr(upperOp, "DSN=")
    If pos > 0 Then
        keyLen = 4
    Else
        pos = InStr(upperOp, "DSNAME=")
        If pos > 0 Then
            keyLen = 7
        Else
            S3_ExtractDsn = "（DSNなし）"
            Exit Function
        End If
    End If

    Dim after As String
    after = Mid(operands, pos + keyLen)

    Dim commaPos As Integer
    commaPos = InStr(after, ",")
    If commaPos > 0 Then
        S3_ExtractDsn = Trim(Left(after, commaPos - 1))
    Else
        S3_ExtractDsn = Trim(Split(after, " ")(0))
    End If
End Function

' DDカードからDISP値を取得（括弧の深さを追跡）
Private Function S3_ExtractDisp(operands As String) As String
    Dim upperOp As String
    upperOp = UCase(operands)

    Dim pos As Integer
    pos = InStr(upperOp, "DISP=")
    If pos = 0 Then
        S3_ExtractDisp = "（DISPなし）"
        Exit Function
    End If

    Dim after As String
    after = Mid(operands, pos + 5)

    If Left(after, 1) = "(" Then
        Dim depth As Integer
        Dim i     As Integer
        Dim c     As String
        depth = 0
        For i = 1 To Len(after)
            c = Mid(after, i, 1)
            If c = "(" Then depth = depth + 1
            If c = ")" Then
                depth = depth - 1
                If depth = 0 Then
                    S3_ExtractDisp = Left(after, i)
                    Exit Function
                End If
            End If
        Next i
        S3_ExtractDisp = after
    Else
        Dim commaPos As Integer
        commaPos = InStr(after, ",")
        If commaPos > 0 Then
            S3_ExtractDisp = Left(after, commaPos - 1)
        Else
            S3_ExtractDisp = Trim(Split(after, " ")(0))
        End If
    End If
End Function

' OPEN区分 → R/W区分に変換
Private Function S3_OpenModeToRW(openMode As String) As String
    Select Case UCase(Trim(openMode))
        Case "INPUT"          : S3_OpenModeToRW = "R（入力）"
        Case "OUTPUT"         : S3_OpenModeToRW = "W（出力）"
        Case "I-O"            : S3_OpenModeToRW = "R/W（入出力）"
        Case "EXTEND"         : S3_OpenModeToRW = "W（追記）"
        Case "（OPEN文未検出）" : S3_OpenModeToRW = "（不明）"
        Case Else             : S3_OpenModeToRW = openMode
    End Select
End Function

既知の限界
ケース挙動SELECT と ASSIGN TO が別行にあるASSIGN TO未取得 → 紐付けなしDSN/DISP が継続行にある未取得PROC呼び出しのDD[PROC:名前] と表示され紐付けなしASSIGN TO に環境固有プレフィックスあり（例：DA-INFILE）DDname不一致で紐付けなし
オレンジセルが多い場合は上記のどれかに該当します。動かして確認してみてください。
